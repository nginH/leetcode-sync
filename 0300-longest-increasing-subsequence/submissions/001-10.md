class Solution {


int sol(vector<int>&nums, int idx, int lastIdx, vector<vector<int>>&dp){

    if(idx>=nums.size()){
        return 0;
    }
   
    if (dp[idx][lastIdx + 1] != -1) {
        return dp[idx][lastIdx + 1];
    }
    int take=0;
    int skip=sol(nums, idx+1, lastIdx, dp);
     if (lastIdx == -1 || nums[idx] > nums[lastIdx]) { // bada hai tabli allowd ..
        take=1+sol(nums, idx+1,idx,  dp);
    }
     return dp[idx][lastIdx + 1] = max(take, skip);


    
}


public:
    int lengthOfLIS(vector<int>& nums) {
        vector<vector<int>>dp(nums.size()+1, vector<int>(nums.size()+1,  -1));
        return sol(nums, 0, -1, dp);
    }
};
