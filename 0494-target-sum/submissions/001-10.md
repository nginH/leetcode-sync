class Solution {

int dfs(vector<int>& nums, int target, int idx, int curr,  vector<vector<int>>& pos, vector<vector<int>>& neg){


    if(idx>=nums.size()){
        if(target==curr){
            return 1;
        }
        return 0;
    }
      // Positive curr
    if (curr >= 0 && pos[idx][curr] != -1) {
            return pos[idx][curr];
    }
        // Negative curr
    if (curr < 0 && neg[idx][-curr] != -1) {
            return neg[idx][-curr];
    }
  

    // addd..
    int add= dfs(nums, target, idx+1,curr+nums[idx], pos, neg);
    // or su=b
    int sub = dfs(nums, target, idx+1, curr-nums[idx], pos, neg);
    int result = add+sub;
    if (curr >= 0)
            pos[idx][curr] = result;
        else
            neg[idx][-curr] = result;

    return result;
}

public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int sum = accumulate(nums.begin(), nums.end(), 0);
        vector<vector<int>> pos(
            nums.size(),
            vector<int>(sum + 1, -1)
        );

        vector<vector<int>> neg(
            nums.size(),
            vector<int>(sum + 1, -1)
        );

        return dfs(nums, target , 0, 0, pos, neg);
    }
};
