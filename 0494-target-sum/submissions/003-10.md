class Solution {
int soln=0;

void dfs(vector<int>&nums, int k, int idx){

    
    if(idx==nums.size()){
        if(k==0){
            soln++;
        }
        return;
    }
    
    dfs(nums, k-nums[idx], idx+1);
    dfs(nums, k+nums[idx], idx+1);
}


public:
    int findTargetSumWays(vector<int>& nums, int target) {
        dfs(nums, target, 0);
        return soln;
    }
};
