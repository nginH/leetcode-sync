class Solution {

int dfs(vector<int>& nums, int target, int idx, int curr){


    if(idx>=nums.size()){
        if(target==curr){
            return 1;
        }
        return 0;
    }

    // addd..
    int add= dfs(nums, target, idx+1,curr+nums[idx]);
    // or su=b
    int sub = dfs(nums, target, idx+1, curr-nums[idx]);
    return add+sub;
}

public:
    int findTargetSumWays(vector<int>& nums, int target) {
        return dfs(nums, target , 0, 0);
    }
};
