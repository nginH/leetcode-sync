class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        

        int ans=INT_MAX;
        int left=0;
        int sum=0;
        for(int i=0; i<nums.size(); i++){
            sum+=nums[i];
            while(sum>=target){ // move left cursor
                ans=min(ans, (i-left)+1);
                sum-=nums[left];
                left++;
            }
        }
        return ans==INT_MAX?0:ans;
    }
};
