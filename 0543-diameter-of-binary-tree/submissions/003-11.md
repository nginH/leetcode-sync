/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:

    int maxCalc(TreeNode* node){

        if(!node){
            return 0;
        }
        int a=maxCalc(node->left)+1;
        int b=maxCalc(node->right)+1;
        return max(a,b);
    }

    int diameterOfBinaryTree(TreeNode* root) {
        if(!root){
            return 0;
        }
        return maxCalc(root->left)+maxCalc(root->right);
    }
};
