class Solution {
public:
    int ladderLength(string beginWord, string endWord,
                     vector<string>& wordList) {
        unordered_set<string> st(wordList.begin(), wordList.end());
        queue<string> q;
        q.push(beginWord);

        unordered_map<string, int> mp;

        int level = 0;
        while (!q.empty()) {
            int sz = q.size();
            level++;
            while (sz--) {
                string curr = q.front();
                q.pop();
                //
                if (curr == endWord) {// word matched
                    return level;
                }
                // i want to al
                string wd = "";
                for (int j = 0; j < curr.size(); j++) {
                    wd = curr;
                    wd[j] = 'a';
                    for (int i = 0; i < 26; i++) {
                        wd[j] ='a'+i; // hit-> ait, bit ,cit ....hat, hbt
                        if (wd != curr) {
                            if (st.contains(wd) && !mp.contains(wd)) {
                                mp[wd]++;   // visited..
                                q.push(wd); // for next explore
                            }
                        }
                    }
                }
            }
        }
        return 0;
    }
};
