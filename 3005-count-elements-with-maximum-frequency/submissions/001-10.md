class Solution {
public:
    int maxFrequencyElements(vector<int>& nums) {
        unordered_map<int , int> m;
        int maxfreq=0;
        for (auto i:nums){
            m[i]++;
            maxfreq=max(maxfreq, m[i]);
        }


        int ans=0;
        for (auto& p : m) {
            if (p.second == maxfreq) {
                ans += p.second;
            }
        }
        return ans;
    }
};
