        {1,0},

        {0,1},
        {0,-1}
    };

    void dfs(vector<vector<char>>& grid, int i, int j){

        if(i<0 || j<0 || i>=n || j>=m || grid[i][j]=='0'){
            return;
        }
        grid[i][j]='0';
      
        dfs(grid, i+1, j);
         dfs(grid, i+1, j);
          dfs(grid, i+1, j);

    }


public:
    int numIslands(vector<vector<char>>& grid) {
         n=grid.size();
         m=grid[0].size();
        int ans=0;


        for(int i=0; i< n; i++){
            for(int j=0; j<m ; j++){
                if(grid[i][j]=='1'){
                    ans++;
                    dfs(grid,i, j);
                }
            }
        }   

        return ans;
    }
};
