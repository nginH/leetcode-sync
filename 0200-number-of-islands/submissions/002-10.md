class Solution {

private: 
    vector<pair<int, int>> dir={
        {0,1},
        {1,0},
        {-1,0},
        {0,-1}
    };
    int ans=0;
    int m,n;
    void dfs(vector<vector<char>>& grid, int row, int col){
        if(row<0 || col<0 || row>=m || col>=n){
            return ; // end 
        }
        if(grid[row][col]=='#' || grid[row][col]=='0'){
            return;
        }
        grid[row][col]='#'; // mark as visited..
        for(auto &[dx, dy]:dir){
            dfs(grid, row+dx, col+dy);
        }
    }   




public:
    int numIslands(vector<vector<char>>& grid) {
        m=grid.size();
        n=grid[0].size();
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(grid[i][j]!='#')
                    if(grid[i][j]=='1'){
                        dfs(grid, i, j); // update all the cell
                        ans++; // 
                    }
            }
        }
        return ans;
    }
};
