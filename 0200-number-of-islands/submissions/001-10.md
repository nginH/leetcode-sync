class Solution {

    vector<pair<int, int>> dir = {{1, 0}, {0, 1}, {0, -1}, {-1, 0}};
    int m, n;

    void dfs(vector<vector<char>>& grid, int r, int c) {
        if(r>=m ||c>=n || r<0 || c<0 ){ // out of bound check!
            return ;
        }

        if(grid[r][c]=='0'|| grid[r][c]=='2'){
            return; // 0 mean water 2 mean vistited;
        }
        grid[r][c]='2';
        for(auto &[dx, dy]:dir){
            dfs(grid, dx+r, dy+c);
        }
    }

public:
    int numIslands(vector<vector<char>>& grid) {
        int ans=0;
        m=grid.size();
        n=grid[0].size();

        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(grid[i][j]=='1'){
                    ans++;
                    dfs(grid, i, j);
                }
            }
        }


        return ans;


    }
};
