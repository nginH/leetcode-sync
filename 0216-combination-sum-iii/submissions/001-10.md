class Solution {

vector<vector<int>> ans;
vector<int>subset;

void dfs(int sz, int idx, int target){
  if (subset.size() == sz) {
    if (target == 0)
        ans.push_back(subset);
    return;
   }

    if(target<0){
        return;
    }

    for(int i=idx; i<=9;i++ ){
        
        if(i>target){
            break;
        }

        subset.push_back(i);
        dfs(sz,i+1, target-i);
        subset.pop_back();
        
    

    }
}



public:
    vector<vector<int>> combinationSum3(int k, int n) {
        dfs(k, 1, n);
        return ans;

    }
};
