class Solution {

vector<vector<int>> ans;
vector<int>subset;

void dfs(vector<int>&arr,int sz, int idx, int target){
    if(target==0){
        if(subset.size()==sz)
           ans.push_back(subset);
        return;
    }

    if(arr.size()==idx || target<0){
        return;
    }

    for(int i=idx; i<arr.size();i++ ){

        if(i>idx && arr[i]==arr[i-1]) continue;

        subset.push_back(arr[i]);
        dfs(arr, sz,i+1, target-arr[i]);
        subset.pop_back();
        
    

    }
}



public:
    vector<vector<int>> combinationSum3(int k, int n) {
        vector<int> arr={1,2,3,4,5,6,7,8,9};
        dfs(arr, k, 0, n);
        return ans;

    }
};
