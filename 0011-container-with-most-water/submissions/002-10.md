#include<bits/stdc++.h>

class Solution {
public:
    int maxArea(vector<int>& h) {
       int ans=0;
       int i=0, j=h.size()-1;
        while(i<=j){
            int hg=min( h[i] , h[j] );
            int b=abs(j-i);
            ans=max(ans, (hg*b));

            h[i]>h[j]?j--:i++;
       }
       return ans;
    }
};
