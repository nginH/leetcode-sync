#include<bits/stdc++.h>

class Solution {
public:
    int maxArea(vector<int>& arr) {
        int i=0;
        int ans=0;
        int j=arr.size()-1;

        while(i<j){
            int h=min(arr[i], arr[j]);
            int len=j-i;
            ans=max(ans, h*len);

            if(arr[i]<arr[j]){
                i++;
            }else{
                j--;
            }
        }
        return ans;
    }
};
