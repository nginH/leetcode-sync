class Solution {
public:
    int maxArea(vector<int>& h) {
        int ans=0;
        for(int i=0; i<h.size(); i++){
            for(int j=0; j<=i;j++){
                if(h[i]<j){
                    continue;
                }
                ans= max(ans, j*h[i]);
            }
        }
        return ans;
    }
};
