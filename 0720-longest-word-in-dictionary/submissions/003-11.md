class Solution {
public:
    string longestWord(vector<string>& words) {
        sort(words.begin(), words.end());
        unordered_set<string> st;
        priority_queue<pair<int, string>>pq;

        for(auto word:words){
            string temp="";
            int countFound=0;
            for(int i=0; i<word.size(); i++){
                temp+=word[i];
                if(!st.empty() && st.contains(temp)){
                    countFound++;
                }else{
                    break;
                }
            }
            st.insert(word);
            pq.push({countFound, word});
        }
        vector<string> ans;
        int maxOccur = pq.top().first;
        while (!pq.empty()) {
            int n = pq.top().first;
            if (n == maxOccur) {
                ans.push_back(pq.top().second);
            } 
            else {
                break;
            }
            pq.pop();
        };
        sort(ans.begin(), ans.end());
        return ans[0];
    }
};
