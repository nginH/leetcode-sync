class Solution {
public:
    string longestWord(vector<string>& words) {

        sort(words.begin(), words.end());

        unordered_set<string> st;
        priority_queue<pair<int, string>> pq;

        for (auto word : words) {
            string temp = "";
            int countFound = 0;
            bool valid = true;
            for (int i = 0; i < word.size() - 1; i++) {
                temp += word[i];
                if (st.contains(temp)) {
                    countFound++;
                }
                else {
                    valid = false;
                    break;
                }
            }
            st.insert(word);
            if (valid) {
                pq.push({countFound, word});
            }
        }
        if (pq.empty()) {
            return "";
        }

        vector<string> ans;
        int maxOccur = pq.top().first;
        while (!pq.empty()) {
            int n = pq.top().first;
            if (n == maxOccur) {
                ans.push_back(pq.top().second);
            }
            else {
                break;
            }
            pq.pop();
        }
        sort(ans.begin(), ans.end());
        return ans[0];
    }
};
