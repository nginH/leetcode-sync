class Solution {
public:
    bool isValid(string s) {
        stack<char> st;

        for(auto x:s){
            if(!st.empty()){
             char c= st.top();

            if(
                    (c=='(' && x==')' ) ||
                    (c=='{' && x=='}' ) ||
                    (c=='[' && x==']' )
            ){
                st.pop();
            }else{
                    st.push(x);
            }
            }else{
                st.push(x);
            }
        }

        return st.empty() ? true:false;
    }
};
