class Solution {
public:
    bool isValid(string s) {
        stack<char> st;
        int i=0;
        int n=s.size();
        while(i<n){
            char c=s[i];

            switch (c) {
                case '(':
                case '{':
                case '[':
                    st.push(c);
                    break;
                case ')':
                    if(!st.empty() && st.top()=='(')
                        st.pop();
                    else{
                        st.push(')');
                    }
                    break;

                case '}':
                    if(!st.empty() && st.top()=='{')
                        st.pop();
                    else{
                        st.push('}');
                    }

                    break;

                case ']':
                    if(!st.empty() && st.top()=='[')
                        st.pop();
                    else{
                        st.push(']');
                    }
                    break;
            }
            i++;
        }

        return !st.size();

    }
};
