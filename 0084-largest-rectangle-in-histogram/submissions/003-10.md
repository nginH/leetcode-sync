class Solution {
public:
    int largestRectangleArea(vector<int>& arr) {
        int ans=0;

        int n=arr.size();
        vector<int> pre(n), post(n);
        stack<int>st;
        
        for(int i=0; i<n; i++){
            while(!st.empty() && arr[st.top()] >=arr[i]){
                st.pop();
            }
            if(st.empty()){
                pre[i]=-1;
            }else{
                pre[i]=st.top();
            }
            st.push(i);
        }
        while(!st.empty())st.pop();

        for(int i=n-1; i>=0; i--){
            while(!st.empty()&& arr[st.top()]>=arr[i]){
                st.pop();
            }
            if(st.empty()){
                post[i] = n;
            }else{
                post[i]=st.top();
            }
            st.push(i);
        }

        for(int i=0; i<n; i++){
            ans=max(ans, arr[i]*(post[i]-pre[i]-1));
        }



        return ans;
    }
};
