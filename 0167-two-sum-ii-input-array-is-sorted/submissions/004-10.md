class Solution {


int bsearch(vector<int>& numbers, int target, int startIdx){

        int i=startIdx;
        int j=numbers.size()-1;

        while(i<=j){
            int mid=(i+j)/2;
            if(numbers[mid]==target){
                return mid;
            }else if(numbers[mid]<target){// mid is less then target//
                i=mid+1;
            }else{
                j=mid-1;
            }
        }
        return -1;
}



public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        int i=0;
        for(auto x:numbers){
            int raim=target-x;
            int searchResult=bsearch(numbers, raim,i+1);
            if(searchResult!=-1){
                return {i+1, searchResult+1};
            }
            i++;
        }
        return{};// as exactly one soln..
    }
};
