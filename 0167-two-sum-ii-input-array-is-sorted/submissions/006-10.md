class Solution {
public:
    vector<int> twoSum(vector<int>& numb, int target) {

        unordered_map<int, int> m;
        for(int i=0; i< numb.size(); i++){
          int cp=target-numb[i];
          
          if(m[numb[i]]>0){
            return { m[numb[i]],i+1};
          }  
          m[cp]=i+1;
        }

        return {};
    }

};
