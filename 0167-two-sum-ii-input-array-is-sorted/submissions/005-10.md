class Solution {
public:
    vector<int> twoSum(vector<int>& numb, int target) {

        bool found=false;
        bool ck=true;
        int a;
        for(int i=0; i< numb.size(); i++){
            int idx= bsearch(i+1, numb, target-numb[i]);
            if(idx!=-1){
                return{i+1, idx+1};
            }   
        }

        return {};
    }

    int bsearch(int start,  vector<int> & v, int t){
        int end= v.size();
        while(start<end){
            int mid= (start+end)/2;
            if(v[mid]==t){
                return mid;
            }else if(t-v[mid]>0){
                start=mid+1;
            }else{
                end=mid;
            }
        }
     return -1;
    }
};
