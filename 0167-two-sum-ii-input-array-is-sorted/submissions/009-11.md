class Solution {
public:
    vector<int> twoSum(vector<int>& numb, int target) {

        bool found=false;
        bool ck=true;
        int a;
        for(int i=0; i< numb.size(); i++){
          
            if(ck){
                found= bsearch(i+1, numb, target-numb[i]);
                a=i;
            }
            if(found){
                if(numb[i]==target-numb[a]){
                    return {a+1,i+1};
                }
                ck=false;
            }   
        }

        return {};
    }

    int bsearch(int start,  vector<int> & v, int t){
        int end= v.size();
        while(start<end){
            int mid= (start+end)/2;
            if(v[mid]==t){
                return true;
            }else if(t-v[mid]>0){
                start=mid+1;
            }else{
                end=mid;
            }
        }
     return false;
    }
};
