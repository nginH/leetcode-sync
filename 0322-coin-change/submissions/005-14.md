class Solution {

int sol(vector<int>& coins, int amount){

    if(amount==0){
        return 0;
    }
    if(amount<0){
        return INT_MAX;
    }

    int minAns=INT_MAX;
    for(auto x:coins){
        int res=sol(coins, amount-x);
        if(res!=INT_MAX){
            minAns=min(minAns, res+1);
        }
    }
    
    return minAns;
}


public:
    int coinChange(vector<int>& coins, int amount) {
        int ans= sol(coins, amount);
       return ans==INT_MAX?-1:ans;
       
    }
};
