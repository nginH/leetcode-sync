class Solution {

vector<int>dp;
int sol(vector<int>& coins, int amount){

    if(amount==0){
        return 0;
    }
    if(amount<0){
        return INT_MAX;
    } 
    if(dp[amount]!=-1){
        return dp[amount];
    }

    int minAns=INT_MAX;
    for(auto x:coins){
        int res=sol(coins, amount-x);
        if(res!=INT_MAX){
            minAns=min(minAns, res+1);
        }
    }
    
    return dp[amount]=minAns;
}


public:
    int coinChange(vector<int>& coins, int amount) {
        dp.resize(amount+1, -1);
        int ans= sol(coins, amount);
       return ans==INT_MAX?-1:ans;
       
    }
};
