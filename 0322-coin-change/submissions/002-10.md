class Solution {

vector<vector<int>>dp;

int sol(vector<int>& coins, int amount, int idx){

    if(amount==0){
        return 0;
    }
    if (amount < 0 || idx >= (int)coins.size()) return INT_MAX;
    if(dp[amount][idx]!=-1){
        return dp[amount][idx];
    }
    int skip=sol(coins, amount, idx+1);
    int take=sol(coins, amount-coins[idx], idx);
    return dp[amount][idx]= min(skip, take==INT_MAX?INT_MAX:take+1);
}


public:
    int coinChange(vector<int>& coins, int amount) {
        dp.resize(amount+1, vector<int>(coins.size()+1, -1));
        int ans= sol(coins, amount,0);
       return ans==INT_MAX?-1:ans;
       
    }
};
