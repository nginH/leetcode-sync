class Solution {

    int count(int i){
        int cnt=0;
        while(i){
            i&=i-1;
            cnt++;
        }
        return cnt;
    }
public:
    vector<int> countBits(int n) {
        vector<int> ans;

        for(int i=0; i<=n ; i++){
            ans.push_back(count(i));
        }
        return ans;
    }
};
