/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
    struct comp{
        bool operator()(ListNode*a, ListNode*b){
            return a->val>b->val;
        }
    };
public:
    ListNode* mergeKLists(vector<ListNode*>& arr) {
       priority_queue<ListNode*, vector<ListNode*>, comp> pt;

        for(auto i:arr){
            if(i)pt.push(i); // constucting priorty que
        }
        ListNode dummy;
        ListNode*tail=&dummy;
        while(!pt.empty()){
            tail->next=pt.top();
            pt.pop();
            tail=tail->next;
            if(tail->next){
                pt.push(tail->next);
            }
            
        };
        tail->next=nullptr;

        return dummy.next;
    }
};
