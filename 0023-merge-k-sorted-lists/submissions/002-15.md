/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {

private:
ListNode* merge2(ListNode* l1, ListNode *l2){
     ListNode dummy(0);
    ListNode* tail = &dummy;

    while(l1 && l2){
        if(l1->val >l2->val){
            tail->next=l2;
            l2=l2->next;
        }else{
            tail->next=l1;
            l1=l1->next;
        }
        tail=tail->next;
    }
    if(l1){
        tail->next=l1;
    }
    if(l2){
        tail->next=l2;
    }
    return dummy.next;
}

void dnq(vector<ListNode*>& lists, int l ,int h){
    if(l<h){
        int mid= (l+h)/2;

        dnq(lists, l, mid);
        dnq(lists, mid+1, h);
       lists[l]= merge2(lists[l], lists[mid+1]);
    }
    
}

public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {  
        dnq(lists, 0, lists.size()-1);
        return lists[0];  
    }
};
