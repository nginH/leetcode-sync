class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        
        int row=0, col=0;
        int n=matrix.size();
        int m=matrix[0].size();

        while(row<n and col+1 <m){

            if(matrix[row][col]==target){
                return true;
            }else if(matrix[row][col+1]>target){
                // if(matrix[row][col+1]==target){
                //     return true;
                // }
                row+=1;
            }else{
                col+=1;
            }
        }
        return false;
    }
};
