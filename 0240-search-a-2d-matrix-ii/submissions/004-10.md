class Solution {
public:
    bool searchMatrix(vector<vector<int>>& arr, int k) {

            int row= arr.size();
            int col=arr[0].size();

            int r=0;
            int c=col-1;

            while(r<row && c>=0){

                if(arr[r][c]==k){
                    return true;
                }else if(arr[r][c]>k){
                    c--;
                }else{
                    r++;
                }


            }
            return false;
    }
};
