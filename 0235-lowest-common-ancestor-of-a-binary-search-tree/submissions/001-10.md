/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */

class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        
        if(!root){
            return root;
        }

        while(root){
            if(root->val > p->val  && root->val > q->val){
                root=root->left; // dono chota move left
            }else if( root->val < p->val  && root->val < q->val ){ // p,q bada then root. more right
                root=root->right;
            }else{
                return root;
            }
        }

        return nullptr;
        
    }
};
