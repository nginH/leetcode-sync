class Solution {
public:
    vector<int> addToArrayForm(vector<int>& nums, int k) {
        vector<int>ans;
        vector<int>kbreak;

        while(k){
           kbreak.push_back(k%10);
           k/=10; 
        }
        reverse(kbreak.begin(), kbreak.end());
        int i=nums.size()-1;
        int j=kbreak.size()-1;
        
        int carry=0;

        while(i>=0 || j>=0 || carry){

            if(i>=0){
                carry+=nums[i];    
                i--;
            }
            if(j>=0){
                carry+=kbreak[j];
                j--;
            }

            ans.push_back(carry%10);
            carry/=10;
        }
        reverse(ans.begin(), ans.end());
        return ans;


    }
};
