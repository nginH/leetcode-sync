/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    bool isCousins(TreeNode* root, int x, int y) {
        // bfs//
        queue<TreeNode*>q;
        q.push(root);
        bool isXfound=false;
        bool isYfound=false;
        while(!q.empty()){ // O(n)
            int sz=q.size();
            while(sz--){
                auto node=q.front();
                q.pop();

                if (node->left && node->right) { // sibling
                    if ((node->left->val == x && node->right->val == y) ||
                        (node->left->val == y && node->right->val == x)) {
                        return false;
                    }
                }
                if(node->val==x){
                    isXfound=true;
                }
                if(node->val==y){
                    isYfound=true;
                }
                if(node->right){
                    q.push(node->right);
                }
                if(node->left){
                    q.push(node->left);
                }
            }
            if(isYfound && isXfound){
                return true;
            }
            else if(isXfound || isYfound){ // if any of two is toggle return false 
                return false;
            }

        }
        return false;


    }
};
