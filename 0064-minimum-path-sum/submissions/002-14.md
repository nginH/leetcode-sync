class Solution {
private:
int m;
int n;
    vector<pair<int, int>> dir{
        {1, 0},
        {0, 1}
    };
    int sol(int row, int col, vector<vector<int>>& grid){
        if (row >= m || col >= n) {
            return INT_MAX;
        }

        if (row == m-1 && col == n-1) {
           return grid[row][col];
        }

        int curr=grid[row][col];

        int upRes=sol(row+1, col, grid);
        int dowRes=sol(row, col+1, grid);

       return curr+ min(upRes,dowRes);
    }


public:
    int minPathSum(vector<vector<int>>& grid) {
         m= grid.size();
         n=grid[0].size();
        return sol(0,0,grid);
    }
};
