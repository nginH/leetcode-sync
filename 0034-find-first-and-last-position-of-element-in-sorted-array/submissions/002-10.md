class Solution {


int lb(vector<int>& nums, int target){
    int low=0;
    int high=nums.size()-1;
    int ans=nums.size();
    while(low<=high){
        int mid=low+(high-low)/2;
        if(nums[mid]>=target){
            // may be in left lower
            ans=mid;
            high=mid-1;
        }else{
            low=mid+1;
        }
    }
    return ans;
}
int ub(vector<int>& nums, int target){
    int low=0;
    int high=nums.size()-1;
    int ans=nums.size();
    while(low<=high){
        int mid=low+(high-low)/2;
        if(nums[mid]>target){
            // may be in left lower
            ans=mid;
            high=mid-1;
        }else{
            low=mid+1;
        }
    }
    return ans;
}


public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int a=lb(nums, target);
        int b=ub(nums, target)-1;
        if (a == nums.size() || nums[a] != target)
            return {-1, -1};
        return{a, b};
    }   
};
