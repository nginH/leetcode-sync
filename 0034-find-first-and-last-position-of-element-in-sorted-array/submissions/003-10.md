class Solution {
public:

    int lbs(vector<int>&arr, int target){
       int low=0, high=arr.size()-1, lb=arr.size()+1;
        while(low<=high){
            int mid=(low+high)/2;
            if(arr[mid]==target){
                lb=min(lb, mid);
                high=mid-1;
            }else if(arr[mid]>target){
                high=mid-1;
            }else{
                low=mid+1;
            }
        } 
        return lb==arr.size()+1?-1:lb;
    }

    int ubs(vector<int>&arr, int target){
       int low=0, high=arr.size()-1, ub=-1;
        while(low<=high){
            int mid=(low+high)/2;
            if(arr[mid]==target){
                ub=max(ub, mid);
                low=mid+1;
            }else if(arr[mid]>target){
                high=mid-1;
            }else{
                low=mid+1;
            }
        } 
        return ub;
    }


    vector<int> searchRange(vector<int>& arr, int target) {

        int l=lbs(arr, target);
        int u=ubs(arr, target);
        return {l, u};
    }
};
