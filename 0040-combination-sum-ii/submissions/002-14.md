class Solution {
set<vector<int>>st;

void solve(vector<int>& candidates, int target, vector<int>&curr, int idx){

    if(target==0){
        st.insert(curr);
        return;
    }

    if(idx>=candidates.size() ||target<0){
        return;
    }

    curr.push_back(candidates[idx]);
    solve(candidates, target-candidates[idx],curr, idx+1); // also increment by one ..
    curr.pop_back();
    solve(candidates,target, curr, idx+1);
}



public:
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        vector<int>curr;
        sort(candidates.begin(), candidates.end()); // sorting preser 
        solve(candidates, target, curr, 0);
        vector<vector<int>> ans= {st.begin(), st.end()};
        return ans;
    }
};
