class DSU {
    vector<int> parent, size;
    public:
    DSU(int n ){
        parent.resize(n);
        size.resize(n, 1);
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int n){
        if(parent[n]==n){
            return n;
        }
        return parent[n]=find(parent[n]);
    }
    bool unite(int a, int b){
        a=find(a);
        b=find(b);
        if(a==b){
            return true;
        }

        if(size[a]>size[b]){
            parent[b]=a;
            size[a]+=size[b];
        }else{
            parent[a]=b;
            size[b]+=size[a];
        }
        return false;
    }

};



class Solution {
public:
    int removeStones(vector<vector<int>>& stones) {
        const int OFFSET = 10001;
        int n=stones.size();
        DSU dsu(20002);
        unordered_set<int>nodes;
        for(auto x:stones){
            dsu.unite(x[0], x[1]+OFFSET);
            nodes.insert(x[0]);
            nodes.insert(x[1]+OFFSET);
        }

        unordered_set<int>st; // parent list..

        for(auto node:nodes){
            st.insert(dsu.find(node));
        };
        // 2 group.2 parents  1 parents . 
        return stones.size()-st.size(); // total nodes - total groups .. 

    }
};
