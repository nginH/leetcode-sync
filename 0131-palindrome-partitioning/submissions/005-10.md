class Solution {

bool validPalindron(string &s, int i, int j){
    if(i>j){
        return false;
    }
    while(i<=j){
        if(s[i]!=s[j]){
            return false;
        }
        i++;
        j--;
    }
    return true;
}

vector<vector<string>> ans;

void  sol(string &s, int idx, vector<string> &curr ){

    if(idx==s.size()){
        ans.push_back(curr);
        return;
    }
    
    for(int i=idx; i<s.size(); i++){
        if(validPalindron(s, idx, i)){
            curr.push_back(s.substr(idx,  i-idx+1));
            sol(s, i+1, curr);
            curr.pop_back();
        }
    }
}

public:
    vector<vector<string>> partition(string s) {
        vector<string> curr;
        sol(s, 0,curr);
        return ans;
    }
};
