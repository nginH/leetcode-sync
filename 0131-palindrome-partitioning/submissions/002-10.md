class Solution {

int dp[20][20];

bool validPalindrome(string &s, int i, int j) {

    if (i >= j)
        return true;

    if (dp[i][j] != -1)
        return dp[i][j];

    if (s[i] != s[j])
        return dp[i][j] = false;

    return dp[i][j] = validPalindrome(s, i + 1, j - 1);
}

vector<vector<string>> ans;

void  sol(string &s, int idx, vector<string> &curr ){

    if(idx==s.size()){
        ans.push_back(curr);
        return;
    }
    
    for(int i=idx; i<s.size(); i++){
        if(validPalindrome(s, idx, i)){
            curr.push_back(s.substr(idx,  i-idx+1));
            sol(s, i+1, curr);
            curr.pop_back();
        }
    }
}

public:
    vector<vector<string>> partition(string s) {
        memset(dp, -1, sizeof(dp));
        vector<string> curr;
        sol(s, 0,curr);
        return ans;
    }
};
