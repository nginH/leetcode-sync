```cpp
class Solution {

int dp[20][20];

bool validPalindron(string &s, int i, int j){
    if(i>j){
        return false;
    }
    if(dp[i+1][j+1]!=-1){ // cache hit.
        return dp[i][j];
    }
    while(i<=j){
        if(s[i]!=s[j]){
            return dp[i+1][j+1]=false;
        }
        i++;
        j--;
    }
    return  dp[i+1][j+1]=true;
}

vector<vector<string>> ans;

void  sol(string &s, int idx, vector<string> &curr ){

    if(idx==s.size()){
        ans.push_back(curr);
        return;
    }
    
    for(int i=idx; i<s.size(); i++){
        if(validPalindron(s, idx, i)){
            curr.push_back(s.substr(idx,  i-idx+1));
            sol(s, i+1, curr);
            curr.pop_back();
        }
    }
}

public:
    vector<vector<string>> partition(string s) {
        memset(dp, -1, sizeof(dp));
        vector<string> curr;
        sol(s, 0,curr);
        return ans;
    }
};
```
