class Solution {
public:
    int maxNumberOfBalloons(string text) {
        string b = "balloon";
        unordered_map<char, int> mp2;
        for (auto c : text) {
            mp2[c]++;
        }

        int ans=0;
        while (mp2.size() != 0) { // cha
            for (auto x:b){
                if(!mp2.contains(x)){
                    return ans;
                }
                mp2[x]--;
                if(mp2[x]==0){
                    mp2.erase(x);
                }
            }
            ans++;
        }

        return ans;
    }
};
