class Solution {
public:
    int maxNumberOfBalloons(string text) {
        int cnt[26] = {};

        for (char c : text) {
            cnt[c - 'a']++;
        }

        cnt['l' - 'a'] /= 2;
        cnt['o' - 'a'] /= 2;

        return min({
            cnt['b' - 'a'],
            cnt['a' - 'a'],
            cnt['l' - 'a'],
            cnt['o' - 'a'],
            cnt['n' - 'a']
        });
    }
};
