class Solution {
public:
    void sortColors(vector<int>& nums) {
        
        int h=nums.size()-1,l=0,mid=0;
        while(l<=h){
            if(nums[mid]==0){
                swap(nums[l], nums[mid]);
                l++;
                mid++;
            }else if(nums[mid]==2){
                mid++;
            }else{
                swap(nums[mid], nums[h]);
                h--;
            }
          
        }

    }
};
