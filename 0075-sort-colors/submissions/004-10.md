class Solution {
public:
    void sortColors(vector<int>& nums) {
        
        int a=0,b=0,c=0;
        for(auto x:nums){
            if(x==0){
                a++;
            }else if(x==1){
                b++;
            }else if(x==2){
                c++;
            }
        }

        for(int i=0; i<nums.size(); i++){
             if(a>0){
                nums[i]=0;
                a--;
            }else if(b>0){
                nums[i]=1;
                b--;
            }else if(c>0){
                nums[i]=2;
                c--;
            }
        }

    }
};
