class Solution {
    bool solve(int i, int j, string s, string p){
        if(j==p.length()){
            return i==s.length();
        }
        //*

        bool isCharMatch= false;
        if(j<p.size() && i<s.size()){
            isCharMatch= p[j]==s[i] || p[j]=='.';
        } 

        if(j+1<p.length() && p[j+1]=='*'){
            bool skip= solve(i, j+2, s, p);
            bool take= isCharMatch && solve(i+1, j, s,p);
            return skip || take;
        }

        return isCharMatch && solve(i+1, j+1, s, p);

    }
public:
    bool isMatch(string s, string p) {
        return solve(0, 0, s, p);
    }
};
