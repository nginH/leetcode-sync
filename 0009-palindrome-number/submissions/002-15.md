class Solution {
public:
    bool isPalindrome(int x) {
        int rev=0;
        int copy=x;
        if(x<0){
            return false;
        }
        while(x){
            rev=rev*10+x%10;
            x/=10;
        }

        return rev==copy;
    }
};
