#define min(a, b) a<b?a:b;
#define max(a, b) a>b?a:b;
class Solution {

private:

  bool canEat(vector<int>& piles, int h, int k) {

    long long hrs = 0;

    for (int x : piles) {
        hrs += ((long long)x + k - 1) / k;
    }

    return hrs <= h;
   }

public:
    int minEatingSpeed(vector<int>& piles, int h) {
       long long low=1;
       long long high=INT_MIN;

       for(auto x: piles){
            high=max(high, x);
       }
    long long ans=-1; 
       while(low<=high){
            long long mid = low + (high - low) / 2;
            bool flag=canEat(piles, h,mid);
            if(flag){
                // eat kar liya try lower
                ans=mid;
                high=mid-1;
            }else{
                low=mid+1;
            }

       }

       return ans;
    }
};
