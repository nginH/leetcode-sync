#define min(a, b) a<b?a:b;
#define max(a, b) a>b?a:b;
class Solution {

private:

    bool canEat(vector<int>&piles, int hr, double k){
        int hrs=0;
        for(auto x:piles){
             hrs+=ceil(x/k);
        }
        return hrs<=hr;

    }

public:
    int minEatingSpeed(vector<int>& piles, int h) {
       long long low=0;
       long long high=INT_MIN;

       for(auto x: piles){
            high=max(high, x);
       }
    int ans=-1; 
       while(low<=high){
            int mid = low + (high - low) / 2;
            bool flag=canEat(piles, h,mid);
            if(flag){
                // eat kar liya try lower
                ans=mid;
                high=mid-1;
            }else{
                low=mid+1;
            }

       }

       return ans;
    }
};
