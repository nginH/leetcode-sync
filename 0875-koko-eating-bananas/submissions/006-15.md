class Solution {
public:

    int calc(vector<int>&arr, int r){
        int ans=0;
        for(auto x:arr){
            ans+=(x+r-1)/r;
        }
        return ans;
    }


    int minEatingSpeed(vector<int>& arr, int k) {
        int l=0; int h=*max_element(arr.begin(), arr.end());
        int ans=h;
        while(l<=h){
            int mid=(l+h)/2;
            int val=calc(arr,mid );
            if(val<=k){ // mean migh we much lesser value exist in left side of the array..
                ans=mid;
                h=mid-1;
            }else{
                l=mid+1;
            }
        }

        return ans;

    }
};
