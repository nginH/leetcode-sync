class Solution {
public:

    long long int calc(vector<int>&arr, int speed){
         long long hours = 0;

        for (int bananas : arr) {
            hours += (bananas + speed - 1) / speed; // ceil division
        }

        return hours;
    }


    int minEatingSpeed(vector<int>& arr, int k) {
        int l=1; int h=*max_element(arr.begin(), arr.end());
        int ans=h;
        while(l<=h){
            int mid = l + (h - l) / 2;
                long long int val=calc(arr,mid );
            if(val<=k){ // mean migh we much lesser value exist in left side of the array..
                ans=mid;
                h=mid-1;
            }else{
                l=mid+1;
            }
        }

        return ans;

    }
};
