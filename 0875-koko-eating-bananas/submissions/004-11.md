class Solution {

private:

    bool canEat(vector<int>&piles, int hr, double k){
        int hrs=0;
        for(auto x:piles){
             hrs+=ceil(x/k);
        }
        return hrs<=hr;

    }

public:
    int minEatingSpeed(vector<int>& piles, int h) {
       int low=INT_MAX;
       int high=INT_MIN;

       for(auto x: piles){
            low=min(low, x);
            high=max(high, x);
       }
        int ans=-1;

       while(low<=high){

            int mid=(low+high)/2;
            bool flag=canEat(piles, h,mid);

            if(flag){
                // eat kar liya try lower
                ans=mid;
                high=mid-1;
            }else{
                low=mid+1;
            }



       }

       return ans;
    }
};
