class Solution {

    vector<pair<int, int>> dir = {
        {1, 0},
        {0, 1},
        {-1, 0},
        {0, -1},
    };
    void dfs(vector<vector<int>>& grid, int row, int col, int& ans) {
        if (row < 0 || col < 0 || row >= grid.size() || row >= grid[0].size()) {
            return;
        }

        if (grid[row][col] == 0 || grid[row][col] == '*') {
            return;
        }
        grid[row][col] = '*';
        ans++;
        for (auto& [dx, dy] : dir) {
            dfs(grid, row + dx, col + dy, ans);
        }
    }

public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int mx=0;

        for(int i=0; i<grid.size(); i++){
            for(int j=0; j<grid[0].size(); j++){
                int ans=0;
                dfs(grid, i, j, ans);
                mx=max(mx, ans);
            }
        }


        return mx;
    }
};
