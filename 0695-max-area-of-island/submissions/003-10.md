class Solution {

private: 

    int m, n;

    int ans=0;

    void dfs(vector<vector<int>>& grid, int i, int j, int &k){

        if(i<0 || j< 0 || i>=m || j>=n || grid[i][j]==0){
            return;
        }
        k++;
        ans=max(ans, k);
        grid[i][j]=0;
        
        dfs(grid, i+1, j, k);
        dfs(grid, i-1, j, k);
        dfs(grid, i, j+1, k);
        dfs(grid, i, j-1, k);

    }




public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        m=grid.size();
        n=grid[0].size();

        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(grid[i][j]==1){
                    int kk=0;
                    dfs(grid, i, j, kk);
                }
            }
        }

        return ans;
    }
};
