class StockSpanner {
public:
    StockSpanner() {
        
    }
    vector<pair<int,int>>arr;

    int next(int price) {
        int res=0;
        int idx=(int)arr.size();
        arr.push_back({price,idx});
        while(!arr.empty() && idx>=0 && price>=arr[idx].first){
            idx--;
            res++;
        }

        return res;


    }
};
