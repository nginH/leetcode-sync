class StockSpanner {
public:
    StockSpanner() {
        
    }
    vector<pair<int,int>>arr;

    int next(int price) {
        int span=1;
        while(!arr.empty()&& arr.back().first<=price){
            span+=arr.back().second;
            arr.pop_back();
        }
        arr.push_back({price, span});

        return span;


    }
};
