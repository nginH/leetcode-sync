class DSU{
private:
    vector<int> parent;
    vector<int> size;


public:
    DSU(int n){
        parent.resize(n+1);
        size.resize(n+1,1);
        for(int i=1; i<=n; i++){
            parent[i]=i; // parent of own;
        }
    }

    int find (int node){
        if(parent[node]==node){
            return node;
        }
        return parent[node]=find(parent[node]);
    }

    bool unite(int u, int v){

        int up=find(u);
        int vp= find(v);

        if(up==vp){
            return false; // already unite;
        }

      if (size[up] < size[vp]) {
            parent[up] = vp;
            size[vp] += size[up];
      } else {
            parent[vp] = up;
            size[up] += size[vp];
      }
        return true;
    }   

};


class Solution {

public:
    vector<int> findRedundantConnection(vector<vector<int>>& edges) {

        int n= edges.size();
        DSU dsu(n);

        for(auto x:edges){
            int u=x[0];
            int v=x[1];

            if(!dsu.unite(u, v)){
                // unite nhi hua .. mean already in cycle
                return {u, v}; // and its gurantee that only one cycle
            }
        }        
        return {}; 
    }
};
