class DSU{
    vector<int>parent, size;
    public:
     DSU(int n){
        parent.resize(n+1);
        size.resize(n+1);
        for(int i=0; i<=n; i++){
            parent[i]=i; // every node is parent of itself;
        }
    }

    int find(int u){
        if(parent[u]==u){
            return parent[u];
        }
        return parent[u]=find(parent[u]);
    }

    vector<int> unite(int u, int v){
        int up=find(u);
        int vp=find(v);
        if(up==vp){
            return {u, v};
        }

        if(size[up]>size[vp]){
            parent[vp]=up;
            size[up]+=size[vp];
        }else{
            parent[up]=vp;
            size[vp]+=size[up];
        }
        return {};
    }
};


class Solution {
public:
    vector<int> findRedundantConnection(vector<vector<int>>& edges) {
        DSU dsu(edges.size()+1);
        vector<int>ans;
        for(int i=0; i<edges.size(); i++){

            auto temp=dsu.unite(edges[i][0], edges[i][1]);
            if(temp.size()==0){
                // skip no same parent;
            }else{
                return temp;
            }

        }

        return ans;
    }
};
