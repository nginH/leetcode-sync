class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int prod=1;
        int ans=INT_MIN;
        int minProd=1;
        for(auto x:nums){
            if(x>0){
                prod=max(x, x*prod);
                minProd=min(x, x*minProd);
            }else{
                prod=max(x, x*prod);
                minProd=max(x, x*minProd);
            }

            ans=max({ans, prod, minProd});
        }
        return ans;
    }
};
