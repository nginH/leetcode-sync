class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int prod=1;
        int ans=INT_MIN;
        int minProd=1;
        for(auto x:nums){
            int oldProd=prod;
            int oldMinProd=minProd;

            prod=max({x, x*oldProd, x*oldMinProd});
            minProd=min({x, x*oldProd, x*oldMinProd});

            ans=max(ans, prod);
        }   
        return ans;
    }
};
