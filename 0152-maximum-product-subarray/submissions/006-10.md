#include<bits/stdc++.h>

class Solution {
public:
    int maxProduct(vector<int>& nums) {

        int maxProd=1;
        int minProd=1;
        int ans=INT_MIN;
        for(auto x:nums){
            if(x<0){
                swap(maxProd, minProd);
            }

            maxProd=max(x, x*maxProd);
            minProd=min(x, x*minProd);

            ans=max(maxProd, ans);

        }
        return ans;



    }
};
