class Solution {

public:
    vector<int> eventualSafeNodes(vector<vector<int>>& graph) {
        int n=graph.size();
        vector<vector<int>>adj(n+1);
        vector<int>inDegree(n+1, 0);

        for (int i = 0; i < n; i++) {
            for (int neighbor : graph[i]) {
                adj[neighbor].push_back(i);
                inDegree[i]++;
            }
        }

        queue<int>q;
        for(int i=0; i<n; i++){
            if(inDegree[i]==0){
                q.push(i);
            }
        }
        vector<int>topo;
        while(!q.empty()){
            int node= q.front();
            q.pop();
            topo.push_back(node);
            for(auto x: adj[node]){
                inDegree[x]--;
                if(inDegree[x]==0){
                    q.push(x);
                }
            }
        }
        sort(topo.begin(), topo.end());
        return topo;
    }
};
