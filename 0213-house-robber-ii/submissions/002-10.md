class Solution {

private:
int sol(vector<int>&nums, int low, int high){
    int prevMax=0;
    int currMax=0;

    for(int i=low; i<=high; ++i){
        int tMax=max(currMax, prevMax + nums[i]);
        prevMax=currMax;
        currMax=tMax;
    }
    return max(prevMax, currMax);
}

public:
    int rob(vector<int>& nums) {
        if(nums.size()==1){
            return nums[0];
        }
        return max(sol(nums, 0, nums.size()-2), sol(nums, 1, nums.size()-1));
    }
};
