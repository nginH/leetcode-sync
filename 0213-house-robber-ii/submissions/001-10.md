class Solution {


struct state{
    int take;
    int skip;
};

struct state sol(vector<int>&nums, int idx, int end){
    if(idx>end){
        return {0, 0};
    }

    struct state val=sol(nums, idx+1,end);
    return {
        (val.skip+ nums[idx]),
        max(val.skip, val.take)
    };
}


public:
    int rob(vector<int>& nums) {
        if(nums.size()==0){
            return 0;
        }
          if(nums.size()==1){
            return nums[0];
        }
        int n=nums.size();
        auto a= sol(nums, 0, n-2); // exclue end include first 
        auto b= sol(nums, 1, n-1); // inclue end exclude first

        return max(
            max(a.take, a.skip),
            max(b.take, b.skip)
        );
    }
};
