class Solution {


private:
    bool dfs(vector<vector<int>> &adj,int src,  int des, vector<bool>&vis){
        if(src==des){
            return true;
        }
        vis[src]=true;
        for(auto x:adj[src]){
            if(!vis[x]){
                if(dfs(adj, x, des, vis)){
                    return true;
                }
            }
        }
        return false;
    }
