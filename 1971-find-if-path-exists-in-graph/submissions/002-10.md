class Solution {




public:
    bool validPath(int n, vector<vector<int>>& edges, int source, int destination) { 

        vector<vector<int>>adj(n);

        queue<int>q;
        for(auto x:edges){
            adj[x[0]].push_back(x[1]);
            adj[x[1]].push_back(x[0]);
        }
        q.push(source);
        vector<int>vis(n, false);
        vis[source]=true;
        
        while(!q.empty()){
            int node= q.front();
            q.pop();
            if(node==destination){
                return true;
            }
            for(auto x: adj[node]){ // adges is adjency list.
                if(vis[x]==false){
                    vis[x]=true;
                    q.push(x);
                }
            }
        }

        return false;

    }
};
