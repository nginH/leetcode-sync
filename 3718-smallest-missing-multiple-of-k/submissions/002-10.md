class Solution {
public:
    int missingMultiple(vector<int>& nums, int k) {
        int smallest=k;
        int ans=0;

        unordered_set<int>st;
        for(auto x:nums){
            if(x%k==0){
                st.insert(x);
            }
        }
        for(int i=1; i<=1e9; i++){ // infinity
            if(!st.contains(k*i)){
                return k*i;
            }
        }
        return -1;
    }
};
