class Solution {
public:
    int missingMultiple(vector<int>& nums, int k) {
        unordered_set<int>st;
        for(auto x:nums){
            if(x%k==0){
                st.insert(x);
            }
        }
        for(int i=1; i<=nums.size()+1; i++){ // infinity
            if(!st.contains(k*i)){
                return k*i;
            }
        }
        return -1;
    }
};
