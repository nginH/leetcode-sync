/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:

    int length(ListNode *temp){
        int res=0;
        while(temp){
            res+=1;
            temp=temp->next;
        }
        return res;
    }


    ListNode* rotateRight(ListNode* head, int k) {
        int len=length(head);
        
        if(!head || k==0){
            return head;
        }
        k%=len;

        int step= len-k;
        if(k==0){
            return head;
        }
        ListNode *headCopy=head;
        ListNode *temp=head;
        ListNode *prev=nullptr;
        ListNode *last=head;

        while(step--){
            prev=temp;
            temp=temp->next;
        }

        while(last->next){
            last=last->next;
        }

        prev->next=nullptr;
        last->next=headCopy;
        return temp;  
    }
};
