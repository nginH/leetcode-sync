class Solution {

int point=0;

vector<int> dp;

int dfs(vector<int>&arr, int idx,unordered_map<int, int> &mp){
    
    if(idx>=arr.size()){
        return 0;
    }
    if(dp[idx]!=-1){
        return dp[idx];
    }

    int skip = dfs(arr, idx + 1, mp);
    int next=idx+1;

    if(next<arr.size() && arr[next]==arr[idx]+1){ // not constructive value..
        next++;
    }

    int take = arr[idx] * mp[arr[idx]] + dfs(arr, next, mp);

    dp[idx]= max(skip, take);
    return dp[idx];
    

 }




public:
    int deleteAndEarn(vector<int>& nums) {
        unordered_map<int, int> mp;

        
        for(auto x:nums){
            mp[x]++;
        }
        vector<int> deDup;
        for(auto x:mp){
            deDup.push_back(x.first);
        }
        sort(deDup.begin(), deDup.end());
        dp.assign(deDup.size()+1, -1);
        return  dfs(deDup, 0, mp);
    }
};
