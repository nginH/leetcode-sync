class Solution {
public:
    int countSpecialIntegers(vector<int>& nums) {
        if(nums.size()<=1){
            return nums.size();
        }
        int n=nums.size();
        int ans=-1;
        bool isToggle =false;
        int prev=nums[0];
        for(int i=1; i<nums.size(); i++){
            
            if(nums[i]==nums[i-1]){
                isToggle=true;
            }else{
                prev=nums[i];
                if(isToggle)
                    isToggle=false;
                    ans++;
            }
        }

        if(prev==nums[n-1]){
            ans++;
        }

        return ans;
    }
};
