class Solution {
public:
    int countSpecialIntegers(vector<int>& nums) {
        if(nums.size()<=1){
            return nums.size();
        }
        int n=nums.size();
        int ans=0;
        bool isToggle =false;
        int prev=nums[0];

        unordered_map<int, int>mp;
        
        for(int i=0; i<nums.size(); i++){
            if(i==0|| nums[i]!=nums[i-1]){
                mp[nums[i]]++;
            }
        }
        for(auto x:mp){
            if(x.second==1){
                ans++;
            }
        }

        return ans;
    }
};
