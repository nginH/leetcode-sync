class Solution {
public:
    
    bool canJump(vector<int>& nums) {

        vector<bool> maxVis(nums.size(), false);

        for(int i=0; i<nums.size(); i++){
            int x=nums[i];
            if(x>=nums.size()){
                return true;
            }
            maxVis[x+i]=true;
       }
        return maxVis[nums.size()-1];
    }
};
