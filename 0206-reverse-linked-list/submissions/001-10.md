/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {

private:
    ListNode * sol(ListNode*node){
        ListNode *prev=nullptr, *curr=node, *next=node;

        while(curr){
            next= curr->next;
            curr->next=prev;
            prev=curr;
            curr=next;
        }
        return prev;
     
    }

public:
    ListNode* reverseList(ListNode* head) {

        return sol(head);

    }
};
