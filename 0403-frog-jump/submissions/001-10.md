unordered_map<int, bool>mp;

class Solution {
    bool canCrossSub(
        vector<int>& stones,
        int idx,
        int jump,
        unordered_set<int>& st,
        vector<vector<int>>&dp
    ) {
        if (idx == stones.size() - 1) {
            return true;
        }
        
        if(dp[idx][jump]!=-1){
            return dp[idx][jump];
        }


        int currentPos = stones[idx];

        // jump - 1
        if (jump > 1) {
            int nextPos = currentPos + jump - 1;

            if (st.contains(nextPos)) {
                int nextIdx = lower_bound(
                    stones.begin(),
                    stones.end(),
                    nextPos
                ) - stones.begin();

                if (canCrossSub(stones, nextIdx, jump - 1, st, dp))
                    return dp[idx][jump]=true;
            }
        }

        // jump
        {
            int nextPos = currentPos + jump;

            if (st.contains(nextPos)) {
                int nextIdx = lower_bound(
                    stones.begin(),
                    stones.end(),
                    nextPos
                ) - stones.begin();

                if (canCrossSub(stones, nextIdx, jump, st, dp))
                    return dp[idx][jump]= true;
            }
        }

        // jump + 1
        {
            int nextPos = currentPos + jump + 1;

            if (st.contains(nextPos)) {
                int nextIdx = lower_bound(
                    stones.begin(),
                    stones.end(),
                    nextPos
                ) - stones.begin();

                if (canCrossSub(stones, nextIdx, jump + 1, st, dp))
                    return dp[idx][jump]= true;
            }
        }

        return dp[idx][jump]= false;
    }

public:
    bool canCross(vector<int>& stones) {
        unordered_set<int> st(stones.begin(), stones.end());
        // auto it= max_element(stones.begin(), stones.end());
        vector<vector<int>>dp(stones.size()+1, vector<int>(stones.size()+1, -1));
        // First jump must be exactly 1
        if (stones.size() > 1 && stones[1] != 1) {
            return false;
        }

        return canCrossSub(stones, 1, 1, st, dp);
    }
};
