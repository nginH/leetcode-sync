/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {

        if(!head){
            return nullptr;
        }
        unordered_map<Node*, Node*> mp;
        Node*temp=head;


        // copu
        while(temp){
            mp[temp]=new Node(temp->val);
            temp=temp->next;
        }


        // next and random

        temp=head;
        while(temp){
            mp[temp]->next= mp[temp->next];

            if(temp->random){
                mp[temp]->random=mp[temp->random];
            }

            temp=temp->next;

        }
        return mp[head];

    }
};
