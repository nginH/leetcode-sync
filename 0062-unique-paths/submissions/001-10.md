class Solution {
public:
    vector<pair<int, int>> dir{
        {0,1},// move up ;
        {1, 0}// move left;
    };
    int m, n;
    int ans=0;
    int soln(int row, int col ,vector<vector<int>> &dp){
        if(row<0 || col<0 || row>m-1 || col>n-1){
            return 0;
        }
        if(row==m-1 && col==n-1){
            return 1;
        }
        if(dp[row][col]!=-1){
            return dp[row][col];
        }
        
        int ans=0;
        
        for(auto [dx, dy]:dir){ 
            ans+=soln(row+dx, col+dy, dp); 
        }

        return dp[row][col] = ans;
    }

    int uniquePaths(int m, int n) {

        vector<vector<int>> dp(m, vector<int>(n, -1)); // nothing is visisted,..
 
        this->m=m;
        this->n=n;
        return soln(0, 0, dp);

    }   
};
