class Solution {

    vector<pair<int, int>> dir = {{1, 0}, {0, 1}, {-1, 0}, {0, -1}};

    int m, n;

    void dfs(vector<vector<int>>& grid, int row, int col) {
        if (row < 0 || col < 0 || row >= m || col >= n) {
            return;
        }
        if (grid[row][col] == 0) {
            return;
        }

        grid[row][col] = 0;
        for (auto& [dx, dy] : dir) {
            dfs(grid, row + dx, col + dy);
        }
    }

public:
    int numEnclaves(vector<vector<int>>& grid) {

        m = grid.size();
        n = grid[0].size();

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
            }
        }

        // boundary

        for (int i = 0; i < m; i++) {
            dfs(grid, i, 0);
            dfs(grid, i, n-1);
        }

        for (int i = 0; i < n; i++) {
            dfs(grid, 0, i);
            dfs(grid, m-1, i);
        }

        int ans = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1) {
                    ans++;
                }
            }
        }

        return ans;
    }
};
