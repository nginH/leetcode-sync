class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        int cnt1=0;
        int maj1=NULL;
        int cnt2=0;
        int maj2=NULL;

        for(int x:nums){

            if (cnt1 == 0 && x != maj2) {
                maj1 = x;
                cnt1 = 1;
            }
            else if (cnt2 == 0 && x != maj1) {
                maj2 = x;
                cnt2 = 1;
            }
            else if (x == maj1) {
                cnt1++;
            }
            else if (x == maj2) {
                cnt2++;
            }
            else {
                cnt1--;
                cnt2--;
            }
        }
        vector<int>ans;
        int c1=0, c2=0;
        for(auto x:nums){
            if(x==maj1)c1++;
            if(x==maj2)c2++;
        }
        int n=nums.size();
        if(c1>n/3)ans.push_back(maj1);
        if(c2>n/3 && maj1!=maj2)ans.push_back(maj2);
        return ans;
    }
};
