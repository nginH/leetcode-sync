class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        // only two answer;

        int a=0, ca=0;
        int b=0, cb=0;
        for(auto x:nums){
            if(x==a){
                ca++;
            }else if(x==b){
                cb++;
            }
           else if(ca==0){
                a=x;
                ca+=1;
            }else if(cb==0){
                b=x;
                cb+=1;
            }else{
                ca--;
                cb--;
            }
        }
        int tempa=0; int tempb=0;
        for(auto x:nums){
            if(x==a){
                tempa++;
            }
            else if(x==b){
                tempb++;
            }
        }
        int n=nums.size();
        vector<int>ans;
        if(tempa>n/3){
            ans.push_back(a);
        }
        if(tempb>n/3){
            ans.push_back(b);
        }

        return ans;

    }
};
