class Solution {

struct state{
    int take;
    int skip;
};

struct state sol(vector<int>& nums ,int idx){
    if(idx>=nums.size()){
        return {0,0};
    }
    auto val=sol(nums, idx+1);
    return {
        val.skip+nums[idx],// take curr
        max(val.take, val.skip)
    };

}


struct state sol2(vector<int>& nums){

    struct state curr={0,0};
    for(int i=0; i<nums.size(); i++){
        auto temp= curr;
        curr.take=nums[i]+curr.skip;
        curr.skip= max(temp.take, temp.skip);
    }
    return curr;
}

public:
    int rob(vector<int>& nums) {
        int idx=0;
        // auto ans= sol(nums, idx);
        auto ans= sol2(nums);
        return max(ans.take,ans.skip);
    }
};
