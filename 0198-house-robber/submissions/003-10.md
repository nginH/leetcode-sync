    int rob(vector<int>& nums) {
        int a=0;// i+1
        int b=0; //i+2

        for(int i=nums.size()-1; i>=0 ; i--){
            int take=nums[i]+b;
            int skip=a;

            int curr= max(take, skip);
            b=a;
            a=curr;
        }

        return a;
