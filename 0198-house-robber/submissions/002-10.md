class Solution {

struct state{
    int take;
    int skip;
};

struct state sol(vector<int>& nums ,int idx){
    if(idx>=nums.size()){
        return {0,0};
    }
    auto val=sol(nums, idx+1);
    return {
        val.skip+nums[idx],// take curr
        max(val.take, val.skip)
    };

}


public:
    int rob(vector<int>& nums) {
        int idx=0;
        auto ans= sol(nums, idx);
        return max(ans.take,ans.skip);
    }
};
