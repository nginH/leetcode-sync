class Solution {
public:
    int maximumUniqueSubarray(vector<int>& nums) {
        int maxScoreOfUnique=0;
        unordered_set<int>st;
        int left=0;
        int sum=0;
        for(int right=0; right<nums.size(); right++){
            sum+=nums[right];
            while(st.contains(nums[right])){
                sum-=nums[left];
                st.erase(nums[left]);
                left++;
            }
            st.insert(nums[right]);
            
            maxScoreOfUnique=max(maxScoreOfUnique,sum);

        }



        return maxScoreOfUnique;
    }
};
