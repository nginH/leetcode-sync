class Solution {

private:
    set<vector<int>>st;
    vector<int>subset;

    void sol(vector<int>& arr, int idx){
        if(idx==arr.size()){
            st.insert(subset);
            return;
        }


        subset.push_back(arr[idx]);
        sol(arr, idx+1);
        subset.pop_back();
        sol(arr, idx+1);
    }

public:
    

    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        sol(nums, 0);
        vector<vector<int>> ans;
        ans={st.begin(), st.end()};
        return ans;
    }
};
