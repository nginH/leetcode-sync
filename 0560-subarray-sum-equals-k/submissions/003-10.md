class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
       unordered_map<int, int> mp;
       mp[0]=1;
       int preSum=0, cnt=0;
       for(auto x:nums){
            preSum+=x;
            int rem=preSum-k;
            cnt+=mp[rem];
            mp[preSum]+=1;
       } 
       return cnt;
    }
};
