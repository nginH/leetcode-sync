class Solution {



public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> mp;
        int ans=0;
        mp[0]=1;

        int preSum=0;
        for(auto x:nums){
            preSum+=x;
            int need=preSum-k;
            if(mp.contains(need)){
                ans+=mp[need];
            }
            mp[preSum]++;
        }


        return ans;
    }
};
