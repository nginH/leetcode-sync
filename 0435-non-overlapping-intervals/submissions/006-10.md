class Solution {
public:
    int eraseOverlapIntervals(vector<vector<int>>& intervals) {
        sort(intervals.begin(), intervals.end());

        vector<vector<int>> ans;
        int k=0;
        for(auto x:intervals){
            if(ans.size()==0){
                ans.push_back(x);
            }else if(ans.back()[1]>x[0]){
                k+=1;
                ans.back()[1]=min(ans.back()[1], x[1]);
            }else{
                ans.push_back(x);
            }
        }

        return k;

    }
};
