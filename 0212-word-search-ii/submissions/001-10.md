class TrieNode{
    public:
        TrieNode * links[26];
        string word;

        TrieNode(){
            for(int i=0; i<26; i++){
               links[i]=nullptr;
            }
            word="";
        }

        void put(char c, TrieNode *node){
            links[c-'a']=node;
        }

        bool isContain(char c){
            return links[c-'a']!=nullptr;
        }

        TrieNode *get(char c){
            return links[c-'a'];
        }
        
};




class Solution {

private:

    TrieNode *root;

    vector<string>ans;

    void insert(string & word){
        TrieNode *temp=root;

        for(auto x:word){
            if(!temp->isContain(x)){
                temp->put(x, new TrieNode());
            }
            temp=temp->get(x); 
        }

        temp->word=word;
    }


    void dfs(vector<vector<char>>& board,int r, int c, TrieNode *node){
            if(r<0 || c<0 || r>=board.size() || c>=board[0].size()){
                return;
            }

            char ch=board[r][c];

            if(ch=='#' || !node->isContain(ch)){
                return;
            }

            node=node->get(ch);
            // found
            if(node->word!=""){
                ans.push_back(node->word);
                node->word="";
            }


            board[r][c]='#'; // mark visited;


            dfs(board, r+1, c, node);
            dfs(board, r-1, c, node);
            dfs(board, r, c+1, node);
            dfs(board, r, c-1, node);

            board[r][c]=ch;

    }






public:
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {

            root=new TrieNode();


            for(auto x:words){
                insert(x);
            }

            for(int i=0; i<board.size();i++ ){
                for(int j=0; j<board[0].size(); j++){
                    dfs(board, i, j, root);
                }
            }

            return ans;
    }
};
