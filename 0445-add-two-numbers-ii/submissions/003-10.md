/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {

private:
    ListNode *head=nullptr;

    int length(ListNode*l1){
        int ans=0;
        while(l1){
            l1=l1->next;
            ans++;
        }
        return ans;
    }
   ListNode* pad(ListNode* head, int cnt) {
    while (cnt--) {
        ListNode* node = new ListNode(0);
        node->next = head;
        head = node;
         }
        return head;
    }


    int solve(ListNode*l1, ListNode*l2){
        if(!l1 || !l2){
            return 0;
        }
        int carry=solve(l1->next,l2->next);

        int sum= l1->val+
                 l2->val+
                 carry;

        ListNode *node = new ListNode(sum%10);

        node->next=head;
        head=node;

        return sum/10;

    }



public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {

        int len1=length(l1);
        int len2=length(l2);
        int k= abs(len1-len2);

       int carry= solve(
            len1<len2?pad(l1, k):l1,
            len1>len2?pad(l2,k):l2
            );

        if(carry){
            ListNode* node = new ListNode(carry);
            node->next = head;
            head = node;
        }


        return head;
    }
};
