class Solution {
public:
    vector<int> findDisappearedNumbers(vector<int>& nums) {

        vector<int> ans;
        int n=nums.size();

        for(int i=0; i<nums.size(); i++){

            int idx= abs(nums[i])-1;

            if(nums[idx]>0){
                nums[idx]=-nums[idx];
            }


        }

        for(int i=0; i<nums.size(); i++){
            if(nums[i]>0 ){ // not vistited
                ans.push_back(i+1); // add the index
            }
        }
        return ans;
    }
};
