class Solution {
public:
    vector<int> findDisappearedNumbers(vector<int>& nums) {
        bool arr[nums.size()+1];
        memset(arr, false, nums.size()+1);

        vector<int> ans;
        int n=nums.size();

        for(int i=0; i<nums.size(); i++){
            arr[nums[i]]=true; // mark that value is vistited
        }

        for(int i=1; i<=nums.size(); i++){
            if(arr[i]==false){ // not vistited
                ans.push_back(i); // add the index
            }
        }
        return ans;
    }
};
