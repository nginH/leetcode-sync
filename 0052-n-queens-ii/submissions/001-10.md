class Solution {

  set<int> cols, pos, neg;
  int ans=0;

  void dfs(int n, int row){

        if(n==row){
            ans+=1;
            return;
        }

        for(int col=0; col<n; col++){
            
            if(
                cols.count(col) ||
                pos.count(row+col)||
                neg.count(row-col)
            )continue;
            cols.insert(col);
            pos.insert(row+col);
            neg.insert(row-col);
            dfs(n, row+1);
            cols.erase(col);
            pos.erase(row+col);
            neg.erase(row-col);
        }
  }


public:

  

    int totalNQueens(int n) {
      dfs(n, 0);
      return ans;  
    }
};
