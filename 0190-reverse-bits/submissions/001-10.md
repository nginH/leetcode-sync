class Solution {
public:
    int reverseBits(int n) {
       uint32_t ans = 0;

        for (int i = 0; i < 32; i++) {

            // Get rightmost bit
            int bit = n & 1;

            // Make space in ans
            ans = ans << 1;

            // Put extracted bit into ans
            ans = ans | bit;

            // Remove rightmost bit from n
            n = n >> 1;
        }

        return ans;   
    }
};
