class Solution {
public:
    string removeKdigits(string num, int k) {

        string st="";
        for(int i=0; i<num.size(); i++){
            while(k>0 && !st.empty() && st.back()>num[i]){ //ascii comperision
                st.pop_back();
                k--;
            }
            if(st.size()==0 && num[i]=='0'){
                continue;
            }
            st.push_back(num[i]);
        }
        return st.size()==0?"0":st;


    }
};
