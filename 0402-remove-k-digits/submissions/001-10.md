class Solution {
public:
    string removeKdigits(string num, int k) {

        if (num.size() == k) {
            return "0";
        }

        string st = "";

        // Monotonic increasing stack
        for (int i = 0; i < num.size(); i++) {

            while (k > 0 && !st.empty() && st.back() > num[i]) {
                st.pop_back();
                k--;
            }

            st.push_back(num[i]);
        }

        // If k is still remaining, remove from the back
        while (k > 0 && !st.empty()) {
            st.pop_back();
            k--;
        }

        // Remove leading zeros
        int start = 0;

        while (start < st.size() && st[start] == '0') {
            start++;
        }

        if (start == st.size()) {
            return "0";
        }

        return st.substr(start);
    }
};
