class Solution {
public:
    string removeKdigits(string num, int k) {

        if(num.size()==k){
            return "0";
        }
        string st="";
        for(int i=0; i<num.size(); i++){
            while(k>0 && !st.empty() && st.back()>num[i]){ //ascii comperision
                st.pop_back();
                k--;
            }
            st.push_back(num[i]);
        }
        string finalAns="";
        int start=0;
        for(int i=0; i<st.size(); i++){
            if(st[start]=='0'){
                start++;
                continue;
            }
            finalAns+=st[i];
        }
        return finalAns.size()==0?"0":finalAns;


    }
};
