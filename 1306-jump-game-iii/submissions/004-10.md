class Solution {

bool dfs(vector<int>&arr, int idx){
    if(idx<0 || idx>=arr.size() || arr[idx]<0){
        return false;
    }
    if(arr[idx]==0){
        return true;
    };
    arr[idx]*=-1;
    bool a= dfs(arr, idx-arr[idx]);
    bool b= dfs(arr, idx+arr[idx]);
    return a||b;

}


public:
    bool canReach(vector<int>& arr, int start) {
        return dfs(arr, start);
    }
};
