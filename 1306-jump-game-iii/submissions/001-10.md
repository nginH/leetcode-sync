class Solution {

bool dfs(vector<int>&arr, int idx){
    if(idx<0 || idx>=arr.size() || arr[idx]<0){
        return false;
    }
    if(arr[idx]==0){
        return true;
    };
    arr[idx]*=-1;
    bool a= dfs(arr, idx-arr[idx]);
    bool b= dfs(arr, idx+arr[idx]);
    return a||b;

}


public:
    bool canReach(vector<int>& arr, int start) {
        queue<int> q;
        q.push(start);

        while(!q.empty()){
            int idx=q.front();
            q.pop();
            if(arr[idx]<0){
                continue;
            }
            if(arr[idx]==0){
                return true;
            }
            int jump=arr[idx];
            arr[idx]*=-1;

            if(idx+jump<arr.size()){
                q.push(idx+jump);
            }

            if(idx-jump>=0){
                q.push(idx-jump);
            }
        }
        return false;
    }
};
