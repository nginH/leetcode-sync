/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverse(ListNode* head) {
        ListNode *curr = head, *next = nullptr, *prev = nullptr;
        while (curr) {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        return prev;
    }

    void reorderList(ListNode* head) {

        // 1->2->3->4->5

        if(!head || !head->next){
            return;
        }
        ListNode *fast = head, *slow = head;

        ListNode* t = nullptr;

        while (fast and fast->next) {
            fast = fast->next->next;
            t = slow;
            slow = slow->next;
        }

        t->next=nullptr;

        ListNode* last = reverse(slow);
        // 1->2->3<-4<-5

        ListNode* temp = head;
        while (last and temp) {
            // presevring the node
            ListNode* a = temp->next;
            ListNode* b = last->next;

            // adjusting the link
            temp->next = last;

            if (!a)
                break;

            last->next = a;

            temp = a;
            last = b;
        }
    }
};
