class Solution {

private:

    bool dfs(vector<int>& nums, int idx, int &sum, int curr){
       
        if(idx>=nums.size()){
            return true;
        }
        if(curr==sum/2){
            return true;
        }
        if(sum%2!=0){
            return false;
        }
        // take.
        curr+=nums[idx];
        bool take=dfs(nums, idx+1, sum, curr);
        curr-=nums[idx];
        bool skip=dfs(nums, idx+1, sum, curr);

        return take|| skip;
    }



public:
    bool canPartition(vector<int>& nums) {
        int sum=accumulate(nums.begin(), nums.end(), 0);
        return dfs(nums,0, sum, 0);
    }
};
