class Solution {

private:

    bool dfs(vector<int>& nums, int idx, int sum, int curr){
        if(curr==sum){
            return true;
        }

        if(curr>sum || idx>=nums.size()){
            return false;
        }
      
        // take.
        curr+=nums[idx];
        bool take=dfs(nums, idx+1, sum, curr);
        curr-=nums[idx];
        bool skip=dfs(nums, idx+1, sum, curr);

        return take|| skip;
    }



public:
    bool canPartition(vector<int>& nums) {
        int sum=accumulate(nums.begin(), nums.end(), 0);
        if(nums.size()<=1 || sum%2==1){
            return false;
        }
        return dfs(nums,0, sum/2, 0);
    }
};
