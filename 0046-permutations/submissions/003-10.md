class Solution {
public:

    void solve(vector<int> arr,  vector<vector<int>> &ans, int index){
        if(index>=arr.size()){
            ans.push_back(arr);
            return;
        }
        for(int j=index; j<arr.size() ; j++){
            swap(arr[j], arr[index]);
            solve(arr, ans, index+1);
            swap(arr[j], arr[index]);
        }
    }

    vector<vector<int>> permute(vector<int>& nums) {
        vector<vector<int>> ans;
        solve(nums, ans, 0);
        return ans;

    }
};
