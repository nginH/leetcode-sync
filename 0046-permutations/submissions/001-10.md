class Solution {

 vector<vector<int>>  ans;

void solve(vector<int>&nums, int idx){
    if(idx==nums.size()){
        ans.push_back(nums);
        return;
    }
   
   for(int i=idx; i<nums.size(); i++){
        swap(nums[idx],nums[i]);// swap
        solve(nums, idx+1);
        swap(nums[idx],nums[i]);// backtrack undo
   }
}


public:
    vector<vector<int>> permute(vector<int>& nums) {
        solve(nums, 0);
        return ans;
    }
};
