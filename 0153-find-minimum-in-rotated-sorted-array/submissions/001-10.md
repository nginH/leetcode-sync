class Solution {
public:
    int findMin(vector<int>& nums) {
        
        int ans=INT_MAX;

        int low=0; int high=nums.size()-1;

        while(low<=high){
            int mid=low+(high-low)/2;


            // low->mid
            // mid->high

            if(nums[low]<=nums[mid]){
                ans=min(ans,nums[low]); // answer stored less check other half
                low=mid+1;
            }else{ 
                ans=min(ans, nums[mid]);
                high=mid-1;
            }
        }

        return ans;



    }
};
