class Solution {
public:
    string reverseWords(string s) {
        vector<string> st;

        for(int i=0; i<s.size();i++){
            if(s[i]!=' '){
                int len=0;
                int inital=i;
                while(s[i]!=' ' && i <s.size()){
                    len++;
                    i++;
                }
                string x=s.substr(inital, len);
                st.push_back(x);
            }
        }
        int i=0, j=st.size()-1;

        while(i<j){
            auto a= st[i];
            auto b= st[j];
            st[j]=a;
            st[i]=b;
            i++;
            j--;
        }
        string ans;
        for(auto x:st){
            ans+=x;
            ans+=' ' ;
        }
        return ans.substr(0, ans.size()-1);

    }
};
