class Solution {
    vector<vector<string>> ans;
    set<int>cols, posdig, negdig;


    void dfs(int n, int row, vector<string>&board ){

        if(row==n){ // iterated till last
            ans.push_back(board);
            return;
        }

        for(int col=0; col<n ; col++){

            if(cols.count(col) ||  
                posdig.count(row-col)||
                negdig.count(row+col)
            ) continue;

            cols.insert(col);
            posdig.insert(row-col);
            negdig.insert(row+col);
            board[row][col]='Q';
            dfs(n, row+1, board);

            cols.erase(col);
            posdig.erase(row-col);
            negdig.erase(row+col);
            board[row][col]='.';
        }
    }





public:
    vector<vector<string>> solveNQueens(int n) {
        vector<string> board(n, string(n, '.'));
        dfs(n, 0, board);
        return ans;
    }
};
