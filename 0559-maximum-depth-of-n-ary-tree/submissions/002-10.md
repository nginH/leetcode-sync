/*
// Definition for a Node.
class Node {
public:
    int val;
    vector<Node*> children;

    Node() {}

    Node(int _val) {
        val = _val;
    }

    Node(int _val, vector<Node*> _children) {
        val = _val;
        children = _children;
    }
};
*/

class Solution {





public:
    int maxDepth(Node* root) {


        // bfs.. level order       

        queue<Node*> q;
        if(!root){
            return 0;
        }
        q.push(root);
        int level=0;

        while(!q.empty()){
            int sz=q.size();
            level++;
            while(sz--){
                Node *f= q.front();
                q.pop();
                for(auto x:f->children){
                    q.push(x);
                }
            }
        }
        return level;



    }
};
