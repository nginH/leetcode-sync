class Solution {
    vector<pair<int, int>> dir = {
        {0, 1},
        {1, 0},
        {0, -1},
        {-1, 0}
    };

public:
    int minimumEffortPath(vector<vector<int>>& heights) {
        int n = heights.size();
        int m = heights[0].size();

        // dist[r][c] = minimum effort required to reach (r,c)
        vector<vector<int>> dist(
            n, vector<int>(m, INT_MAX)
        );

        // {effort, row, col}
        priority_queue<
            tuple<int, int, int>,
            vector<tuple<int, int, int>>,
            greater<tuple<int, int, int>>
        > pq;

        dist[0][0] = 0;
        pq.push({0, 0, 0});

        while (!pq.empty()) {
            auto [effort, row, col] = pq.top();
            pq.pop();

            // Ignore stale entry
            if (effort > dist[row][col])
                continue;

            // Destination
            if (row == n - 1 && col == m - 1)
                return effort;

            for (auto [dx, dy] : dir) {
                int newRow = row + dx;
                int newCol = col + dy;

                if (newRow < 0 || newCol < 0 ||
                    newRow >= n || newCol >= m) {
                    continue;
                }

                int edgeCost = abs(
                    heights[newRow][newCol] -
                    heights[row][col]
                );

                // Path effort = maximum edge cost so far
                int newEffort = max(effort, edgeCost);

                if (newEffort < dist[newRow][newCol]) {
                    dist[newRow][newCol] = newEffort;
                    pq.push({
                        newEffort,
                        newRow,
                        newCol
                    });
                }
            }
        }

        return 0;
    }
};
