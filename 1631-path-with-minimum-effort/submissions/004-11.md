class Solution {
public:
    int minimumEffortPath(vector<vector<int>>& heights) {
        // {dist, {row, col}}'
        std::priority_queue<
            pair<int,pair<int, int>>,
            vector<pair<int,pair<int, int>>>, 
            greater<>>pq;// greater without arg only work on c++17>=

        int n=heights.size();
        int m=heights[0].size();
        vector<vector<int>> dist(n , vector<int>(m, 1e9));
        vector<pair<int, int>> direction={
            {0,1},
            {1,0},
            {-1, 0},
            {0, -1}
        };

        //0->n-1,m-1
        pq.push({0, {0,0}}); // dist 0 for {0,0} row col
        dist[0][0]=0;

        while(!pq.empty()){
            auto [distance, temp]= pq.top();
            auto [row, col]=temp;
            pq.pop();

            if(row==n-1 && col==n-1){
                return distance;
            }
            for(auto [dx, dy]: direction){
                int newRow= dx+row;
                int newCol= dy+col;
                if(newRow<0 || newCol<0 || newRow>=n || newCol>=m){
                    continue; // out of bound;
                }
                int newEffort= max(abs(heights[row][col]- heights[newRow][newCol]),distance);
                if(newEffort<dist[newRow][newCol]){
                    dist[newRow][newCol]=newEffort; // newMin;
                    pq.push({newEffort, {newRow, newCol}});
                }

            }
        }

        return 0;


    
    }
};
