class Solution {

vector<pair<int , int>> dir={
    {0,1},
    {1,0},
    {0, -1},
    {-1, 0}
};

public:
    int minimumEffortPath(vector<vector<int>>& heights) {

        priority_queue<
            tuple<int, int, int>,
            vector<tuple<int, int, int>>,
            greater<>
        > pq;
        int n=heights.size();
        int m=heights[0].size();
        vector<vector<int>> dist(n,vector<int>(m,INT_MAX));
        pq.push({0,0,0});
        dist[0][0]=0;

        while(!pq.empty()){
            int sz=pq.size();
            while(sz--){
                auto [cost, currRow , currCol]=pq.top();
                pq.pop();
                if(cost>dist[currRow][currCol]){ // stall state..
                    continue;
                }

                for(auto [dx, dy]:dir){
                    int newRow=dx+currRow;
                    int newCol=dy+currCol;
                    if(newRow<0 ||newCol<0 || newRow>=n || newCol>=m){
                        continue;  // no need to check..
                    }

                   int  newCost= max(
                        cost, 
                        abs(heights[newRow][newCol]-heights[currRow][currCol])
                   );

                   if(dist[newRow][newCol]>newCost){
                        dist[newRow][newCol]=newCost;
                        pq.push({newCost , newRow, newCol});

                   }

                }
            }
        }
        return dist[n-1][m-1];
    }
};
