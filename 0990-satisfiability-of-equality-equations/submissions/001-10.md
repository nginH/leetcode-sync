class DSU {
    vector<int> parent, size;
    public:
    DSU(int n){
        parent.resize(n+1);
        size.resize(n+1, 1);
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int u){
        if(parent[u]==u){
            return u;
        }
        return parent[u]=find(parent[u]);
    }

    void unite(int u, int v){
        int up=find(u);
        int vp=find(v);
        
        if(up==vp){
            return;
        }
        if(size[up]>size[vp]){
            parent[vp]=up;
            size[up]+=size[vp];
        }else{
            parent[up]=vp;
            size[vp]+=size[up];
        }
    }
};


class Solution {
public:
    bool equationsPossible(vector<string>& equations) {
        DSU dsu(26);
        for(auto &x:equations){
            if(x[1]=='=')
                dsu.unite(x[0]-'a', x[3]-'a');
        }

        for(auto &x:equations){

            if(x[1]=='!'){
                if(dsu.find(x[0]-'a') == dsu.find(x[3]-'a')){
                    return false;
                }
            }
        }

        return true;
    }
};
