class Solution {


bool isValidSplit(vector<int>& nums, int k, int limit){

    int split=1;
    int sum=0;
    for(auto x:nums){
        if(x+sum>limit){
            split++;
            sum=x;
        }else{
            sum+=x;
        }
    }

    return split<=k; // true if splited else false..
}


public:
    int splitArray(vector<int>& nums, int k) {
        int low=0;
        int high=0;
        for(auto x:nums){
            low=max(low, x);
            high+=x;
        }
        int ans=0;
        while(low<=high){
            int mid= (low+high)/2;
            bool val=isValidSplit(nums, k, mid);
            if(val){
                ans=mid;
                high=mid-1;
            }else{
                low=mid+1;
            }
        }
        return ans;
    }
};
