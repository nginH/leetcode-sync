class Solution {
public:

    bool canSplit(vector<int> &arr, int k, int sum ){
        int cnt=1;
        int temp=0;
        for(int i=0; i<arr.size(); i++){
            if(temp+arr[i]>sum){
                cnt++;
                temp=arr[i];
            }else{
                temp+=arr[i];
            }
        }

        return cnt<=k;
    }

    int splitArray(vector<int>& arr, int k) {
        
        int high=0;
        int low=0;
        int ans=-1;
        for(auto x:arr){
            low=max(low, x);
            high+=x;
        }
        while(low<=high){
            int mid=(low+high)/2;
            if(canSplit(arr, k, mid)){
                ans= mid;
                high=mid-1;
            }else{
                low=mid+1;
            }
        }

        return ans;



    }
};
