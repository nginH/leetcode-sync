    int leastInterval(vector<char>& tasks, int n) {
        // a=3.   a=2
        // b=3.   b=2
        // n=2    //c 1
                 //d1   
        //---8

        vector<int> freq(26,0);

        for(auto x:tasks){
            freq[x-'A']++;
        };
        int mx= *max_element(freq.begin(), freq.end());

        // number of ma
        
        int mxi=0;
        for(int i=0; i<26; i++){
            if(freq[i]==mx){
                mxi++;
            }
        }
        int slot= (int) (mx-1)*(n+1)+mxi;

        return max(slot,(int)tasks.size());
    }
};
