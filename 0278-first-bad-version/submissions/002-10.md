// The API isBadVersion is defined for you.
// bool isBadVersion(int version);

class Solution {
public:

    int firstBadVersion(int n) { //pattern binary search on answer;

        int low=0, high=n;
        int ans=n;
        while(low<=high){
            int mid=low+(high-low)/2;
            int isBad= isBadVersion(mid);

            if(isBad==true){
                ans=mid;
                high=mid-1;
            }else{
                // false..
                low=mid+1;
            }

        }
        return ans;
    }
};
