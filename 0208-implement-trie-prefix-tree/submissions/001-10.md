class Trie {


    struct TrieNode {
        bool isEndOfWord;
        TrieNode* kids[26];
    };

    struct TrieNode *root;


public:
    Trie() {
        root= create();
    }

    struct TrieNode* create(){
        struct TrieNode *temp= (struct TrieNode*)malloc(sizeof(struct TrieNode));
        for(int i=0;i<26; i++){
            temp->kids[i]=NULL;
        }
        temp->isEndOfWord=false;
        return temp;
    }
    
    void insert(string word) {
        struct TrieNode* crawl=root;
        for(char ch:word){
            if(!crawl->kids[ch-'a']){
                crawl->kids[ch-'a']=create();
            }
            crawl=crawl->kids[ch-'a'];
        }
        crawl->isEndOfWord=true;
    }
    
    bool search(string word) {

        struct TrieNode*crawl=root;
        for(char ch:word){
            if(crawl->kids[ch-'a']==NULL){
                return false;
            }
            crawl=crawl->kids[ch-'a'];
        }
        return crawl->isEndOfWord; // of crawl read at final index..
    }
    
    bool startsWith(string prefix) {
         struct TrieNode*crawl=root;
        for(char ch:prefix){
            if(crawl->kids[ch-'a']==NULL){
                return false;
            }
            crawl=crawl->kids[ch-'a'];
        }
        return true; // of crawl read at final index.. force return ture if iteration complete
    }
};
