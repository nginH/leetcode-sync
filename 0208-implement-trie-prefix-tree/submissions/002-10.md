class TrieNode{
    public:
        TrieNode * links[26];
        bool isLast;

        TrieNode(){
            for(int i=0; i<26; i++){
               links[i]=nullptr;
            }
            isLast=false;
        }

        bool isEnd(){
            return isLast;
        }
        void end(){
            isLast=true;
        }
        void put(char c, TrieNode *node){
            links[c-'a']=node;
        }

        bool isContain(char c){
            return links[c-'a']!=nullptr;
        }

        TrieNode *get(char c){
            return links[c-'a'];
        }
        
};

private:

public:

        }

            if(!temp->isContain(word[i])){

            temp= temp->get(word[i]);
