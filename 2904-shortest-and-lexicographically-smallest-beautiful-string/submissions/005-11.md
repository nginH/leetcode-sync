class Solution {
public:
    string shortestBeautifulSubstring(string s, int k) {
        string ans=s+"-"; // inf  
        int left=0;

        for(int right=0; right<=s.size(); right++){
            if(s[right]=='1'){
                k--;
            }
            if(k==0){
                string temp=s.substr(left, right-left+1);
                left++;// increment 1 show that it can use next
                ans=ans.size()>temp.size()?temp:ans;
                k++;
            }
            while(left<s.size( )&& s[left]!='1'){
                left++;
            };
        }
        return ans.size()==s.size()+1?"":ans;
    }
};
