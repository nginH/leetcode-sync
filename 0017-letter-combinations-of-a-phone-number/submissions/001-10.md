class Solution {

    vector<string> mapping = {
        "", "", "abc", "def", "ghi",
        "jkl", "mno", "pqrs", "tuv", "wxyz"
    };

void solve(vector<string>s, int i, int j, string &curr, vector<string> &ans){

    if(i==s.size()){
        ans.push_back(curr);
        return;
    }

    if(j==s[i].size()){
        // current string group finsh invalid path
        return;
    }

    // take..
    curr.push_back(s[i][j]);
    solve(s, i+1, 0, curr, ans);
    curr.pop_back(); // backtrack and try other path..
    solve(s, i, j+1, curr, ans); // skip.. increament the j ..
}


public:
    vector<string> letterCombinations(string digits) {
        
        vector<string>s;
        for(char x:digits){
            s.push_back(mapping[x-'0']);
        }

        vector<string>ans;
        string curr;
        if (digits.empty())
            return ans;


        solve(s, 0, 0, curr, ans);
        return ans;
        

    }
};
