class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        
        int i=0;
        int j=nums.size()-1;
        while(j>0 && i<nums.size()){
            int a=abs(nums[i]);
            int b=abs(nums[j]);

            if(a>b){
                swap(nums[i], nums[j]);
            }else{
                j--;
            }
        }
        for(int i=0; i<nums.size(); i++){
            nums[i]*=nums[i];
        }

        return nums;


    }
};
