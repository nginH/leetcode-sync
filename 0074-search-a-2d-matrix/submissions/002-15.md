class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        // make matrix as 1d array for searcj
        int n=matrix.size();
        int m=matrix[0].size();

        int low=0;
        int high=n*m-1;


        while(low<=high){
            int mid=low+(high-low)/2;

            auto row=mid/n;
            auto col =mid%n;

            if(matrix[row][col]==target)
                return true;
            else if(matrix[row][col]>target)//target smaller
                high=mid-1;
            else
                low=mid+1;
            
        }

        return false;

    }
};
