class Solution {
public:

    int bsearch(vector<int>arr, int n, int k){
        int low=0, high=n-1;
        while(low<=high){
            int mid=(low+high)/2;

            if(arr[mid]==k){
                return true;
            }else if(arr[mid]>k){
                high=mid-1;
            }else{
                low=mid+1;
            }
        }
        return false;
    }

    bool searchMatrix(vector<vector<int>>& arr, int k) {


        int row=arr.size();
        int col=arr[0].size();
        int low=0, high=row*col-1; // flatten to 1d

        while(low<=high){
            int mid=(low+high)/2;

           int  r=(int) mid/col;
            int c=mid%col;

            if(arr[r][c]==k){
                return true;
            }else if(arr[r][c]>k){
                high=mid-1;
            }else{
                low=mid+1;
            }
        }
        return false;
    
    return false;
    
    }

};
