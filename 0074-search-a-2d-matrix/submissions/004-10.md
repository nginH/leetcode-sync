class Solution {
public:

    int bsearch(vector<int>arr, int n, int k){
        int low=0, high=n-1;
        while(low<=high){
            int mid=(low+high)/2;

            if(arr[mid]==k){
                return true;
            }else if(arr[mid]>k){
                high=mid-1;
            }else{
                low=mid+1;
            }
        }
        return false;
    }

    bool searchMatrix(vector<vector<int>>& arr, int target) {
    

        for(int i=0; i<arr.size(); i++){
            if(target >= arr[i][0] && target <= arr[i][arr[i].size()-1]){
                return bsearch(arr[i], arr[i].size(), target);
            }
        }

    
    return false;
    
    }

};
