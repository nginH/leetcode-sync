class KthLargest {
public:

    priority_queue<int, vector<int>, greater<>> pq;//min heap..
    int k;
    
    void adjust(){
          while(pq.size()>this->k){
            pq.pop();
       }
    }

    KthLargest(int k, vector<int>& nums) {
        this->k=k;

       for(auto x:nums){
            pq.push(x);
       }
      adjust();
    }
    
    int add(int val) {
        pq.push(val);
        adjust();
        return pq.top();
    }
};
