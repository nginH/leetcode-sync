class Solution {
public:
    int findMin(vector<int>& arr) {
        
        int ans=INT_MAX;

        if(arr.size()==1){
            return arr[0];
        }


        int low=0; 
        int high=arr.size()-1;

        while(low<=high){
            int mid=(low+high)/2;

            if(arr[low]<arr[high]){ // entire array is sorted/
                return arr[low];
                break;
            }


            if(arr[mid]==arr[high]){
                high--;
                continue;
            }


            if(arr[mid]>=arr[low]){ // left half sorted.
                ans=min(ans, arr[low]);
                low=mid+1; 
            }else{
                ans=min(ans, arr[mid]);
                high=mid-1;
            }
        }

        return ans;



    }
};
