class Solution {
public:
    int findMin(vector<int>& arr) {
        int ans = arr[0], low = 0, high = arr.size() - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (arr[low] == arr[high]) {
                ans = min(ans, arr[low++]);
            } else if (arr[mid] >= arr[low]) {
                ans = min(ans, arr[low]);
                low = mid + 1;
            } else {
                ans = min(ans, arr[mid]);
                high = mid - 1;
            }
        }

        return ans;
    }
};
