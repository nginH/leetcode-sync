class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        vector<int> st;

        for(auto node:asteroids){
            bool alive=true;
            if(!st.empty() && st.back() > 0 && node < 0){
               while (alive && !st.empty() && st.back() > 0 && node < 0) {
                    if (st.back() + node < 0) {
                        st.pop_back();
                    }
                    else if (st.back() + node == 0) {
                        st.pop_back();
                        alive = false;
                    }
                    else {
                        alive = false;
                    }
                }
            }
            if(alive){
                st.push_back(node);
            }
        }

        return st;
    }
};
