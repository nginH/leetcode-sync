class Solution {
public:

    bool isPalindrom(string &s, int l, int r){
        while(l<=r){
            if(s[l]==s[r]){
                l++;
                r--;
            }else{
                return false;
            }
        }
        return true;
    }

    string longestPalindrome(string s) {
        string res="";
        for(int i=0; i<s.size(); i++){
            for(int j=i; j<s.size(); j++){
                int k=j-i+1;
                if(isPalindrom(s, i, j)){
                    if(k>=res.size()){
                        res=s.substr(i, k);
                    }
                }

            }
        }
        return res;
    }
};
