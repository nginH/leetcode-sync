public:
    string multiply(string num1, string num2) {
      
        long long n1=0;

        for(int i=0;i<num1.size(); i++ ){
            n1=n1*10+(num1[i]-'0');
        }
        long long n2=0;
        for(int i=0;i<num2.size(); i++ ){
            n2=n2*10+(num2[i]-'0');
        }

        int res=1LL*n1*n2;
        
        string ans="";
        while(res){
            ans+=res%10+'0';
            res/=10;
        }
        reverse(ans.begin(), ans.end());
        return ans;



    }
};
