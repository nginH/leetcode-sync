class Solution {
public:
    string multiply(string num1, string num2) {
        
        int n=num1.size();
        int m=num2.size();
        vector<int> res(n + m, 0);
        for(int i=n-1; i>=0; --i){ // 1 2 3
            for(int j=m-1; j>=0; --j){ // 6
                int a=num1[i]-'0';
                int b=num2[j]-'0';

                int pos1 = i + j + 1;
                int pos2 = i + j;

                int val=res[pos1]+a*b;
                res[pos1]= val%10;
                res[pos2]+= val/10;
            }
        }

        int start = 0;
        while (start < res.size() - 1 && res[start] == 0) {
            start++;
        }
        string ans;

        for (int i = start; i < res.size(); ++i) {
            ans.push_back(res[i] + '0');
        }
       return ans;
    }
};
