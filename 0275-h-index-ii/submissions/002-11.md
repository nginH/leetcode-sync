class Solution {
public:
// arr[i]>=i+1
    int hIndex(vector<int>& arr) {
        int l=0, h=arr.size();

        int ans=0;


        while(l<=(h-1)){
            int mid=(l+h)/2;

            if(arr[mid]>=h-mid){
                ans=h-mid;
                h=mid-1;
            }else{
                l=mid+1;
            }
        }
        return ans;
    }
};
