class Solution {
public:
// arr[i]>=i+1
    int hIndex(vector<int>& arr) {
        int l=0, h=arr.size()-1;

        int ans=0;

        if(arr.size()==1 && arr[0]==0){
            return 0;
        }

        while(l<=h){
            int mid=(l+h)/2;

            if(arr[mid]<=mid+1){
                ans=mid+1;
                l=mid+1;
            }else{
                h=mid-1;
            }
        }
        return ans;
    }
};
