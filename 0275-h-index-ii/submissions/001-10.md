class Solution {
public:
// arr[i]>=i+1
    int hIndex(vector<int>& arr) {
        int l=0, n=arr.size();
        int h=n-1;
        int ans=0;


        while(l<=h){
            int mid=(l+h)/2;

            if(arr[mid]>=n-mid){
                ans=n-mid;
                h=mid-1;
            }else{
                l=mid+1;
            }
        }
        return ans;
    }
};
