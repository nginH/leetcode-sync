/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {

private:
    ListNode* solve(ListNode* node, int &n) {
        if (node == nullptr)
            return nullptr;

        node->next = solve(node->next, n);

        n--;

        if (n == 0) {
            ListNode* temp = node->next;
            delete node;
            return temp;
        }

        return node;
    }


public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
       
       return solve(head, n);
    }
};
