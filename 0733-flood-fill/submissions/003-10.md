class Solution {


private:

    int m, n;

    void dfs(vector<vector<int>>& image, int i, int j,int k,  int color){

        if (i < 0 || j < 0 || i >= m || j >= n || image[i][j] !=k){
            return ;
        }
     
        image[i][j]=color;

        dfs(image, i+1, j,k, color);
        dfs(image, i-1, j,k, color);
        dfs(image, i, j+1,k, color);
        dfs(image, i, j-1,k, color);
    }



public:
    vector<vector<int>> floodFill(vector<vector<int>>& image, int i, int j, int color) {

        m=image.size();
        n=image[0].size();

        if (image[i][j] == color)
            return image;


        dfs(image, i, j, image[i][j],  color);
        return image;
    }
};
