class Solution {


vector<pair<int, int>> dir={
    {1, 0},
    {0, 1},
    {-1, 0},
    {0, -1},
};

void dfs(vector<vector<int>>& image, int r, int c, int color, int scolor){

    if(r<0 || c<0 || r>=image.size()|| c>=image[0].size()){
        return;
    }
    
    if(scolor!=image[r][c]){
        return ; // 
    }

    if(image[r][c]==color){
        return;
    }
    image[r][c]=color;

    for(auto &[dx, dy]:dir){
        dfs(image, r+dx, c+dy, color,scolor );
    }
}

public:
    vector<vector<int>> floodFill(vector<vector<int>>& image, int sr, int sc, int color) {


       dfs(image, sr, sc, color, image[sr][sc]);
       return image; 


    }
};
