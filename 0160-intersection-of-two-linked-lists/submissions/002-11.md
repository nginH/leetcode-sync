/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        unordered_set<ListNode*> st;
        while(headA || headB){

            if(st.contains(headA)){
                return headA;
            }
            if(st.contains(headB)){
                return headB;
            }
            if(headA){
                st.insert(headA);
                headA=headA->next;
            }

            if(headB){
                st.insert(headB);
                headB=headB->next;
            }
        }   
        return nullptr;
    }
};
