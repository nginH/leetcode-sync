/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        unordered_set<ListNode*> st;
        while(headA || headB){
            if(headA){
                if(st.contains(headA)){
                    return headA;
                }
                st.insert(headA);
                headA=headA->next;
            }
            if(headB){
                if(st.contains(headB)){
                    return headB;
                }
                st.insert(headB);
                headB=headB->next;
            }
          
        }   
        return nullptr;
    }
};
