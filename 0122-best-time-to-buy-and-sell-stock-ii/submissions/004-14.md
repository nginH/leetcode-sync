class Solution {

int soln(vector<int>& prices, int idx, int canBuy){
    
    if(idx>=prices.size()){
        return 0;
    }

    // skip.
   int skip = soln(prices, idx+1,canBuy);
   int take=0;
   if(canBuy){
        take=-prices[idx]+soln(prices, idx+1, 0);
   }else{
        take=prices[idx]+soln(prices, idx+1, 1);
   }
   return max(skip, take);

}
