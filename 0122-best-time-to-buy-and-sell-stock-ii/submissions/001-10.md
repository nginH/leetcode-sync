class Solution {

int soln(vector<int>& prices, int idx, int canBuy,  vector<vector<int>> &dp){
    
    if(idx>=prices.size()){
        return 0;
    }

    if(dp[idx][canBuy]!=-1){
        return dp[idx][canBuy];
    }


    // skip.
   int skip = soln(prices, idx+1,canBuy, dp);
   int take=0;
   if(canBuy){
        take=-prices[idx]+soln(prices, idx+1, 0, dp);
   }else{
        take=prices[idx]+soln(prices, idx+1, 1, dp);
   }
   return dp[idx][canBuy]= max(skip, take);

}


public:
    int maxProfit(vector<int>& prices) {
        vector<vector<int>>dp( prices.size()+1, vector<int>(2, -1));
        dp[0][0]=true;
        int curr=INT_MAX;
        return soln(prices, 0, 1, dp);
    }
};
