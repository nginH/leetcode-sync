class Solution {
    vector<int>dp;

    int dfs(vector<int> &arr, int target){

        if(target==0){
            return 1;
        }
        if(dp[target]!=-1){
            return dp[target];
        }

        int ways=0;
        for(auto x:arr){
            if(x<=target){
                ways+=dfs(arr, target-x);
            }
        }

        return dp[target]=ways;
    
    }

public:
    int combinationSum4(vector<int>& nums, int target) {
        dp.assign(target + 1, -1);
        dp[0]=1;
        return dfs(nums, target);
    }
};
