class MedianFinder {
public:
    vector<double>arr;
    int size;
    MedianFinder() {
        this->size=0;
    }
    
    void addNum(int num) {
        double x=(double)num;
        arr.push_back(x);
        this->size+=1;
    }
    
    double findMedian() {
        if(size%2){// odd one median
            return (double)arr[size/2];
        }else{
            double a= floor(arr[this->size/2-1]);
            double b= ceil(arr[this->size/2]);
            return (a+b)/2;
        }
    }
};
