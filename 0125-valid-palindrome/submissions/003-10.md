class Solution {
public:
    bool isPalindrome(string s) {
        
        string sw="";

        for(int i=0; s[i]; i++){
            if(s[i]>='a' && s[i]<='z' || s[i]>='A' && s[i]<='Z' || s[i]>='0' && s[i]<='9'){
                sw+=tolower(s[i]);
            }
        }

        string rv=sw;

        reverse(sw.begin(), sw.end());

        return sw==rv;

    }
};
