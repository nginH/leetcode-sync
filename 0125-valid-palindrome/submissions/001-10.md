class Solution {
public:
    bool isPalindrome(string s) {
        string norm="";

        for(char c:s){
            if(c>='a'&& c<='z'){
                norm+=c;
            }
            if(c>='A' && c<='Z'){
                norm+=static_cast<char>(std::tolower(c));;
            }
            if(c>='0' && c<='9'){
                norm+=c;
            }
        }

        int i=0; 
        int j=norm.size()-1;

        while(i<=j){
            if(norm[i]!=norm[j]){
                return false;
            }
            i++;
            j--;
        }
        return true;
    }
};
