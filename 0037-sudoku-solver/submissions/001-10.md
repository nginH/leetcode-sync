class Solution {
    bool isValid(vector<vector<char>>& board, int row, int col, char c){
    for(int i=0; i<9; i++){

        if( i!=row && board[i][col]==c){
            return false;
        }
        if( i !=col && board[row][i]==c){
            return false;
        }
        
        int r = 3 * (row / 3) + i / 3;
        int cc = 3 * (col / 3) + i % 3;

        if ((r != row || cc != col) && board[r][cc] == c)
            return false;
    }
    return true;
}

  bool solve(vector<vector<char>>& board) {
        for(int i=0; i<9; i++){
            for(int j=0; j<9; j++){
                if(board[i][j]=='.'){
                    for(int k='1'; k<='9'; k++){
                        if(isValid(board, i, j,k)){
                            board[i][j]=k;
                            if(solve(board)){  
                                return true;
                            }else
                                board[i][j]='.';
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }

public:
    void solveSudoku(vector<vector<char>>& board) {
       solve(board);
    }
};
