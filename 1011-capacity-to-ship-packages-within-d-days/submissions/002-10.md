class Solution {
public:

    int canShip(vector<int> arr, int d, int c){
        int usedData=1;
        int temp=0;
        for(auto x: arr){
            if(temp+x>c){
                usedData++;
                temp=x;
            }else{
                temp+=x;
            }
        }
        return usedData<=d;   // if 0 possible if greater then one ..
    }

    int shipWithinDays(vector<int>& arr, int days) {

        int l=0,h=0;
        for(auto x:arr){
            l=max(x,l);
            h+=x;
        }

        int ans=0;
        while(l<=h){
            int mid=(l+h)/2;
            int t=canShip(arr, days, mid);


            if(t){ //target time kam laga ..
                ans=mid;
                h=mid-1;
                
            }else{
                l=mid+1;
            }
        }
        return ans;
    }
};
