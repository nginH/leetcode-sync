class Solution {

private:

   bool ck(vector<int>& weights, int days, int capacity) {

    int day = 1;
    int sum = 0;

    for (int x : weights) {

        if (sum + x > capacity) {
            day++;
            sum = x;
        } else {
            sum += x;
        }
    }

    return day <= days;
}

public:
    int shipWithinDays(vector<int>& weights, int days) {
        int low=INT_MIN;
        int high=0;


        // return ck(weights , days, 14);
        for(auto x:weights){
            low=max(low, x); // isse kam tho saman le nhi jaa payega.
            high+=x; // sab ek sath ..
        }
        int ans;

        while(low<=high){
            int mid=low+(high-low)/2;
            bool isval=ck(weights, days, mid);

            if(isval){
                ans=mid;
                high=mid-1;
            }else{
                low=mid+1;
            }
            
        }

        return ans;

        
    }
};
