class Solution {
public:
    vector<int> findMissingAndRepeatedValues(vector<vector<int>>& grid) {
        map<int, int> m;
        for(int i=0; i<grid.size(); i++){
            for(int j=0; j<grid.size(); j++){
                m[grid[i][j]]++;
             }
        }
        int dubleOccurance=0;
        int secondAnswer=0;
        int k=0;
        bool ck=true;
        for(auto x:m){
            k++;
            if(x.second>1){
                dubleOccurance= x.first;
            }
            if(x.first!=k && ck){
                secondAnswer=k;
                ck=false;
            }
        }
        int a=grid.size();
        return {dubleOccurance,secondAnswer==0 ?a*a:secondAnswer };

    }
};
