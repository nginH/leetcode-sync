class Solution {
public:
    int longestSubarray(vector<int>& nums, int k) {

        unordered_map<int, vector<int>> mp;// number and prime count
        for (int x : nums) {
            if (mp.count(x)) {
                continue;
            }

            int num = x;
            vector<int> factors;

            for (int p = 2; p * p <= num; p++) {

                if (num % p == 0) {
                    factors.push_back(p);
                    while (num % p == 0) {
                        num /= p;
                    }
                }
            }
            if (num > 1) {
                factors.push_back(num);
            }
            mp[x] = factors;
        }

        unordered_map<int, int> freq;

        int left = 0;
        int distinct = 0;
        int ans = 0;
        for (int right = 0; right < nums.size(); right++) {
            for (int p : mp[nums[right]]) {
                if (freq[p] == 0) {
                    distinct++;
                }
                freq[p]++;
            }
            while (distinct > k) {
               for (int p : mp[nums[left]]) {
                    freq[p]--;
                    if (freq[p] == 0) {
                        distinct--;
                    }
                }
                left++;
            }
            ans = max(ans, right - left + 1);
        }

        return ans;
    }
};
