class DSU{
    vector<int>parent, size;
    public:
        DSU(int n){
            parent.resize(n);
            size.resize(n, 1);
            iota(parent.begin(), parent.end(), 0);
        }

        int find(int n){
            if(parent[n]==n){
                return n;
            }
            return parent[n]=find(parent[n]);
        }

        bool unite(int n,int m){
            int u=find(n);
            int v=find(m);
            if(u==v){
                return true;
            }
            if(size[u]>size[v]){
                parent[v]=u;
                size[u]+=size[v];
            }else{
                parent[u]=v;
                size[v]+=size[u];
            }
            return false;
        }
};


class Solution {
public:
    int makeConnected(int n, vector<vector<int>>& connections) {
        if(connections.size()+1<n){ //more cable is okay 
            return -1; // less cable return -1
        }
        DSU dsu(n);
        int ans=0;
        unordered_set<int>st;
        for(auto x:connections){
            st.insert(x[0]);
            st.insert(x[1]);
            if(dsu.unite(x[0], x[1])){
                ans++;
            }
        }



        if(connections.size()-st.size() <= ans){
            return n-st.size() ; // one these many needed
        }// 4 
        return ans; // no ring presnt then 0
    }
};
