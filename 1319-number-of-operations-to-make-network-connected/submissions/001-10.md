class DSU {
    vector<int> parent, size;

public:
    DSU(int n) {
        parent.resize(n);
        size.resize(n, 1);
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int n) {
        if (parent[n] == n)
            return n;

        return parent[n] = find(parent[n]);
    }

    bool unite(int n, int m) {
        int u = find(n);
        int v = find(m);

        if (u == v)
            return true;

        if (size[u] > size[v]) {
            parent[v] = u;
            size[u] += size[v];
        } else {
            parent[u] = v;
            size[v] += size[u];
        }

        return false;
    }
};

class Solution {
public:
    int makeConnected(int n, vector<vector<int>>& connections) {

        if (connections.size() < n - 1)
            return -1;

        DSU dsu(n);

        int extra = 0;

        for (auto& edge : connections) {
            if (dsu.unite(edge[0], edge[1])) {
                extra++;
            }
        }

        int components = 0;

        for (int i = 0; i < n; i++) {
            if (dsu.find(i) == i)
                components++;
        }

        return components - 1;
    }
};
