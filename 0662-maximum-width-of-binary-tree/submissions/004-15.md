/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */





class Solution {

struct Internal{
    TreeNode *node;
    int num;
    int level;
};

public:
    int widthOfBinaryTree(TreeNode* root) {
        int res=0;
        queue<Internal> q;
        q.push({root, 1, 0});

        int prevLevel=0, prevNum=1;

        while(!q.empty()){
            auto x=q.front();
            q.pop();

            if(x.level>prevLevel){
                prevLevel=x.level;
                prevNum=x.num;
            }

            res=max(res, x.num-prevNum+1);

            if(x.node->left){
                q.push({
                    x.node->left,
                    2*x.num,
                    x.level+1
                 });
            }
            if(x.node->right){
                q.push({
                    x.node->right,
                    2*x.num+1,
                    x.level+1
                 });
            }

        }



        return res;
    }
};
