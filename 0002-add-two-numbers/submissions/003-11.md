/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:

    ListNode* reverse(ListNode*head){
        ListNode*curr=head, *prev=nullptr, *next=nullptr;
        while(curr){
            next=curr->next;
            curr->next=prev;
            prev=curr;
            curr=next;
        }

        return prev;
    }

    ListNode* addTwoNumbers(ListNode* ll1, ListNode* ll2) {
        ListNode *l1=reverse(ll1);
        ListNode *l2=reverse(ll2);

        int carry=0;
        
        ListNode *dummy=nullptr;
        while(l1 or l2 or carry){
            int val= (l1 ? l1->val:0) + 
                     (l2 ? l2->val:0) +
                     carry;

            carry=val/10; // carry
            val%=10;
            ListNode*temp= new ListNode(val);
            temp->next=nullptr;
            if(dummy!=nullptr){
                temp->next=dummy; // adding to front so i dont have to reverese
                dummy=temp;
            }else{
                dummy=temp;
            }

            if(l1){
                l1=l1->next;
            }
            if(l2){
                l2=l2->next;
            }
        }

        return reverse(dummy);
    }
};
