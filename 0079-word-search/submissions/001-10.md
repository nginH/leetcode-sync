class Solution {

private:

    vector<pair<int, int>>dir={
        {1,0},
        {0,1},
        {-1,0},
        {0,-1}
    };

    bool sol(int i, int j,vector<vector<char>>& board,string word, int idx){
        // out of bound.
        
        if (i < 0 || j < 0 ||
            i >= board.size() ||
            j >= board[0].size() ||
            board[i][j] != word[idx]) {
            return false;
        }
        if (board[i][j] == '#')
            return false;


        if(idx==word.size()-1){ // move to end.. 
            return true;
        }
        char orig=board[i][j];
        board[i][j]='#';


        for(auto &[dx, dy]:dir){
            if(sol(i+dx, j+dy, board, word, idx+1)){ // kisi v point pe true mila then return true,
                return true;
            }
        }
        board[i][j]=orig;
        return false;
    }


public:
    bool exist(vector<vector<char>>& board, string word) {
        int R = board.size();
        int C = board[0].size();

        if (word.size() > R * C)
            return false;

         for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
               if (board[i][j] == word[0] &&
                    sol(i, j, board, word,0)) {
                    return true;
                }
            }
        }
        return false;
    }
};
