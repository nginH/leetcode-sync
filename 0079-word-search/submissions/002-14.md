class Solution {

private:

    vector<pair<int, int>>dir={
        {1,0},
        {0,1},
        {-1,0},
        {0,-1}
    };

    bool sol(int i, int j,vector<vector<char>>& board, string & curr, string word){
        // out of bound.
        if(i<0 || j<0 || i>=board.size() || j>=board[0].size()){
            return false;
        }
        if (board[i][j] == '#')
            return false;


        curr.push_back(board[i][j]);
        if(word==curr){
            return true;
        }
        char orig=board[i][j];
        board[i][j]='#';


        for(auto &[dx, dy]:dir){
            if(sol(i+dx, j+dy, board,curr, word)){ // kisi v point pe true mila then return true,
                return true;
            }
        }
        board[i][j]=orig;
        curr.pop_back();
        return false;
    }


public:
    bool exist(vector<vector<char>>& board, string word) {
        string curr="";
         for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                if (sol(i, j, board, curr, word))
                    return true;
            }
        }
        return false;
    }
};
