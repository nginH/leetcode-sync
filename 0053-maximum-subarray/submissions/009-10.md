#include <limits>

class Solution {
public:
    int maxSubArray(vector<int>& arr) {
        int ans=INT_MIN;
        int sum=0;
 if(arr.size()==1){
            return arr[0];
        }
        
        for(int i=0; i<arr.size(); i++){
            if(sum<0 ){
                sum=arr[i];
            }else{
                sum+=arr[i];
            }
            ans=max(sum, ans);
        }
        return ans;
    }
};
