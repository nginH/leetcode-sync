#include <limits>

class Solution {
public:
    int maxSubArray(vector<int>& arr) {
        int ans=INT_MIN;
        int sum=0;

        
        for(int i=0; i<arr.size(); i++){
            sum+=arr[i];
            if(sum<0 ){
                sum=0;
            }
            ans=max(sum, ans);
        }
        return ans;
    }
};
