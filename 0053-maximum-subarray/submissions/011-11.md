#include <limits>

class Solution {
public:
    int maxSubArray(vector<int>& arr) {
        int ans=INT_MIN;
        int sum=arr[0];

        for(int i=1; i<arr.size(); i++){
            if(sum<0){
                sum=arr[i];
            }else{
                sum+=arr[i];
            }
            ans=max(sum, ans);
        }
        return ans;
    }
};
