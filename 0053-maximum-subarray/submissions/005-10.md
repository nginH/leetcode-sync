class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int maxSum=0;
        int ans=INT_MIN;
        for(auto x:nums){
            maxSum=max(x, maxSum+x);
            ans=max(maxSum, ans);
        }
        return ans;
    }
};
