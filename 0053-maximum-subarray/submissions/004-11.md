class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int sum=0;
        int intCurrMax=0;
        for(int x:nums){
            intCurrMax=max(x, x+intCurrMax);
            sum=max(sum,intCurrMax);

        }
        return sum;


    }
};
