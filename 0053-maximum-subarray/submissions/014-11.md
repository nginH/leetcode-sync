#include <limits>

class Solution {
public:
    int maxSubArray(vector<int>& arr) {
           int ans=0;
        int sum=0;

        for(int i=0; i<arr.size(); i++){
            if(sum<0){
                sum=arr[i];
            }else{
                sum+=arr[i];
            }
            ans=max(sum, ans);
        }
        return ans;
    }
};
