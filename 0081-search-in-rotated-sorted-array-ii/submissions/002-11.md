class Solution {
public:
    bool search(vector<int>& arr, int target) {
        

        int left=0, right=arr.size()-1;

        while(left<=right){
            int mid=(left+right)/2;


            if(arr[mid]==target){
                return true;
            }

            if(arr[left]<arr[mid]){// let sorted
                if(arr[left]<=target && target>=arr[mid]){
                    right=mid-1;
                }else{
                    left=mid+1;
                }
            }else{
                if(arr[right]>=target && target<arr[mid]){
                    left=mid+1;
                }else{
                    right=mid-1;
                }
            }
        }

        return false;



    }
};
