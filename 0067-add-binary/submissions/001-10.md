class Solution {
public:
    string addBinary(string a, string b) {
        string ans="";
        int i=a.size()-1; int j=b.size()-1;
        int carry=0;
        while(i>=0 || j>=0 || carry){
                int suba=0;
                if(i>=0){
                 suba=a[i]-'0';
                }
                int subb=0;
                if(j>=0){
                  subb= b[j]-'0';
                }

                int k=suba+subb+carry;
                
                int val=k%2;
                 carry=k/2;
                ans+=to_string(val);
                i--;
                j--;
        }
        reverse(ans.begin(), ans.end());
        return ans;
    }
};
