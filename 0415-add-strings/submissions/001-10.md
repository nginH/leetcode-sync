class Solution {
public:
    string addStrings(string num1, string num2) {
        int m=num1.size();
        int n=num2.size();
        reverse(num1.begin(), num1.end());
        reverse(num2.begin(), num2.end());

        int i=0;
        int j=0;
        int carry=0;

        string ans="";
        
        while(i<=m-1 || j<=n-1 || carry){
            if(i<=m-1){
                carry+=num1[i]-'0';
            }
            if(j<=n-1){
                carry+=num2[j]-'0';
            }

            ans.push_back((carry%10+'0'));
            carry/=10;
            i++;
            j++;
        }

        reverse(ans.begin(), ans.end());
        return ans.substr(0, ans.size());

    }
};
