class Solution {

      void dfs(
        string curr,
        string start,
        unordered_map<string, vector<string>>& parent,
        vector<string>& path,
        vector<vector<string>>& res
    ) {
        if (curr == start) {
            vector<string> temp = path;
            reverse(temp.begin(), temp.end());
            res.push_back(temp);
            return;
        }

        for (auto& wd : parent[curr]) {
            path.push_back(wd);
            dfs(wd, start, parent, path, res);
            path.pop_back();
        }
    }

public:
    vector<vector<string>> findLadders(string beginWord, string endWord,
                                       vector<string>& wordList) {
        vector<vector<string>> res;
        unordered_set<string> st(wordList.begin(), wordList.end());
        queue<string> q;
        q.push(beginWord);

        unordered_map<string, vector<string>> parent;

        vector<string> temp;
        bool found = false;
        while (!q.empty()) {
            int sz = q.size();

            unordered_set<string> levelVisited;

            while (sz--) {
                auto curr = q.front();
                q.pop();
                temp.push_back(curr);

                if (curr == endWord) {
                    found = true;
                }
                string wd = "";
                for (int j = 0; j < curr.size(); j++) {
                    wd = curr;
                    for (int i = 0; i < 26; i++) {
                        wd[j] = 'a' + i;
                        if (wd == curr) {
                            continue;
                        }
                        if (!st.contains(wd)) {
                            continue;
                        }

                        if (!levelVisited.contains(wd)) {
                            levelVisited.insert(wd);
                            q.push(wd);
                        }
                        parent[wd].push_back(curr);
                    }
                }
            }
            for (auto& word : levelVisited) {
                st.erase(word);
            }
        }
        vector<string> path;
        path.push_back(endWord);
        dfs(endWord, beginWord, parent, path, res);
        return res;
    }
};
