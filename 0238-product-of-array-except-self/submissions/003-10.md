class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        
        vector<int> arr;
        

        int pd=1;
        for(int i=0;i<nums.size(); i++){
            arr.push_back(pd);
            pd*=nums[i]; // prefix . 
        }
        pd=1;
        for(int i=nums.size()-1; i>=0; i--){
            arr[i]*=pd;
            pd*=nums[i];
            
        }
        return arr;

    }
};
