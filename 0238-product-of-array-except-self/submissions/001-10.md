class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int n=nums.size();
        vector<int>postMultiple(n, 1);
        vector<long long> preMultiple(nums.size(), 1);
        
        for(int i=1;i<nums.size(); i++){
            postMultiple[i]=postMultiple[i-1]*nums[i-1];
        }

        for(int i=nums.size()-2; i>=0; --i){
            preMultiple[i]=1LL*preMultiple[i+1]*nums[i+1];
        }

        for(int i=0; i<postMultiple.size(); i++){
            postMultiple[i]=postMultiple[i]* preMultiple[i];
        }

        return postMultiple;
        
    }
};
