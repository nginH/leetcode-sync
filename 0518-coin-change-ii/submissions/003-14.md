class Solution {


int ans=0;
int dfs(int amount, vector<int>& coins, int idx, int curr){

    if(idx>=coins.size()){
        return 0; // out of bound.
    }

    if(amount==curr){
        ans++;
        return 1;
    }

    if(amount<curr){
        return 0;
    }

    // take;
    curr+=coins[idx];
    int skip= dfs(amount , coins, idx, curr);// start over again ..
    curr-=coins[idx];
    int take= dfs(amount, coins, idx+1, curr);
    return skip+take;
}



public:
    int change(int amount, vector<int>& coins) {
         dfs(amount, coins, 0, 0);
        return ans;
    }
};
