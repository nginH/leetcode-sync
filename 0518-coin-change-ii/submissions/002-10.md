class Solution {


int ans=0;
int dfs(int amount, vector<int>& coins, int idx, int curr,vector<vector<int>> &dp ){

    if(idx>=coins.size()){
        return 0; // out of bound.
    }

    if(amount==curr){
        ans++;
        return 1;
    }

    if(amount<curr){
        return 0;
    }

    if(dp[idx][curr]!=-1){
        return dp[idx][curr]; 
    }

    // take;
    curr+=coins[idx];
    int skip= dfs(amount , coins, idx, curr ,dp);// start over again ..
    curr-=coins[idx];
    int take= dfs(amount, coins, idx+1, curr, dp);  // curr is also changin.. 
    return dp[idx][curr]= skip+take;
}



public:
    int change(int amount, vector<int>& coins) {
        vector<vector<int>> dp(coins.size()+1, vector<int>(50000, -1)); // hopwfully work coz 5000 mac value + max len 5000 so probablue summation will give out of bound 
        return   dfs(amount, coins, 0, 0, dp);
    }
};
