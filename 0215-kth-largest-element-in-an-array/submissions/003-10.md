class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        
        priority_queue<int, vector<int>,less<int>> pq;
        for(auto x:nums){
            pq.push(x);
        }
        int last;
        while(k--){
            
           last= pq.top();
           pq.pop();
        }
        return last;
    }
};
