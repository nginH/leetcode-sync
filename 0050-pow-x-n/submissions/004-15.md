class Solution {
public:
    double myPow(double x, int n) {
         double ans=1;
       
       long long int _n;
       bool isNeg=false;
       if(n<0){
            isNeg=true;
           _n= (long long )(-n);
       }else{
        _n=(long long)n;
       }
        while(_n>0){
            if(_n&1){
                ans=ans*x;
            }
            x=x*x;
            _n>>=1;
        }

        return !isNeg?ans: (1/ans);

    }
};
