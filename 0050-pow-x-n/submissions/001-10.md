class Solution {
    double res;
private: 

    double solve(double x, long long int n){
        if(n==0){
            return 1.0;
        }
        if(x==0){
            return 0;
        }

        res=solve(x,n/2);
        res*=res;
        return n%2==1 ?x*res:res;

    }

public:
    double myPow(double x, int n) {
        long long int k=(long long int)n;
        if(n<0){
            k=-k;
        }

     double val= solve(x,k)  ;
      return n<0?1/val:val; 
    }
};
