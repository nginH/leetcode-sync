class Solution {
public:
    double myPow(double x, int n) {
         double ans=1;
       
       bool isNeg=false;
       if(n<0){
            isNeg=true;
           n=-(n);
       }
        while(n>0){
            if(n&1){
                ans=ans*x;
            }
            x=x*x;
            n>>=1;
        }

        return !isNeg?ans: (1/ans);

    }
};
