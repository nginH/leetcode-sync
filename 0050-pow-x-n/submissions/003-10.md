class Solution {
public:
double myPow(double x, int n) {
        double ans = 1;
        long long _n = n;  

        if (_n < 0) {
            _n = -_n;
            x = 1 / x; 
        }

        while (_n > 0) {
            if (_n & 1) {
                ans *= x;
            }
            x *= x;
            _n >>= 1;
        }

        return ans;
    }
};
