        }

        }

        res=solve(x,n/2);
        res*=res;
        return n%2==1 ?x*res:res;

    }

public:
    double myPow(double x, int n) {
        double val= solve(x,abs(n))  ;
      return n<0?1/val:val; 
    }
};
