class Solution {
public:

    int lb(const vector<int>& arr, int k) {
        int ans = -1;
        int low = 0, high = arr.size() - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (arr[mid] >= k) {
                ans = mid;
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }

        return ans;
    }
    


    vector<int> findRightInterval(vector<vector<int>>& arr) {

        vector<int> ans;
        unordered_map<int,int> mp;
        vector<int> a;


        for(int i=0; i<arr.size();i++ ){
            mp[arr[i][0]]=i; 
            a.push_back(arr[i][0]);
        }

        sort(a.begin(), a.end());

        for(int i=0; i<arr.size(); i++){
            int res= lb(a,arr[i][1]);
            if(res==-1){
                ans.push_back(-1);
            }else{
                ans.push_back(mp[a[res]]);
            }
        }




        return ans;


    }
};
