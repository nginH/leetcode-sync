class Solution {
public:


    vector<int> findRightInterval(vector<vector<int>>& arr) {

        vector<int> ans;
        unordered_map<int,int> mp;
        vector<int> a;


        for(int i=0; i<arr.size();i++ ){
            mp[arr[i][0]]=i; 
            a.push_back(arr[i][0]);
        }

        sort(a.begin(), a.end());

        for(int i=0; i<arr.size(); i++){

            auto res= lower_bound(
                a.begin(),
                a.end(),
                arr[i][1]
                );

            if(res==a.end()){
                ans.push_back(-1);
            }else{
                ans.push_back(mp[*res]);
            }
        }




        return ans;


    }
};
