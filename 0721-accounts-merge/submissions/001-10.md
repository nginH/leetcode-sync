class DSU{
private:
    vector<int> parent, size;
public:
    DSU(int n){
        parent.resize(n+1);
        size.resize(n+1,1);
        for(int i=0; i<=n; i++){
            parent[i]=i;
        }
    }

    int find(int node){ // return parent
        if(parent[node]==node){
            return node;
        }
        return parent[node]= find(parent[node]); 
    }

    void unite(int u, int v){
        int up= find(u);
        int vp= find(v);
        if(up==vp){
            return;
        }

        if(size[up]>size[vp]){
            parent[vp]=up;
            size[up]+=size[vp];
        }else{
            parent[up]=vp;
            size[vp]+=size[up];
        }

    }
};






class Solution {
public:
    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        int n=accounts.size();
        DSU dsu(n);
        unordered_map<string, int>mp; // email->index
        for(int i=0; i<accounts.size(); i++){
            for(int j=1; j<accounts[i].size(); j++){
                if(!mp.contains(accounts[i][j])){
                    mp[accounts[i][j]]=i; 
                }else{
                dsu.unite(i, mp[accounts[i][j]]);
               }
            }
        }

        unordered_map<int, set<string>>groups;
        for(auto &[email, idx]: mp){
            int parent=dsu.find(idx);
            groups[parent].insert(email);
        }
        vector<vector<string>>ans;

        for(auto &[parent, emails]:groups){
            vector<string>temp;
            temp.push_back(accounts[parent][0]);
            for(auto email:emails){
                temp.push_back(email);
            }
            ans.push_back(temp);

        }


        return ans;


   }
};
