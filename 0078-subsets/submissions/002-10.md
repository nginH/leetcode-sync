class Solution {
private:
    vector<vector<int>> ans;
    vector<int>subset;

    void soln(vector<int>& nums, int idx){
        if(idx==nums.size()){
            ans.push_back(subset);
            return;
        };
        subset.push_back(nums[idx]);
        soln(nums,idx+1);

        subset.pop_back();
        soln(nums,idx+1);
    }


public:


    vector<vector<int>> subsets(vector<int>& nums) {

        soln(nums, 0);
        return ans;
    }
};
