#include<bits/stdc++.h>
class Solution {
public:
    int maxProfit(vector<int>& arr) {
        int ans=INT_MIN;
        int minPrice=INT_MAX;

        for(auto pric:arr){
            minPrice=min(minPrice, pric);
            ans= max(ans, pric-minPrice);
        }
        return ans;

    }
};
