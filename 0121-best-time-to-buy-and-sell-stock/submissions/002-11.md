class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int minIdx = 0;
        int maxIdx = 0;
        int minimum = INT_MAX;
        int maximum = INT_MIN;
        for (int i = 0; i < prices.size(); i++) {
            if (minimum > prices[i]) {
                minimum = prices[i];
                minIdx = i;
            }
            if (i > minIdx && maximum < prices[i]) {
                maximum = prices[i];
                maxIdx = i;
            }
        }

        if (maxIdx > minIdx) {
            return maximum - minimum;
        } else {
            return 0;
        }
    }
};
