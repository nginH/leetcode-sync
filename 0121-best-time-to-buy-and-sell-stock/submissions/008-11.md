#include<bits/stdc++.h>
class Solution {
public:
    int maxProfit(vector<int>& p) {
        int ans=INT_MIN;
        int i=0, j=p.size()-1;
        while (i<=j){

           if(p[i]>p[j]){
                i++;
           }else{
                int pf=p[j]-p[i];
                ans=max(ans, pf);
                j--;
           }
        }
        return ans;

    }
};
