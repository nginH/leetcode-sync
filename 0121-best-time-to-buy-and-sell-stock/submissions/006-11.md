#include<bits/stdc++.h>
class Solution {
public:
    int maxProfit(vector<int>& arr) {
        int ans=INT_MIN;
        int i=0, j=arr.size()-1;

        while(i<=j){
            int low=arr[i];
            int high=arr[j];
            ans=max(ans, high-low);
            if(low>high){
                i++;
            }else{
                j--;
            }
        }
        return ans;

    }
};
