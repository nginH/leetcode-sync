class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int minimum = INT_MAX;
        int maximum = INT_MIN;

        for(auto x:prices){
            minimum=min(minimum, x);
            maximum=max(maximum, x-minimum);
        }
        return maximum;
    }
};
