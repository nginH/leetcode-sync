class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int mmin=INT_MIN;
        int mmax=INT_MAX;

        for(int x:prices){
            mmax=min(mmax, x);
            mmin=max(mmin, mmax);
        }

        return (mmax-mmin)<0?0:(mmax-mmin);
    }
};
