int dp[100];

class Solution {
    Solution(){
        memset(dp, -1, sizeof(dp));
    }
public:

    int climbStairs(int n) {
        
        if(dp[n]!=-1){
            return dp[n];
        }

        if(n<=3){
            dp[n]= n;
            return dp[n];
        }else{
            dp[n]= climbStairs(n-1)+ climbStairs(n-2); 
            return dp[n];  
        }
    }

   
};
