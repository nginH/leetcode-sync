int x=0;
int dp[100];

class Solution {
public:

 int h2(int n) {
        
        if(dp[n]!=-1){
            return dp[n];
        }

        if(n<=3){
            dp[n]= n;
            return dp[n];
        }else{
            dp[n]= h2(n-1)+ h2(n-2); 
            return dp[n];  
        }
    }


    int climbStairs(int n) {
        memset(dp, -1, sizeof(dp));
        return h2(n);
    }

   
};
