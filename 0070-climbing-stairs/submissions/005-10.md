int dp[10000];

class Solution {
public:
    void init(){
        memset(dp, -1, sizeof(dp));
    }

    int solve(int n){
        if(n==0){
            return 0;
        }   
        else if(n==1){
            return 1;
        }else if(n==2){
            return 2;
        }else{
            if(dp[n]!=-1){
                return dp[n];
            }
            dp[n]= solve(n-1)+ solve(n-2);
            return dp[n];
        } 
    }
    int climbStairs(int n) {
     init();
     return solve(n);
    }
};
