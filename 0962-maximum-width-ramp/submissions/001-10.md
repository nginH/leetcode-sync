class Solution {
public:
    int maxWidthRamp(vector<int>& nums) {
        int n=nums.size();
        int maxRig[nums.size()];
         maxRig[nums.size()-1]=nums[n-1];
        for(int i=nums.size()-2; i>=0; i--){
            maxRig[i]= max(maxRig[i+1], nums[i]);
        }

        int res=0;
        int i=0,j=0;

        while(i<nums.size() && j< nums.size()){
            if(nums[i]<=maxRig[j]){
                res=max(res, j-i);
                j++;
            }else{
                i++;
            }
        }

        return res;
    }
};
