/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {

private:
    ListNode * merge(ListNode *a, ListNode*b){
    ListNode dummy;
    ListNode* ans = &dummy;

        while(a && b){
            if(a->val<=b->val){
                ans->next=a;
                a=a->next;
            }else{
                ans->next=b;
                b=b->next;
            }
            ans=ans->next;
        }
        // final res;
        if(a){
            ans->next=a;
        }
        if(b){
            ans->next=b;
        }
        return dummy.next;
    }

    ListNode *middle(ListNode *a){
        ListNode *fast=a->next; // pause one before
        ListNode *slow=a;
        while(fast && fast->next){
            slow=slow->next;
            fast=fast->next->next;
        }
        return slow;
    }



public:
  ListNode* sortList(ListNode* head) {
        if (!head || !head->next)
            return head;
        ListNode* mid = middle(head);// point to second mid
        ListNode* second = mid->next; //mid 
        mid->next = nullptr; // split

        ListNode* left = sortList(head);
        ListNode* right = sortList(second);
        return merge(left, right);
    }
};
