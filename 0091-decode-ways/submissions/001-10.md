class Solution {
private:

int sol(string &s, int idx, vector<int> &dp){
    if(idx>=s.size()){
        return 1;
    }
    if(s[idx]=='0'){
        return 0;
    }
    if(dp[idx]!=-1){
        return dp[idx];
    }

    int ans= sol(s, idx+1, dp);// take one
    if (idx+1<s.size()){
        string x=s.substr(idx,2);
        if (stoi(x) >= 10 && stoi(x) <= 26)
            ans+=sol(s, idx+2, dp);
    }
    return dp[idx]=ans;

}

public:
    int numDecodings(string s) {

        vector<int> dp(s.size()+1, -1);
        if(s.size()==0){
            return 0;
        }
        return sol(s,0, dp);
        
    }
};
