class Solution {
private:

int sol(string s, int idx){
    if(idx>=s.size()){
        return 1;
    }
    if(s[idx]=='0'){
        return 0;
    }
    int ans= sol(s, idx+1);// take one
    if (idx+1<s.size()){
        string x=s.substr(idx,2);
        if(s[idx+1] !='0' && stoi(x)<=26)
            ans+=sol(s, idx+2);
    }
    return ans;

}

public:
    int numDecodings(string s) {
        if(s.size()==0){
            return 0;
        }
        return sol(s,0);
        
    }
};
