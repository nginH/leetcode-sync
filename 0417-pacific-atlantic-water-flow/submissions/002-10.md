class Solution {
private:
   int m, n;
    int dirs[4][2] = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}};

    void dfs(const vector<vector<int>>& heights, int r, int c, vector<vector<bool>>& ocean) {
        ocean[r][c] = true; // Mark as reachable & visited

        for (auto& dir : dirs) {
            int dx = r + dir[0];
            int dy = c + dir[1];
            if (dx < 0 || dx >= m || dy < 0 || dy >= n || ocean[dx][dy] || heights[dx][dy] < heights[r][c]) {
                continue;
            }
            dfs(heights, dx, dy, ocean);
        }
    }

public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        if (heights.empty() || heights[0].empty()) return {};
    vector<vector<int>> res;
        m = heights.size();
        n = heights[0].size();

        vector<vector<bool>> pacific(m, vector<bool>(n, false));
        vector<vector<bool>> atlantic(m, vector<bool>(n, false));

        for (int c = 0; c < n; ++c) {
            dfs(heights, 0, c, pacific);         // Top row
            dfs(heights, m - 1, c, atlantic);    // Bottom row
        }

        for (int r = 0; r < m; ++r) {
            dfs(heights, r, 0, pacific);         // Left col
            dfs(heights, r, n - 1, atlantic);    // Right col
        }

        for(int i=0;i<m; i++){
            for(int j=0; j<n; j++){
               if (pacific[i][j] && atlantic[i][j]) {
                    res.push_back({i,j});
                }
            }
        }

      return res;
    }       
};
