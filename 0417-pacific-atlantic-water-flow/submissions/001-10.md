class Solution {
int m, n;

vector<pair<int, int>>dir={
    {0,1},
    {1,0},
    {-1, 0},
    {0, -1}
};

void dfs(vector<vector<int>>& heights , int row , int col, int prevHig, vector<vector<bool>>& visited){
    if(row<0|| col<0 || row>=m || col>=n){
        return;
    }
    if(heights[row][col]<prevHig){
        return;
    }
    if(visited[row][col]==true){
        return; // already visited..
    }
    visited[row][col]=true;
    for(auto &[dx, dy]:dir){
        dfs(heights, row+dx, col+dy,heights[row][col], visited);
    }
}



public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        m=heights.size();
        n=heights[0].size();
        vector<vector<bool>>pacific(m, vector<bool>(n, false));
        vector<vector<bool>>atlantic(m, vector<bool>(n, false));

        for(int j=0; j<n ; j++){
            dfs(heights, 0,j, INT_MIN, pacific);
            dfs(heights, m-1,j, INT_MIN, atlantic);
        }

        for(int i=0; i<m ; i++){ // col
            dfs(heights, i,0, INT_MIN, pacific);
            dfs(heights, i,n-1, INT_MIN, atlantic);
        }
        vector<vector<int>> ans;
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++ ){
                if(atlantic[i][j] && pacific[i][j]){
                    ans.push_back({i, j});
                }
            }
        }
        return ans;


    }
};
