class Solution {
private:
    vector<pair<int, int>>direction={
        {0,1},
        {1,0},
        {-1,0},
        {0,-1}
    };
    int m;
     int n;
    void dfs(vector<vector<int>>& heights,int i, int j,  set<pair<int, int>> &vis, int PrevHight){
        if(i<0 || j<0 || i>=m || j>=n || vis.contains({i,j}) || heights[i][j]<PrevHight){
            return;
        }
        vis.insert({i,j});
        for(auto [dx, dy]:direction){
            dfs(heights, i+dx,j+dy,vis,heights[i][j]);
        }
    }

public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        if (heights.empty() || heights[0].empty()) return {};
    vector<vector<int>> res;
        set<pair<int, int>> pacific;
        set<pair<int, int>> atlantic;
        m = heights.size();
        n = heights[0].size();

        for(int i=0; i<m; i++){
            dfs(heights , 0,  i, pacific, heights[0][i]);
            dfs(heights , m-1,i, atlantic, heights[m-1][i]);
        }

        for(int i=0; i<n; i++){
            dfs(heights , i,  0, pacific, heights[i][0]);
            dfs(heights , i,  n-1, atlantic, heights[i][n-1]);
        }

        for(int i=0;i<m; i++){
            for(int j=0; j<n; j++){
                if (pacific.contains({i,j}) && atlantic.contains({i,j})){
                    res.push_back({i,j});
                }
            }
        }

      return res;
    }       
};
