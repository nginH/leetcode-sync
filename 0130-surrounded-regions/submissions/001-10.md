class Solution {

    vector<pair<int, int>> dir = {
        {1, 0},
        {0, 1},
        {-1, 0},
        {0, -1},
    };

    void dfs(vector<vector<char>>& mat, int row, int col) {
        if (row < 0 || col < 0 || row >= mat.size() || col >= mat[0].size()) {
            return;
        }
        // to prevent circular
        if (mat[row][col] == 'X' || mat[row][col] == '-') {
            return;
        }
        mat[row][col] = '-'; // mark visited ..
        for (auto [dx, dy] : dir) {
            dfs(mat, row + dx, col + dy);
        }
    }

public:
    void solve(vector<vector<char>>& board) {

        int rows = board.size();
        int cols = board[0].size();

        for (int i = 0; i < rows; i++) {
            dfs(board, i, 0);
            dfs(board, i, cols - 1);
        }

        for (int j = 0; j < cols; j++) {
            dfs(board, 0, j);
            dfs(board, rows - 1, j);
        }

        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                if (board[i][j] == 'O') {
                    board[i][j] = 'X';
                } else if (board[i][j] == '-') {
                    board[i][j] = 'O';
                }
            }
        }
    }
};
