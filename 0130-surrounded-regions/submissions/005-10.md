class Solution {
private:
vector<pair<int, int>> dir={
    {0,1},
    {1,0},
    {-1, 0},
    {0,-1}
};
int m,  n;

void dfs(vector<vector<char>>& board, int i, int j){
    if(i<0 || j<0 || i>=m || j>=n){
        return;
    }
    if(board[i][j]!='O'){
        return;
    }
    board[i][j]='#'; // marking connected to boarder..

    for(auto &[dx, dy]:dir){
        dfs(board, dx+i, dy+j);
    }
}



public:
    void solve(vector<vector<char>>& board) {
        m=board.size();
        n=board[0].size();
        // top
        for(int r=0; r<m; r++){
            dfs(board, r, 0);
            dfs(board, r, n-1);
        }

         for(int c=0; c<n; c++){
            dfs(board,0, c);
            dfs(board, m-1, c);
        }

        for(int i=0; i<m; i++){

            for(int j=0; j<n; j++){
                if(board[i][j]=='O'){
                    board[i][j]='X';
                }
                if(board[i][j]=='#'){
                    board[i][j]='O';
                }
            }
        }
        
    }
};
