```cpp
class Solution {

vector<pair<int, int>>dir={
    {1,0},
    {0,1},
    {-1,0},
    {0,-1}
};
int m,n;
void dfs(int row, int col,vector<vector<char>>& board){
    if(row<0|| col<0 || row>=m|| col>=n){
        return;// out of bound;
    }
    if(board[row][col]=='X' || board[row][col]=='#'){
        return;
    }
    board[row][col]='#';
    for(auto &[dx, dy]:dir){
        dfs(row+dx, col+dy, board);
    }
}


public:
    void solve(vector<vector<char>>& board) {
        m=board.size();
        n=board[0].size();
        for(int i=0; i<m; i++){// from row boarder..
            dfs(0,i, board);
            dfs(m-1,i, board);
        }

        for(int i=0; i<n; i++){// from row boarder..
            dfs(i,0, board);
            dfs(i,n-1, board);
        }

        // ---------------------- 

        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(board[i][j]=='#'){
                    board[i][j]='O';
                }else if(board[i][j]=='O'){
                    board[i][j]='X';
                }
            }
        }
    }
};
```
