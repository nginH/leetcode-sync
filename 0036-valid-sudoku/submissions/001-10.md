class Solution {
bool sol(vector<vector<char>>& board, int row , int col){

    // row check
    char c=board[row][col];
    for(int i=0; i<9; i++){
        if(c==board[i][col] && i!=row){
            return false;
        }
    }
    //col
    for(int i=0; i<9; i++){
        if(c==board[row][i] && i!=col){
            return false;
        }
    }
    // box

    // row =5 col 5
    // i=4 .. 

    int iStart=(row/3)*3;
    int jStart=(col/3)*3;
    for(int i=iStart; i<3+iStart ;i++){
        for(int j=jStart; j<3+jStart ; j++){
                if (board[i][j] == c && !(i == row && j == col))
                    return false;
                }

    }
    return true;
}



public:
    bool isValidSudoku(vector<vector<char>>& board) {
        
        for(int i=0; i<9; i++){
            for(int j=0; j<9; j++){
                if(board[i][j]!='.')
                    if(sol(board, i, j)==false){
                        return false;
                    }
            }
        }
        return true;
    }
};
