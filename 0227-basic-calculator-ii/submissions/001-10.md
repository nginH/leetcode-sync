    int calculate(string s) {
        s+='#'; // mark as end//

        stack<int>st;
        string value="";

        for(int i=0; i<s.size(); i++){
            if(isdigit(s[i])){
                num=num*10+(s[i]-'0');
            }
            if(!isdigit(s[i]) && s[i]!=' ' || i==s.size()-1){
                int k;
                switch(op){
                    case '+':
                        st.push(num);
                        break;
                    case '-':
                        st.push(-num);
                        break;
                    case '*':
                         k=st.top();
                        st.pop();
                        st.push(num*k);
                        break;
                    case '/':
                          k=st.top();
                        st.pop();
                        st.push(k/num);
                        break;                    
                }
                op=s[i];
                num=0;
            }
        }
        int ansp=0;
        while(!st.empty()){
            ansp+=st.top();
            st.pop();
        }
        return ansp;


    }
};
