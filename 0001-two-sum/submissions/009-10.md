class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {

        unordered_map<int, int>m;
        
        for(int i=0; i<nums.size(); i++){
            int rem= target-nums[i];
            auto it=m.find(rem);
            if(it!=m.end()){
                vector<int>v= {i, it->second};
                sort(v.begin(), v.end());
                return v; // as per question its sure the answer exixt
            }
            m[nums[i]]=i;
        }
        return {-1,-1};

    }
};
