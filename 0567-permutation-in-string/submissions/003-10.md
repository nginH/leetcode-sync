class Solution {
public:
    bool checkInclusion(string s1, string s2) {

            if(s1.size()>s2.size()){
                return false;
            }
        
            unordered_map<char, int>s1mp;
            for(auto x:s1){
                s1mp[x]++;
            }


            for(int right=0;right<s2.size();right++){
                if(s1mp.contains(s2[right])){
                    unordered_map<char,int> temp=s1mp;
                    int left=right;
                    int sz=s1.size();
                    while(sz--){
                        if(!temp.contains(s2[left])){
                            break;
                        }
                        temp[s2[left]]--; // value not exist it'll add neg value increment
                        if(temp[s2[left]]==0){
                            temp.erase(s2[left]);
                        }
                        left++;
                    }
                    if(temp.size()==0){
                        return true;
                    }
                }
            }

        
            return false;

    }
};
