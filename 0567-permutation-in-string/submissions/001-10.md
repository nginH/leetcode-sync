class Solution {
public:
    bool checkInclusion(string s1, string s2) {

            if(s1.size()>s2.size()){
                return false;
            }
        
            unordered_map<char, int>need;
            for(auto x:s1){
                need[x]++;
            }
            unordered_map<char, int>window;
            for(int i=0; i<s1.size(); i++){
                window[s2[i]]++;
            }
            if(need==window){
                return true;
            }
            int left=0;
            for(int i=s1.size(); i<s2.size(); i++){    
                window[s2[i]]++;

                window[s2[left]]--;
                if(window[s2[left]]==0){
                    window.erase(s2[left]);
                }
                left++;
                if(need==window){
                    return true;
                }
            }
            return false;

    }
};
