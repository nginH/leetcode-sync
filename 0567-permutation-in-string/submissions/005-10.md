class Solution {
public:
    bool checkInclusion(string s1, string s2) {

            if(s1.size()>s2.size()){
                return false;
            }

            unordered_map<char, int> mp1, mp2;

            for(auto x:s1){
                mp1[x]++;
            }

            for(int i=0; i<s1.size(); i++){
                mp2[s2[i]]++;
            }

            for(int i=0; i<=s2.size()-s1.size(); i++){ //(o(n))
                if(mp1==mp2){ // 26 char comp o(1)
                    return true;
                }
                mp2[s2[i+s1.size()]]++;
                mp2[s2[i]]--;
                if(mp2[s2[i]]<=0){
                    mp2.erase(s2[i]);
                }
            }

            return false;

    }
};
