class Solution {
public:
    int characterReplacement(string s, int k) {
        
        unordered_map<char , int> mp;
        int ans=0, left=0;

        int maxFreq=0;
        for(int r=0; r<s.size(); r++){
            mp[s[r]]++; // increasing the freq count;
            maxFreq=max(maxFreq, mp[s[r]]);

            while((r-left+1)-maxFreq>k ){
                mp[s[left]]--;
                left++;
            }
            ans=max(ans, r-left+1);   
        }

        return ans;
    }
};
