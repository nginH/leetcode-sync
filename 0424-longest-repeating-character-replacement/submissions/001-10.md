class Solution {
public:
    int characterReplacement(string s, int k) {
        
        unordered_map<char , int> mp;
        int left=0; int right=0;
        int n=s.size();
        int ans=0;

        int maxFreqofKey=0;

        for(right=0; right<n;right++){
            mp[s[right]]++;
            maxFreqofKey=max(maxFreqofKey, mp[s[right]]);
            while((right-left+1)-maxFreqofKey>k){
                mp[s[left]]--;

                if(mp[s[left]]==0){
                    mp.erase(s[left]);
                }
                left++;

                maxFreqofKey=0;
                for(auto x:mp){
                    maxFreqofKey=max(maxFreqofKey, x.second);
                }

            }




            ans=max(ans, right-left+1);
        }
        return ans;

    }
};
