class Solution {
public:
    long long weightedSum(vector<int>& parent, vector<int>& nums) {
        int n=parent.size();
        vector<vector<int>>adj(n);

        for(int i=1; i<parent.size(); i++){
            adj[parent[i]].push_back(i); 
        }
        int h=adj.size();
        /*
         which gives 0=[5]
            1->[1, 2, 3] // level one 
            2->[4, 6]
         */

        queue<int>q;
        q.push(0);

        long long ans=0;
        long long sum=0;
        /*
         E(nums[i])+h-d+1
         E(nums[i])h-E(nums[d])+E(nums[i])

         E(nums[i])*h
         */
        for (int x : nums) // E(nums[i])*h
            sum += x;

        int d=0;
         while(!q.empty()){
            int sz=q.size();
            d++;
            while(sz--){
                auto node =q.front();
                q.pop();

                ans+=1LL*nums[node]*(1-d);//  -E(nums[d])+E(nums[i])

                for(auto &kid:adj[node]){
                    q.push(kid);
                }
            }
         }
         long long kkk=1LL*sum*d ;// E(nums[i])h 
          ans=ans+ kkk; // E(nums[i])h -E(nums[d])+E(nums[i])
        return ans;
    }
};
