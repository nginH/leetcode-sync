class Solution {
public:
    bool isAnagram(string s, string t) {
        unordered_map<char, int> smp; // hash map
        unordered_map<char,int>tmp;
        if(s.size()!=t.size()){
            return false;
        }
        for(int i=0; i< s.size(); i++){
            smp[s[i]]++;
            tmp[t[i]]++;
        }

        return smp==tmp;

    }
};
