class Solution {
public:
    bool isAnagram(string s, string t) {
        unordered_map<char, int> mp; // hash map
        for(auto x:s){
            mp[x]++;
        }

        for(auto x:t){
            if(mp.contains(x)){
                mp[x]--;
            }else{
                return false;
            }
        }

        for(auto x:mp){
            if(x.second!=0){
                return false;
            }
        }
        return true;
    }
};
