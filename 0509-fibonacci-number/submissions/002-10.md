class Solution {
    int dp[31];
public:
    Solution(){
        for(int i=0; i<31; i++){
            dp[i]=-1;
        }
    }
   
    int fib(int n) {

        if(n==0){
            return 0;
        }else if(n==1){
            return 1;
        }
        if(this->dp[n]!=-1){
            return this->dp[n];
        }

        this->dp[n]=fib(n-1)+fib(n-2);
        return this->dp[n];
    }
};
