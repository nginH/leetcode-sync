class Solution {
public:
    bool isVowel(char c){
        return c=='a'|| c=='e'||c=='i'|| c=='o'||c=='u';
    }
    int countVowelSubstrings(string word) {
        unordered_map<char, int>mp;
    
        int ans=0;
        for(int i=0; word[i];i++){
            unordered_map<char, int> freq;
            for(int j=i; j<word.size(); j++){
                if(!isVowel(word[j])) break;
                freq[word[j]]++;
                if(freq.size()==5){
                    ans++;
                }
            } 
        }
        return ans;
    }
};
