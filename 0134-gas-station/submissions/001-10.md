class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int sum=0;
        int tank=0;
        int fir=0;
        for(int i=0; i<gas.size(); i++){
            int val=gas[i]-cost[i];
            sum+=val;
            tank+=val;
            if(tank<0){
                fir=i+1;
                tank=0;
            }

        }
        return sum>=0?fir:-1;
    }
};
