class Solution {

bool isPalindrom(string s){
    int i=0; 
    int j=s.size()-1;
    while(i<j){
        if(s[i]!=s[j]){
            return false;
        }
        i++;
        j--;
    }
    return true;
}

int ans=0;
void dfs(string s, int idx, string curr){
   
    if(isPalindrom(curr)){
        ans=max(ans, (int)curr.size());
    }
     if(idx>=s.size()){
        return;
    }
    curr.push_back(s[idx]);
    dfs(s,idx+1, curr);// take.
    curr.pop_back();
    dfs(s, idx+1, curr);// skip

}


public:
    int longestPalindromeSubseq(string s) {
        string curr="";
        dfs(s, 0,curr);
        return ans;
    }
};
