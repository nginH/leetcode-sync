class Solution {

    int dfs(string& s, int left, int right,
            vector<vector<int>>& dp) {

        if (left > right)
            return 0;

        if (left == right)
            return 1;

        if (dp[left][right] != -1)
            return dp[left][right];

        if (s[left] == s[right]) {
            return dp[left][right] =
                2 + dfs(s, left + 1, right - 1, dp);
        }

        return dp[left][right] =
            max(
                dfs(s, left + 1, right, dp),
                dfs(s, left, right - 1, dp)
            );
    }

public:
    int longestPalindromeSubseq(string s) {

        int n = s.size();

        vector<vector<int>> dp(
            n,
            vector<int>(n, -1)
        );

        return dfs(s, 0, n - 1, dp);
    }
};
