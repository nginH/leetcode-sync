class Solution {
public:
    int trap(vector<int>& height) {
        int ans=0;
        int n= height.size();

        vector<int> preMax(n);
        vector<int> postMax(n);
        int a=INT_MIN;
        int b=INT_MIN;
        for(int i=0; i<n; i++){
            a=max(a, height[i]);
            b=max(b, height[n-i-1]);
            preMax[i]=a;
            postMax[n-1-i]=b;
        }


        for(int i=0; i<n ; i++){
            ans+=min(postMax[i], preMax[i])-height[i];
        }
        return ans;





        return ans;
    }
};
