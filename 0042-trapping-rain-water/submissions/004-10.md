class Solution {
public:
    int trap(vector<int>& height) {
        int n=height.size();
        int preSum[n];
        int postSum[n];
        preSum[0]=height[0];
        postSum[n-1]=height[n-1];
        int j=n-2;
        for(int i=1; i<n; i++){
            preSum[i]=max(preSum[i-1], height[i]);
            postSum[j]=max(postSum[j+1], height[j]);
            j--;
        }
        int ans=0;
        for(int i=0; i<n; i++){
            ans+=min(preSum[i], postSum[i])-height[i];
        };
        return ans;
    }
};
