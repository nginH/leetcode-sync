class Solution {


vector<int>leftMax(vector<int>&height){
    int n=height.size();
    vector<int>res(n);
    res[0]=height[0];
    for(int i=1; i<n ; i++){
        res[i]=max(res[i-1], height[i]);
    }
    return res;
}


vector<int>RightMax(vector<int>&height){
    int n=height.size();
    vector<int>res(n);
    res[n-1]=height[n-1];
    for(int i=n-2; i>=0 ; i--){
        res[i]=max(res[i+1], height[i]);
    }
    return res;
}


public:
    int trap(vector<int>& height) {
        int n=height.size();
        vector<int>preMax(n,0);
        vector<int>postMax(n,0);
        preMax=leftMax(height);
        postMax=RightMax(height);

        int res =0;
        for(int i=1; i<n; i++){
            res+=min(preMax[i], postMax[i])-height[i] ;
        }
        return res;


    }
};
