class Solution {
public:
    void rotate(vector<int>& nums, int k) {
       // brute force;
        
        vector<int> arr={nums.begin(), nums.end()};
        int j=0;
       
        for(int i=nums.size()-k; i<nums.size(); i++){
                nums[j++]= arr[i];
        }

        for(int i=0; i<(nums.size()-k)%nums.size(); i++){
            nums[j++]=arr[i];
        }
    }
};
