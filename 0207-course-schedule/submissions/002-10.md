class Solution {

private:

unordered_map<int, int> mp;

bool dfs(vector<vector<int>>& adj, vector<bool>&pvis, vector<bool> &vis, int node){
    pvis[node]=true;
    vis[node]=true;

    for(auto x:adj[node]){
        if(pvis[x]==true){ // node already visited directly return... cycle detected
            return true;
        }
        if(!vis[x]){ // node not visted yet explore ..
            if(dfs(adj, pvis, vis, x)){
                return true;
            }
        }
    }
    pvis[node]=false;
    return false;
   
}


public:
    bool canFinish(int n, vector<vector<int>>& prerequisites) {
        vector<vector<int>> adj(n);
        vector<int> inDegree(n, 0);
        for(auto x:prerequisites){
            adj[x[1]].push_back(x[0]);
            inDegree[x[0]]++;
        }


        queue<int>q;
        for(int i=0; i<n; i++){
            if(inDegree[i]==0){
                q.push(i);  // node with 0 indegreee 
            }
        }


        int processNode=0;
        while(!q.empty()){
            int u=q.front();
            q.pop();
            processNode++;
            for(auto x:adj[u]){
                inDegree[x]--;
                if(inDegree[x]==0){
                    q.push(x);
                }
            }
        }
        return processNode==n;

    }
};
