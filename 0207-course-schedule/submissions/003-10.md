class Solution {

private:

unordered_map<int, int> mp;

bool dfs(vector<vector<int>>& adj, vector<bool>&pvis, vector<bool> &vis, int node){
    pvis[node]=true;
    vis[node]=true;

    for(auto x:adj[node]){
        if(pvis[x]==true){ // node already visited directly return... cycle detected
            return true;
        }
        if(!vis[x]){ // node not visted yet explore ..
            if(dfs(adj, pvis, vis, x)){
                return true;
            }
        }
    }
    pvis[node]=false;
    return false;
   
}


public:
    bool canFinish(int n, vector<vector<int>>& prerequisites) {
        vector<vector<int>> adj(n+1);
        for(auto x:prerequisites){
            adj[x[0]].push_back(x[1]);
        }

        vector<bool> vis(n+1,false);
        vector<bool> pvis(n+1, false);
        for(int i=0; i<n ; i++){
            if(vis[i]==false){
                // not vis
                if(dfs(adj, pvis, vis, i)==true){
                    // cycle
                    return false;
                }
            }
        }


        return true;
    }
};
