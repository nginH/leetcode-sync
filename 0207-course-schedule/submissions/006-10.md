class Solution {
public:
    bool canFinish(int n, vector<vector<int>>& prerequisites) {
        vector<vector<int>>adj(n);
        vector<int> indegree(n, 0);

        for(auto x:prerequisites){
            adj[x[0]].push_back(x[1]);
            indegree[x[1]]++;
        }

        queue<int>q;

        for(int i=0; i<n; i++){
            if(indegree[i]==0){
                q.push(i);
            }
        }
        vector<int> topo;
        while(!q.empty()){
            int k=q.front();
            q.pop();
            topo.push_back(k);
            for(auto x:adj[k]){
                indegree[x]--;
                if(indegree[x]==0){
                    q.push(x);
                }
            }

        }

        return topo.size()==n;
    }
};
