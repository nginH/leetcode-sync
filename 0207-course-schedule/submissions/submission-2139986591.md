class Solution {

    bool canF(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int> indegree(numCourses);
        vector<vector<int>> adj(numCourses);
        for (auto x : prerequisites) {
            adj[x[1]].push_back(x[0]);
            indegree[x[0]]++;
        }

        queue<int> q;

        for (int i = 0; i < indegree.size(); i++) {
            if (indegree[i] == 0) {
                q.push(i);
            }
        }
        int level = 0;
        while (!q.empty()) {
            level++;
            int k = q.front();
            q.pop();
            for (auto x : adj[k]) {
                indegree[x]--;
                if (indegree[x] == 0) {
                    q.push(x);
                }
            }
        }

        return level == numCourses;
    }

public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        return canF(numCourses, prerequisites);
    }
};
