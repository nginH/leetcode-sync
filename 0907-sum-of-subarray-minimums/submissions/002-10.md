class Solution {
public:
    int sumSubarrayMins(vector<int>& arr) {
        int ans=0;
        const int MOD=1e9+7;
        stack<int> st;

        for(int i=0; i<=arr.size(); i++){
            while(!st.empty() && (i==arr.size() || arr[st.top()]>=arr[i])){
                int mid=st.top();
                st.pop();
                int left= st.empty()?mid+1: mid-st.top();
                int right= i-mid;

                ans=(ans + 1LL * arr[mid]* left*right)%MOD;
            }
            st.push(i);
        }
        return ans;

    }
};
