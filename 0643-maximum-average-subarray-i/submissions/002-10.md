class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        double ans=INT_MIN;


        double sum=0;
        int i=0;
        int j=0;

        while(j<nums.size()){
            if(j-i>=k){
                sum-=nums[i];
                i++;
            }
            sum+=nums[j];
            j++;
            if(j-i==k)
                ans=max(ans, sum/(k*1.0));
        }

        return ans;



    }
};
