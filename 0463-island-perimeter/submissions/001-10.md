class Solution {

private:
    int m, n;

    int dfs(vector<vector<int>>& grid, int i, int j){
        if(i<0 ||j<0 || i>=m ||j>=n){
            return 1;
        }

        if(grid[i][j]==0){
            return 1;
        }

        if(grid[i][j]==-1){
            return 0;
        }
        grid[i][j]=-1;

        return dfs(grid, i+1, j)+
            dfs(grid, i-1, j)+
            dfs(grid, i, j+1)+
            dfs(grid, i, j-1);
    }




public:
    int islandPerimeter(vector<vector<int>>& grid) {
         m=grid.size();
         n=grid[0].size();
       for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(grid[i][j]==1){
                    return dfs(grid, i, j);
                }
            }
       } 
       return 0;
    }
};
