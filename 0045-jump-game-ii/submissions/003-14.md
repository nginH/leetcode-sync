class Solution {
public:

    int f(vector<int> &arr,int idx, int jump){

        if(idx>= arr.size()-1){
            return jump;
        }else{
            int k=INT_MAX;
            for(int i=1; i<=arr[idx]; i++){
                k=min(f(arr, idx+i, jump+1), k);
            }
            return k;
        }
    }


    int jump(vector<int>& nums) {
        return f(nums, 0, 0);
    }
};
