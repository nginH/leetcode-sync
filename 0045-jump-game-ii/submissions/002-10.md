class Solution {
public:
    vector<int> dp;

    int f(vector<int>& nums, int idx) {
        if (idx >= nums.size() - 1) {
            return 0;
        }

        if (dp[idx] != -1) {
            return dp[idx];
        }

        int ans = INT_MAX;

        for (int jump = 1; jump <= nums[idx]; jump++) {
            if (idx + jump < nums.size()) {
                int temp = f(nums, idx + jump);

                if (temp != INT_MAX) {
                    ans = min(ans, 1 + temp);
                }
            }
        }

        return dp[idx] = ans;
    }

    int jump(vector<int>& nums) {
        dp.assign(nums.size(), -1);
        return f(nums, 0);
    }
};
