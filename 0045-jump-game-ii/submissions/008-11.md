class Solution {
public:
    int jump(vector<int>& nums) {
        

        int reach=0;
        int ct=-1;

        for(int i=0; i<nums.size(); i++){
            if(reach<i){
                return -1;
            }else{
                int t=reach;
                reach=max(reach, nums[i]+i);
                if(reach != t){
                    // reach is updated;
                    ct++;
                }
            }
        }
        return ct;
    }
};
