class Solution {
public:
    int jump(vector<int>& nums) {
    queue<int>q;

    vector<bool>visited(nums.size(), false);
    q.push(0); // starting .
    visited[0]=true;

    int steps=0;
    int n=nums.size();
    while(!q.empty()){
        int sz=q.size();
        while(sz--){
            int idx=q.front();
            q.pop();

            if(idx==n-1){
                return steps;
            }

            int fur= min(n-1, idx+nums[idx]);

            for(int i=idx+1; i<=fur && i<nums.size() ; i++){
                if(!visited[i]){
                    visited[i]=true;
                    q.push(i);
                }
            }


        }
        steps++;
    }      
    return -1;
    }
};
