class Solution {

vector<pair<int, int>>dir{
    {0,1},
    {1,0}
};
int m, n;
int dfs(vector<vector<int>>& obstacleGrid, int row, int col, vector<vector<int>>&dp){
  if(row<0 || col<0 || row>=m || col>=n){
    return 0;
  }
  if(obstacleGrid[row][col]==1 || obstacleGrid[row][col]==2){ // 1 for obstical and # for visited..
    return 0; 
  }
  if(row==m-1 && col==n-1){
    return 1;
  }
  if(dp[row][col]!=-1){
    return dp[row][col];
  }
  int ans=0;
  for(auto [dx, dy]:dir){
    obstacleGrid[row][col]=2;
    ans+=dfs(obstacleGrid, row+dx, col+dy, dp);
    obstacleGrid[row][col]=0; // backtrack
  }

  return dp[row][col]= ans;

}


public:
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        m=obstacleGrid.size();
        n=obstacleGrid[0].size();
        vector<vector<int>>dp(m+1, vector<int>(n+1, -1));
        return dfs(obstacleGrid, 0, 0, dp);


    }
};
