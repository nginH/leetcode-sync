class Solution {

vector<vector<int>>ans;
void solve(vector<int>& candidates, int idx, int target, vector<int>&curr){
    if(target==0){
        ans.push_back(curr);
        return;
    }
    if(idx>=candidates.size() || target<=0){
        return;
    }
    curr.push_back(candidates[idx]);
    solve(candidates , idx,target-candidates[idx], curr);// take
    curr.pop_back();
    solve(candidates, idx+1, target, curr);
}


public:
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        vector<int>curr;
        solve(candidates, 0, target, curr);
       return ans; 
    }
};
