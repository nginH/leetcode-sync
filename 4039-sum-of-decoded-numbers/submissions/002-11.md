class Solution {


    int pow(int a, int b){
        int MOD=1000000007;
        a%=MOD;
        long long ans=1;
        while(b>0){
            if(b&1)
                ans= (ans*a)%MOD;
            a=(1LL*a*a)%MOD;
            b>>=1;
        }
        return (long long ) ans;
    }

    
public:
    int sumDecoded(vector<long long>& nums) {
      long long ans=0;
        for(auto x:nums){
            int w=x%10;
            long long  d=floor(x/10);
            string t=to_string(d);
            int x1=stoi(t.substr(0, w));
            int y= stoi(t.substr(w, t.size()-w));
            ans+=pow(1LL*x1, y);
        }
        return ans;
    }
};
