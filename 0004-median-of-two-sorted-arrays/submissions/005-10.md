class Solution {
public:
    double findMedianSortedArrays(vector<int>& n1, vector<int>& n2) {
        vector<int> fArr;
        int i=0, j=0;
        while(i!= n1.size()  && j!=n2.size()){
            if(n1[i]<n2[j]){
                fArr.push_back(n1[i]);
                i++;
            }else{
                fArr.push_back(n2[j]);
                j++;
            }
        }

        for(int k=i; k<n1.size(); k++){
            fArr.push_back(n1[k]);
        }
        for(int k=j; k<n2.size(); k++){
            fArr.push_back(n2[k]);
        }

        if(fArr.size()%2){
            return fArr[ceil(fArr.size()/2)];
        }else{
            int x=(fArr.size()/2 )-1;
            int y=x+1;
            float a= fArr[x];
             float b= fArr[y];

            return float((a+b)/2);
        }


    }
};
