class Solution {
public:
    double findMedianSortedArrays(vector<int>& n1, vector<int>& n2) {
        if(n1.size()>n2.size()){
            return findMedianSortedArrays(n2,n1);
        }
        int m=n1.size();
        int n=n2.size();

        int left=0;
        int right=m;

        int half= (m+n+1)/2;
        
        while(left<=right){

            int i= left+ (right-left)/2;
            int j=half-i;

            int n1left=(i==0) ?INT_MIN:n1[i-1];
            int n1right=(i==m)?INT_MAX:n1[i];

            int n2left=(j==0)?INT_MIN:n2[j-1];
            int n2right=(j==n)?INT_MAX:n2[j];

            if(n1left<=n2right && n2left<=n1right){
                //odd 
                if((m+n)%2==1){
                    return max(n1left, n2left);
                }

                //even
                int leftMax=max(n1left, n2left);
                int rmax=min(n1right, n2right);
                return (leftMax+rmax)/2.0;
            }
            if(n1left>n2right){
                right=i-1;
            }else{
                left=i+1;
            }
        }
        return 0.0;
    }
};
