class Solution {
public:
    int maxVowels(string s, int k) {
        unordered_set<char>st={'a', 'e','i','o','u'};
        
        int ans=0;
        int maxAns=0;
        for(int i=0; i<k; i++){
            if(st.contains(s[i])){
                ans++;
            }
        }
        maxAns=ans;
        int left=0;
        for(int right=k; right<s.size(); right++){

            if(st.contains(s[right])){
                ans++;
            };
            if(st.contains(s[left])){
                ans--;
            }


            left++;
            maxAns=max(maxAns, ans);
        }
    
        return maxAns;



    }
};
