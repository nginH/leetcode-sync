class Solution {


bool isVowel(char c) {
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || 
           c == 'u';
}
public:
    int maxVowels(string s, int k) {
        
        int ans=0;
        int maxAns=0;
        for(int i=0; i<k; i++){
            if(isVowel(s[i])){
                ans++;
            }
        }
        maxAns=ans;
        int left=0;
        for(int right=k; right<s.size(); right++){

            if(isVowel(s[right])){
                ans++;
            };
            if(isVowel(s[left])){
                ans--;
            }

            left++;
            maxAns=max(maxAns, ans);

            if(maxAns==k){
                return maxAns;// all volel .. no further need to check
            }
        }
    
        return maxAns;



    }
};
