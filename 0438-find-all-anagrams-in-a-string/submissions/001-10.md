class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        if(s.size()<p.size()){
            return {};
        }
        unordered_map<char,int>need;
        for(auto x:p){
            need[x]++;
        }
        unordered_map<char, int> window;
        for(int i=0; i<p.size(); i++){
            window[s[i]]++;
        }
        vector<int>ans;
        if(need==window){
            ans.push_back(0); // match at index 0
        }
        int left=0;
        for(int i=p.size(); i<s.size(); i++){
            window[s[i]]++;
            window[s[left]]--;
            if(window[s[left]]==0){
                window.erase(s[left]);
            }
            left++;
            if(need==window){
                ans.push_back(left);
            }
        }
        return ans;
    }
};
