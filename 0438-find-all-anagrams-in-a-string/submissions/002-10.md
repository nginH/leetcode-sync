class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        vector<int> ans;
        if(s.size()<p.size()){
            return {};
        }


        unordered_map<char, int> mp1 , mp2;

        for(auto x:p){
            mp1[x]++;
        }

        for(int i=0; i<p.size(); i++){
            mp2[s[i]]++;
        }


        for(int i=0; i<=s.size()-p.size(); i++){
            if(mp1==mp2){
                ans.push_back(i);
            }
            mp2[s[i]]--; // remove first index
            if(mp2[s[i]]==0){
                mp2.erase(s[i]);
            }
            mp2[s[i+p.size()]]++; // add last one
        }
        return ans;
    }
};
