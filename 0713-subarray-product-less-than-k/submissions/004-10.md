class Solution {
int ans=0;
void dfs(vector<int>& nums, int k , long long currProd, int idx){

    if(idx>=nums.size()){
        return;
    }

    currProd*=nums[idx];
    if(currProd<k){// strictly less
        ans++;
        dfs(nums, k, currProd, idx+1);
    }
}


public:
    int numSubarrayProductLessThanK(vector<int>& nums, int k) {

        for(int i=0; i<nums.size(); i++)
            dfs(nums, k, 1, i);
        return ans;
    };
};
