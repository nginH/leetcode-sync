class Solution {
public:
    int count=0;
    void f(vector<int> &nums, int k, int idx, int prod){
        if(idx>=nums.size()){
            return ;
        }else{
            prod*=nums[idx];
            if(prod<k){
                count++;
                f(nums, k, idx+1, prod);
            }
        }
    }


    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        if(k<=1)return 0;
        int prod=1;
        int l=0;
        for(int i=0; i<nums.size(); i++){
            prod*=nums[i];
            while(prod>=k){
                prod/=nums[l];
                l++;
            }
            count+=(i-l+1);

        }

        return count;
    }
};
