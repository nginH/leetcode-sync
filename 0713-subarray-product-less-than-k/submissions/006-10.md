class Solution {
public:
    int count=0;
    void f(vector<int> &nums, int k, int idx, int prod){
        if(idx>=nums.size()){
            return ;
        }else{
            prod*=nums[idx];
            if(prod<k){
                count++;
                f(nums, k, idx+1, prod);
            }
        }
    }


    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        if(k<=1)return 0;

        for(int i=0; i<nums.size(); i++){
            int prod=1;
            for(int j=i; j<nums.size(); j++){
                prod*=nums[j];
                if(prod<k){
                    count++;
                }else{
                    break;
                }
            }
        }

        return count;
    }
};
