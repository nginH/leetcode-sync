class Solution {

int dfs(vector<int>& nums, int k , long long currProd, int idx,  unordered_map<string, int>&mp){

    if(idx>=nums.size()){
        return 0;
    }
    string key=to_string(idx)+"-"+to_string(currProd);
    if(mp.contains(key)){
        return mp[key];
    }
    currProd*=nums[idx];
    int ans=0;
    if(currProd<k){// strictly less
        ans=1+dfs(nums, k, currProd, idx+1, mp);
    }
    return mp[key]=ans;
}


public:
    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        unordered_map<string, int>mp;
        int ans=0;
        for(int i=0; i<nums.size(); i++)
            ans+=dfs(nums, k, 1, i, mp);
        return ans;
    };
};
