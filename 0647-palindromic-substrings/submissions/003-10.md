class Solution {
public:

    bool isPalandro(string & s, int l, int h){
        while(l<=h){
            if(s[l]==s[h]){
                l++;
                h--;
            }else{
                return false;
            }
        }
        return true;
    }

    int countSubstrings(string s) {
            int count=0;
            for(int i=0; i<s.size(); i++){
                for(int j=i; j<s.size(); j++){
                    if(isPalandro(s, i, j)){
                        count++;
                    }                
                }
            }

            return count;
    }
};
