class Solution {

private:

    int expand(string s, int low, int high){
       int  count=0;
        while(low>=0 && high<s.size() && s[low]==s[high]){
            low--;
            high++;
            count++;
        }
        return count;
    }

public:
    int countSubstrings(string s) { // length = 1000 n2 will work
            int count=0;
           
            for(int i=0; i<s.size(); i++){
                count+= expand(s,i, i);

                count+= expand(s,i, i+1);
            }

            return count;
    }
};
