class Solution {

private:

    bool isPalindron(string s){
        if(s.size()<=1){
            return true;
        }
        int i=0;
        int j=s.size()-1;
        while(i<j){
            if(s[i]!=s[j]){
                return false;
            }
            i++;
            j--;
        }
        return true;
    }

public:
    int countSubstrings(string s) { // length = 1000 n2 will work
            int count=0;
           
            for(int i=0; i<s.size(); i++){
                for(int j=i; j<s.size(); j++){
                    string sub= s.substr(i, j-i+1);
                    if(isPalindron(sub)){
                        count+=1;
                    }
                }
            }

            return count;
    }
};
