class Solution {
public:

    bool isPalandro(string & s, int l, int h){
        while(l<=h){
            if(s[l]==s[h]){
                l++;
                h--;
            }else{
                return false;
            }
        }
        return true;
    }

    int countSubstrings(string s) {
            int count=1;
            for(int i=0; i<s.size(); i++){
                for(int j=i; j<s.size(); j++){
                    int k=j-i+1;
                    if(isPalandro(s, i, k)){
                        count ++;
                    }                
                }
            }

            return count;
    }
};
