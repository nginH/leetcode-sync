/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        
        if(!head || !head->next){ // 0 or 1 node ..
            return nullptr;
        }
        

        ListNode*fast=head->next;
        ListNode *slow= head;
        ListNode *follow;
        while(fast && fast->next && slow!=fast){
            follow=slow;
            slow=slow->next;
            fast=fast->next->next;
        }

        return fast->next==nullptr?nullptr:follow;



    }
};
