/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        
        if(!head){ // 0 or 1 node ..
            return nullptr;
        }
        if(!head->next){
            return nullptr;
        }
        

        ListNode*fast=head->next;
        ListNode *slow= head;
        ListNode *follow;
        while(fast && fast->next){
            follow=slow;
            slow=slow->next;
            fast=fast->next->next;
            if(slow==fast){
                break;
            }
        }

        return fast->next==nullptr?nullptr:follow;



    }
};
