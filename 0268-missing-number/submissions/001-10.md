class Solution {
public:
    int missingNumber(vector<int>& nums) {
        // constrin all number are unique...

        int ans=nums.size();
        for(int i=0; i<nums.size(); i++){
            ans^=i;
            ans^=nums[i];

        }
        return ans;;
    }
};
