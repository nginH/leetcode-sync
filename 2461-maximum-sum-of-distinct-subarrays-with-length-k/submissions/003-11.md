class Solution {
public:
    long long maximumSubarraySum(vector<int>& nums, int k) {
        
        // widow sliding ..
        unordered_set<int>st;

        int left=0;
        int sum=0;
        bool isValid=true;
        for(int i=0; i<k; i++){
            if(st.contains(nums[i])){
                isValid=false;
            } 
            st.insert(nums[i]);
            sum+=nums[i];
        }

       int  ans=isValid?sum:0;
        for(int right=k; right<nums.size(); right++){
            bool isValid=true;

            if(st.contains(nums[right])){
                isValid=false;
            }
            sum-=nums[left];
            sum+=nums[right];

            st.erase(nums[left]); // remove left 
            st.insert(nums[right]); // add right

            left++;
            if(isValid){
                ans=max(ans, sum);
            }


        }

        return ans;



    }
};
