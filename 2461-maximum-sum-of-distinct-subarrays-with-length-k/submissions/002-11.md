class Solution {
public:
    long long maximumSubarraySum(vector<int>& nums, int k) {
        
        // widow sliding ..
        unordered_set<int>st;

        int left=0;
        long long sum=0;
        long long ans=0;

        for(int right=0; right<nums.size(); right++){
           

           sum+=nums[right];
           st.insert(nums[right]);

           if(right-left+1>k){
                sum-=nums[left];
                st.erase(nums[left]);
                left++;
           }
           if(st.size()==k){
                ans=max(ans, sum);
           }


        }

        return ans;



    }
};
