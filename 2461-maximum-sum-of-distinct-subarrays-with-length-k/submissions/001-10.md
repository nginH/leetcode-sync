class Solution {
public:
    long long maximumSubarraySum(vector<int>& nums, int k) {
        
        // widow sliding ..
        unordered_map<int, int>st; // val, freq

        int left=0;
        long long sum=0;
        long long ans=0;

        for(int right=0; right<nums.size(); right++){
           

           sum+=nums[right];
           st[nums[right]]++;

           if(right-left+1>k){
                sum-=nums[left];
                st[nums[left]]--;

                if(st[nums[left]]==0)
                    st.erase(nums[left]);
                left++;
           }
           if(right-left+1==k && st.size()==k){
                ans=max(ans, sum);
           }


        }

        return ans;



    }
};
