class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        unordered_map<int, int> mp;

        for(auto x:nums){
            mp[x]++;
        }
        int i=1;
        for(i=1; i<=nums.size(); i++){
            if(!mp.contains(i)){
                return i;
            }
        }
        return i;
    }
};
