class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
       
        for(int i=0;i<nums.size(); i++){
            if(nums[i]<0){
                nums[i]=0;
            }
        }

         for(int i=0;i<nums.size(); i++){
            if(abs(nums[i])>=1 && abs(nums[i])<=nums.size()){
                int idx= abs(nums[i])-1;
                if(nums[idx]>0){
                    nums[idx]*=-1;
                }else if(nums[idx]==0){
                    nums[idx]=-i*(nums.size()+1);
                }
            }
        }

       int i=0;
        for( int i=1; i<=nums.size(); i++){
            if(nums[i-1]>=0){ // non marked index is answer
                return i;
            }
        }
        return i;

    }
};
