class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        queue<pair<int, int>> q;
        int freshCount = 0;

        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid[0].size(); j++) {

                if (grid[i][j] == 2) {
                    q.push({i, j});
                } else if (grid[i][j] == 1) {
                    freshCount += 1;
                }
            }
        }
        int ans = -1;
        vector<pair<int, int>> dir = {
            {1, 0},
            {0, 1},
            {-1, 0},
            {0, -1},
        };

        if (q.size() == 0) {
            return 0;
        }

        while (!q.empty()) {
            int sz = q.size();
            ans += 1;

            while (sz--) {
                auto [r, c] = q.front();
                q.pop();

                for (auto& [dx, dy] : dir) {
                    int nr = dx + r;
                    int nc = dy + c;

                    if (nr < 0 || nc < 0 || nr >= grid.size() ||
                        nc >= grid[0].size()) {
                        continue;
                    }

                    if (grid[nr][nc] == 2 ||
                        grid[nr][nc] == 0) { // void or rotten
                        continue;
                    }
                    if (grid[nr][nc] == 1) {
                        grid[nr][nc] = 2;
                        q.push({nr, nc});
                        freshCount--;
                    }
                }
            }
        }
        return !freshCount ? ans : -1;
    }
};
