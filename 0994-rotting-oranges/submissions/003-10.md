class Solution {
    int m,n;


    vector<pair<int, int>> dir={
        {0,1},
        {1,0},
        {-1, 0},
        {0, -1}
    };


public:
    int orangesRotting(vector<vector<int>>& grid) {
        queue<pair<int, int>> rq;
        m=grid.size();
        n=grid[0].size();
        int freshCount=0;
        for(int i=0; i<m; i++){
            for(int j=0; j<n; j++){
                if(grid[i][j]==1){
                    freshCount++;
                }else if(grid[i][j]==2){
                    rq.push({i, j}); // this will use to start iteration...
                }
            }
        }


        int ans=-1;
        int frot=0;
        if(freshCount==0){
            return 0;
        }
        while(!rq.empty()){
            ans++;
            int sz=rq.size();
            while(sz--){
                auto [rrow, rcol]=rq.front();
                rq.pop(); // already pop;;
                for(auto [dx, dy]: dir){
                    int ni=dx+rrow;
                    int nj=dy+rcol;

                    if(ni<0 || nj<0 || ni>=m || nj>=n){
                        continue; // out of bound..
                    }
                    if(grid[ni][nj]==1){ // fresh 
                        frot++;
                        grid[ni][nj]=2; // rot that fresh..
                        rq.push({ni, nj});// for next iteration..
                    }
                }
            }
        }

        return (frot==freshCount)? ans:-1; 
    }
};
