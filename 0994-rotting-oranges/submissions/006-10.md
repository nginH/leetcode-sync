class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        // 2 rotten     
        int m=grid.size();
        int n=grid[0].size();

        queue<pair<int, int>>q;
        int fresh=0;
        int min=0;


        for(int i=0;i<m; i++){
            for(int j=0; j<n; j++){
                if(grid[i][j]==2){
                    q.push({i,j});
                }else if(grid[i][j]==1){
                    fresh+=1;
                }
            }
        }

        if(fresh==0){
            return 0;
        }

        vector<pair<int, int>>dir={
            {0, 1},
            {1, 0},
            {-1, 0},
            {0,-1}
        };

        while(!q.empty() && fresh>0){
            min++;
            int sz=q.size();
            while(sz--){
                auto [i, j]=q.front();
                q.pop();

                for(auto [dx, dy]:dir){
                    int ni=dx+i;
                    int nj=dy+j;

                    if(ni<0 || nj<0 || ni>=m || nj>=n){
                        continue ; //out of bound
                    }

                    if(grid[ni][nj]!=1){
                        continue; // either sada hua or water ..
                    }

                    grid[ni][nj]=2;
                    fresh--;
                    q.push({ni, nj}); // for next iteration,..
                }
            }
        }


        return fresh==0? min:-1;
    }
};
