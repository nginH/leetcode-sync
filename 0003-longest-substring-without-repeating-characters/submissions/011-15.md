class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int ans=0;
        int ctr=0;
        vector<int> hash(27, 0);
        for(int i=0 ; s[i] !='\0'; i++){
            ans=max(ans, ctr);
            int k=s[i]-'a';
            
            // cout<<ans<<": "<<k << " "<<hash[k]<<"\n";
            if(hash[k]!=0){
                hash[k]=0;
                ctr=0;
            }
            hash[k]++;
            ctr++;
        }
        return ans;
    }
};
