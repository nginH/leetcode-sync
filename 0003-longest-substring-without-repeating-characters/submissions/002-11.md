class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_map<int, int> mp;
        int ans=0;
        int longest=0;
        for(int i=0; i<s.size(); i++){
            if(!mp.contains(s[i])){
                longest++;
                mp[s[i]]++;

            }else{
                longest=0;
                mp.erase(s[i]);
            }
            ans=max(ans, longest);
        }
        return ans;
    }
};
