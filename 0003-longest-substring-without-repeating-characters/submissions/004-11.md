class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int ans=0;
        int j=0;
        string sumArray;
        unordered_map<char, int> mp;
        for (int i=0; s[i]; i++ ){
            if(mp[s[i]]) {
                string newArray="";
                bool ok=1;
                for(auto t:sumArray){
                    if(ok){
                        if(t==s[i]){
                            ok=0;
                        }
                    }else{
                        newArray+=t;
                    }
                }
                sumArray=newArray;
                mp[s[i]]=0;
            }
            sumArray+=s[i];
            mp[s[i]]=1;
            ans=max(ans, (int)sumArray.size());
        }
        return ans;
    }
};
