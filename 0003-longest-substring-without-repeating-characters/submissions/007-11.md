class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int ans=0;
        int j=0;
        if(s.size()==1){
            return 1;
        }
        unordered_map<char, int> mp;
        for (int i=0; s[i]; i++ ){
            if(mp[s[i]]) {
                ans=max(j, ans);
                j=0;
                mp[s[i]]=0;
            }
            mp[s[i]]=1;
            j++;
        }
        return ans;
    }
};
