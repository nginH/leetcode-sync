```cpp
#define max(a,b) a>b?a:b
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_map<char, int> lastSeen;
        int ans=0;
        int left=0;
        for(int i=0; i<s.size(); i++){
            char c=s[i];
            if(lastSeen.count(c)){
                left=max(left, lastSeen[c]+1);
            }

            lastSeen[c]=i;
            ans=max(ans, i-left+1);
           
        }
        return ans;
    }
};
```
