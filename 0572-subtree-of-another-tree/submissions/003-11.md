/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {


bool dfs(TreeNode *a, TreeNode *b){

    if(!a && !b){
        return true;
    }
    if(!a || !b){
        return false;// any one end before others ...
    }

    bool c1=dfs(a->left, b->left);
    bool c2= dfs(a->right, b->right);
    return c1 && c2; // both shold return true else false..
}


// not bst..

TreeNode* serc(TreeNode* root , int val){
    if(!root){
        return NULL;
    }
    if(root->val ==val){
        return root;
    }
    TreeNode* a =serc(root->left , val);
    if(a!=NULL){
        return a;
    }
    TreeNode *b= serc(root->right, val);
    return b;
}



public:
    bool isSubtree(TreeNode* root, TreeNode* subRoot) {
        
        return dfs(serc(root, subRoot->val), subRoot);

    }
};
