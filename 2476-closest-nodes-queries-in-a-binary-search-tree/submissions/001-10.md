/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {

private:
    vector<int> ans;
    void sol(TreeNode *root){
        if(!root){
            return;
        }

        sol(root->left);
        ans.push_back(root->val);
        sol(root->right);
    }


public:
    vector<vector<int>> closestNodes(TreeNode* root, vector<int>& queries) {
        sol(root);

        vector<vector<int>> k;

        for(auto x:queries){
            auto lb=std::lower_bound(ans.begin(),ans.end() ,x);
            auto ub=std::upper_bound(ans.begin(), ans.end(), x);
            int fl=-1, cl=-1;
            if(lb!=ans.begin()){
                fl=*(prev(lb));
            }
            if(ub!=ans.end()){
                cl=*ub;
            }
            if(ub!=lb){
                fl=cl=*lb;
            }

            k.push_back({fl, cl});
        }

        return k;
    }
};
