class Solution {
public:
    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
        
        vector<vector<pair<int, int>>>adj(flights.size());

        for(auto x:flights){
            int u=x[0];
            int v=x[1];
            int w=x[2];
            adj[u].push_back({v,w});
        }


        queue<pair<int, int>> q;
        q.push({0, src}); // 
        vector<int>dist(n, INT_MAX);

        int layer=-1;
        while(!q.empty()){
            if(layer==k){
                break;
            }
            int sz=q.size();
            while(sz--){
                auto [prewt, v]=q.front();
                q.pop();
                for(auto [u, wt]:adj[v]){
                    if(dist[u]>wt+prewt){ // relax.. 
                        dist[u]=prewt+wt;
                        q.push({dist[u], u});
                    }
                }


            }
            layer++;
        }
        
        return dist[dst]==INT_MAX?-1:dist[dst];
    }
};
