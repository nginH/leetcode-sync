class Solution {
public:


    int findPeakElement(vector<int>& arr) {
        int low=0, high=arr.size()-1, lb=arr.size()+1;


        if(arr.size()==1){
            return 0;
        }
        if(arr.size()==2){
            return arr[0]> arr[1]?0:1;
        }
        while(low<=high){
            int mid=(low+high)/2;
            if(mid>0 && mid<arr.size()-1){

                if(arr[mid]>arr[mid-1] && arr[mid]>arr[mid+1]){
                    return mid;
                }

                if(arr[mid]>arr[mid-1] && arr[mid]<arr[mid+1]){ //increaseing 
                    // so peak should at right 
                    low=mid+1;
                }else{// decreasing 
                    high=mid-1;
                }
            }else{
                low=mid+1; // can both increae lower bound or upper bound
            }
        }

        return arr[0]> arr[arr.size()-1]?0:arr.size()-1;


    }
};
