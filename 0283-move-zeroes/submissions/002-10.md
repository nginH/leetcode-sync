class Solution {
public:
    void moveZeroes(vector<int>& nums) {

       int high=nums.size()-1;
        for(int i=nums.size()-1; i>=0; i--){
            if(nums[i]==0){
                int lower=i;
                while(lower<high){
                    swap(nums[lower], nums[lower+1]);
                    lower++;
                }
                high--;

            }
        }   
    }
};
