class Solution {
public:
    int snakesAndLadders(vector<vector<int>>& board) {
        int n=board.size();
        int m=board.size();

        vector<int>adj(n*m+1);
        int idx=1;
        for (int x = 1; x <= n*m; x++) {
            int row = n - 1 - (x - 1) / n;
            int col = (x - 1) % n;

            // Every alternate
            if ((n - row) % 2 == 0) {
                col = n - 1 - col;
            }

            if (board[row][col] == -1)
                adj[x] = x;
            else
                adj[x] = board[row][col];
        }

        queue<int>q;
        int steps=0;
        vector<bool>vis(n*m+1,  false);
        vis[1]=true;
        q.push(1);
        while(!q.empty()){
           
            
            int sz=q.size();
            while(sz--){
                int toexplore=q.front();
                if(toexplore==n*m){
                    return steps;
                }
                q.pop();
                for(int i=1; i<=6; i++){
                    int ckcount=toexplore+i;
                    if (ckcount > n * n)       // FIX 3
                        continue;
                    int newIdx= adj[ckcount];
                    if(!vis[newIdx]){
                        vis[newIdx]=true;
                        q.push(newIdx);
                    }
                }
            }
            steps++;
        }
        return -1;
    }
};
