class Solution {

int solve(vector<pair<int, int>> &arr, int idx, int m, int n){
    if(idx>=arr.size()){
        return 0 ;
    }

    int skip =solve(arr, idx+1, m, n);
    int take=0;
    if(arr[idx].first<=m && arr[idx].second <=n) // can even take
        take= 1+solve(
                arr,
                idx+1,
                m-arr[idx].first,
                n-arr[idx].second);
    
    return max(take , skip);


}



public:
    int findMaxForm(vector<string>& strs, int m, int n) {
        vector<pair<int, int>> arr;

        for(int i=0;i<strs.size(); i++){
            int one=0;
            int zero=0;
            for(int j=0;j<strs[i].size(); j++){
                if(strs[i][j]=='0'){
                    zero+=1;
                }else{
                    one+=1;
                }
            }
            arr.push_back({zero, one});
        }
       return solve(arr,0, m, n);
       

    }
};
