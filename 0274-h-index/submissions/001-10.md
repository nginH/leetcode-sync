class Solution {
public:
    int hIndex(vector<int>& arr) {
        sort(arr.begin(), arr.end());
        int ans=0;

        int l=0;
        int n=arr.size();
        int h=n-1;

        while(l<=h){
            int mid=(l+h)/2;
            if(arr[mid]>=n-mid){
                ans=n-mid;
                h=mid-1;
            }else{
                l=mid+1;
            }
        }

        return ans;

    }
};
