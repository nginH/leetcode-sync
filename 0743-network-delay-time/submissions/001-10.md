class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
        vector<vector<pair<int, int>>> adj(n+1);

        for(auto edge:times){
            int u=edge[0];
            int v=edge[1];
            int w=edge[2];
            adj[u].push_back({v,w});
        }

        priority_queue<pair<int, int>,
            vector<pair<int, int>>,
            greater<>> pq;// min heap..

        vector<int>dist(n+1,INT_MAX); //every node have inf distance  initially ..

        pq.push({0, k});// weight, node..
        dist[k]=0;

        while(!pq.empty()){
            auto [d, node]= pq.top();
            pq.pop();
            if(d>dist[node]){
                continue;
            }
            for(auto [next,wt]:adj[node]){

                if(dist[next]>wt+d){// relax
                    dist[next]=wt+d;
                    pq.push({dist[next] , next});
                }
            }
        }
        
        int ans=0;

        for(int i=1; i<=n; i++){
            if(dist[i]==INT_MAX){
                return -1;
            }
            ans=max(ans, dist[i]);
        }
        return ans;



    }
};
