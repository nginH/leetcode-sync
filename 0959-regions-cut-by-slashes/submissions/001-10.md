class Solution {

    vector<pair<int, int>> dir = {{1, 0}, {0, 1}, {-1, 0}, {0, -1}};
    void dfs(vector<vector<int>>& grid, int r, int c) {
        if (r < 0 || c < 0 || r >= grid.size() || c >= grid[0].size()) {
            return;
        }
        if (grid[r][c] == 1 || grid[r][c] == 2) {
            return;
        }
        grid[r][c] = 2;
        for (auto& [dx, dy] : dir) {
            dfs(grid, r + dx, c + dy);
        }
    }

public:
    int regionsBySlashes(vector<string>& grid) {
        int n=grid.size();
        vector<vector<int>> expanded(n * 3, vector<int>(n * 3, 0));

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {

                int r = i * 3;
                int c = j * 3;

                if (grid[i][j] == '/') {

                    expanded[r][c + 2] = 1;
                    expanded[r + 1][c + 1] = 1;
                    expanded[r + 2][c] = 1;

                } else if (grid[i][j] == '\\') {

                    expanded[r][c] = 1;
                    expanded[r + 1][c + 1] = 1;
                    expanded[r + 2][c + 2] = 1;
                }
            }
        }

        int ans = 0;
        for (int i = 0; i < expanded.size(); i++) {
            for (int j = 0; j < expanded[0].size(); j++) {
                if (expanded[i][j] == 0) {
                    ans++;
                    dfs(expanded, i, j);
                }
            }
        }
        return ans;
    }
};
