class Solution {
public:
    int findDuplicate(vector<int>& nums) {

        vector<int> arr(1000001, 0);
        
        for(auto x:nums){
            if(arr[x]==0){
                arr[x]++;
            }else{
                return x;
            }
        }

        return -1;
    }
};
