class Solution {
public:
    int majorityElement(vector<int>& nums) {
        int cd=0;
        int count=0;
        for(int x:nums){
            if(count==0){
                cd=x;
            }
            if(cd==x){
                count++;
            }else{
                count--;
            }
        }
        return cd;
    }
};
