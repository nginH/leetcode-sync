class Solution {
public:
    int nearestDrone(vector<vector<int>>& drones, vector<int>& target) {

        int ans=INT_MAX;

        int tx=target[0];
        int ty=target[1];
        int idx=-1;
        for(int i=0; i<drones.size(); i++){
            int x=drones[i][0];
            int y=drones[i][1];
            int r=drones[i][2];
            int val=abs(tx-x)+abs(ty-y);
            if(val>r){
                continue;// skip not reachable 
            }else{
              int newVal= min(val, ans);
              if (newVal!=ans){
                idx=i;
                ans=newVal;
              }
                
            }
        }

        return idx;

    }
};
