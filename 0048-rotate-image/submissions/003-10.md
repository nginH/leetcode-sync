class Solution {
public:
  void rotate(vector<vector<int>>& m) {
         int ans[m.size()][m.size()];
        for(int i=0; i<m.size(); i+=1){
            for(int j=0; j<m[0].size(); j+=1){
                ans[j][m.size()-i-1]= m[i][j];
            }
        }
        
        for(int i=0; i<m.size(); i+=1){
            for(int j=0; j<m[0].size(); j+=1){
                m[i][j]=ans[i][j];
            }
        }


    }
};
