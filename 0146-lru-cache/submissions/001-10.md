class LRUCache {

private:

    list<pair<int, int>> dll; // pair<key, val>
    unordered_map<int, list<pair<int, int>>::iterator> mp; // key, dll,
    int cap;

public:
    LRUCache(int capacity) {
        this->cap=capacity;
    }
    
    int get(int key) {
        if(mp.contains(key)){
            auto node =mp[key]; // move to front
            dll.splice(dll.begin(), dll, node); // move to first ( destination , sourcelist and iterator)
            return node->second;
        }else{
            return -1;
        }
    }
    
    void put(int key, int value) {
        if(mp.contains(key)){
            auto node =mp[key]; // move to front
            node->second=value;
            dll.splice(dll.begin(), dll, node);
        }else{ // not existi so append..
           if(dll.size()==this->cap){
                int key= dll.back().first;
                mp.erase(key);
                dll.pop_back();
           }

           dll.push_front({key, value});
           mp[key]=dll.begin();
        } 
        
    }
};
