class Solution {
public:
    bool isSubsequence(string s, string t) {
        // brute force.. 
        int countMatch=s.size();
        int i=0;
        for(int j =0; j<s.size(); j++){
            for(i; i<t.size(); i++){
                if(s[j]==t[i]){
                    countMatch--;
                    break;
                }
            }
        }
        return countMatch<=0?true:false;
    }
};
