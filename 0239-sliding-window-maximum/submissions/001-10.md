class Solution {

private:

public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        priority_queue<pair<int, int >>pq; // value, idx
        vector<int>ans;

        for(int i=0; i<nums.size(); i++){
            pq.push({nums[i], i});
            if(i>=k-1){//now inside the k window .. 

                while(pq.top().second<i-k+1){
                    pq.pop();
                }
                ans.push_back(pq.top().first);



            }


        }

        return ans;
    }
};
