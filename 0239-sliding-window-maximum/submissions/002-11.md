class Solution {

private:

    

public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        priority_queue<int, vector<int> , less<>>pq;
        queue<int> q;
        vector<int>ans;

        for(int i=0; i<nums.size(); i++){
            q.push(nums[i]);
            pq.push(nums[i]);
            if(q.size()==k){
                q.pop();// left.
                ans.push_back(pq.top());
            }
        }

        return ans;
    }
};
