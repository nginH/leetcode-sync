class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        priority_queue<int, vector<int>,greater<>> pq;
        unordered_map<int, int>mp;
        if(k<1){
            return {};
        }
        for(auto x:nums){
            pq.push(x);
            mp[x]++;
        }

        vector<int>ans;
        int prev=pq.top();
        ans.push_back(prev);
        pq.pop();
        if(k==1){
            k--;
        }
        while(k>1 && !pq.empty()){
            int curr= pq.top();
            pq.pop();
            if(curr!=prev){
                ans.push_back(curr);
                prev=curr;
                k--;
            }
        }


        return ans;
    }
};
