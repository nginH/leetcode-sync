class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {

        unordered_map<int, int>mp;
        for(auto x:nums){
            mp[x]++;
        }

        priority_queue<
        pair<int, int>,
        vector<pair<int,int>>,
        less<>>pq;

        for(auto [a,b]:mp){
            pq.push({b, a});
        }
        vector<int>ans;
        while(k--){
            auto [a, b]=pq.top();
            pq.pop();
            ans.push_back(b);
        }

        return ans;
    }
};
