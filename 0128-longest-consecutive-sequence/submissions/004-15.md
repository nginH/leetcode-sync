class Solution {
public:
    int longestConsecutive(vector<int>& v) {
        std::unordered_map<int, int> mp;
        int mi=INT_MAX;
        for(auto x:v){
            mp[x]++; // store existance of elememnt and numbers.
            mi=min(mi, x); // store mimum value..
        }
        int ans=0;
        while(true){
            ans++;
            mi++;
            if(!mp.contains(mi)){
                break;
            }
        }
        return ans;
    }

};
