class Solution {
public:
    int longestConsecutive(vector<int>& v) {
        if(v.size()==0){
            return 0;
        }
        set<int> st={v.begin(), v.end()};
        vector<int> nums={st.begin(), st.end()};
        int ans=0;
        int temp=1;

        unordered_map<int, int> mp;
        for(int i=1; i<nums.size(); i++){
            mp[nums[i]]+=1;

            if(mp.contains(nums[i]-1)){
                temp+=1;
                ans=max(ans, temp);
            }else{
                temp=1;
            }

        }
        return ans+1;
    }
};
