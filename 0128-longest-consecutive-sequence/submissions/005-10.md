class Solution {
public:
    int longestConsecutive(vector<int>& v) {
        if(v.size()==0){
            return 0;
        }
        unordered_set<int> st={v.begin(), v.end()};
        int ans=0;
        for(auto x:st){
            if(!st.contains(x-1)){  
               int len=1;
                while(st.contains(x+len)){
                    len++;
                }
                ans=max(ans,len);
                
            }
        }
        return ans;
    }
};
