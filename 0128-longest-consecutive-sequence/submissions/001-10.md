class Solution {
public:
    int longestConsecutive(vector<int>& v) {
        std::unordered_set<int>st(v.begin(), v.end());
        int ans=0;
        for(auto x:st){
            if(!st.contains(x-1)){
                int len=1;
                int curr=x;
                while(st.contains(curr+1)){
                    len+=1;
                    curr+=1;
                }
               ans=max(ans, len);
            }
        }
        return ans;
    }

};
