class Solution {
public:
    int longestConsecutive(vector<int>& v) {
        
        set<int> st={v.begin(), v.end()};
        vector<int> nums={st.begin(), st.end()};
        sort(nums.begin(), nums.end());
        int ans=0;
        int temp=0;
        for(int i=1; i<nums.size(); i++){
            int curr=nums[i];
            int prev=nums[i-1];
            int val= curr-prev;
            if(val==1){
                temp+=1;
                ans=max(ans,temp+1);
            }else{
                temp=1;
            }
        }
        return ans;
    }
};
