class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        int n= prerequisites.size();
        vector<vector<int>>adj(numCourses);
        vector<int>indegree(numCourses);

        for(auto x:prerequisites){
            adj[x[1]].push_back(x[0]);
            indegree[x[0]];
        };
        queue<int>q;

        for(int i=0; i<indegree.size(); i++){
            if(indegree[i]==0){
                q.push(i);
            }
        };
        vector<int> ans;
    
        while(!q.empty()){
            int sz=q.size();
            // first iteration
            while(sz--){
                int curr=q.front();
                ans.push_back(curr);
                q.pop();
                for(auto x:adj[curr]){
                    indegree[x]--;
                    if(indegree[x]==0){
                        q.push(x);
                    }
                }
            }
        }
        return ans;
    }
};
