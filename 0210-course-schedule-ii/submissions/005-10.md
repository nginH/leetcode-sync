class Solution {
public:
    vector<int> findOrder(int n, vector<vector<int>>& prerequisites) {

        vector<vector<int>>adj(n);
        vector<int>inDegree(n, 0);

        for(auto x:prerequisites){
            adj[x[1]].push_back(x[0]);
            inDegree[x[0]]++;
        }

        queue<int>q;

        for(int i=0; i<n; i++){
            if(inDegree[i]==0){
                q.push(i);
            }
        }

        vector<int>topo;

        while(!q.empty()){
            int node= q.front();
            q.pop();
            topo.push_back(node);
            for(auto x: adj[node]){
                inDegree[x]--;
                if(inDegree[x]==0){
                    q.push(x);
                }
            }
        }

        vector <int>empty;

        return topo.size()==n?topo:empty;
    }
};
