class TrieNode {
public:
    TrieNode* links[26];
    bool isLast;

    TrieNode() {
        for (int i = 0; i < 26; i++)
            links[i] = nullptr;
        isLast = false;
    }

    bool contains(char c) {
        return links[c - 'a'] != nullptr;
    }

    TrieNode* get(char c) {
        return links[c - 'a'];
    }

    void put(char c, TrieNode* node) {
        links[c - 'a'] = node;
    }

    void setEnd() {
        isLast = true;
    }

    bool isEnd() {
        return isLast;
    }
};



class WordDictionary {

private:
    TrieNode *Root;

    bool dfs (TrieNode *root, string &word, int idx){
        if(idx>=word.size()){
            return root->isEnd();
        }

        char c=word[idx];

        if(c!='.'){
            if(!root->contains(c)){
                return false;
            }
            return dfs(root->get(c), word, idx+1); // next word with next index
        }else{ // every possible calls
            for(int i=0; i<26; i++){
                if(root->contains('a'+i)) // not null
                   if(dfs(root->get('a'+i), word, idx+1)){ // idx+1 done to move next 
                    return true;
                   }
            }
        }
          return false;

    }


public:
    WordDictionary() {
        Root=new TrieNode();
    }
    
    void addWord(string word) {
        TrieNode *temp=Root;

        for(auto x:word){
            if(!temp->contains(x)){
                temp->put(x, new TrieNode());
            }
            temp=temp->get(x);
        }

        temp->setEnd();
    }
    
    bool search(string word) {
        return dfs(Root, word, 0);
    }
};
