class Solution {


public:
    vector<vector<int>> updateMatrix(vector<vector<int>>& mat) {

        std::queue<pair<int, int>>q;

            int m=mat.size();
            int n=mat[0].size();

        for(int i=0; i<mat.size(); i++){
            for(int j=0;j<mat[0].size(); j++){
                if(mat[i][j]==0){
                    q.push({i,j});
                }else {
                    mat[i][j]=-1;
                }
            }
        }

        std::vector<pair<int, int>>dir={
            {1,0},
            {0,1},
            {0,-1},
            {-1,0}
        };

        while(!q.empty()){
            auto [i, j]=q.front();
            q.pop();
            for(auto [dx, dy]:dir){
                int ni= dx+i;
                int nj=dy+j;

                if(ni<0 || nj<0 ||ni>=m || nj>=n){
                    continue; // out of bound;
                }
                if(mat[ni][nj]==-1){
                    mat[ni][nj]=mat[i][j]+1;
                    q.push({ni, nj});
                }

            }
        }

        return mat;



    }
};
