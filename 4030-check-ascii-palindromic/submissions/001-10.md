class Solution {


public:
    bool isPalindromic(string s) {
        string toCheckString="";
        for(int x:s){
            bitset<8> binary8(x);
            toCheckString+=binary8.to_string();
        }

        int i=0;
        int j=toCheckString.size()-1;
        while(i<=j){
            if(toCheckString[i]!=toCheckString[j]){
                return false;
            }
            i++;
            j--;
        }

        return true;

    }
};
