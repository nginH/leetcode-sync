class Solution {
public:
    int maxSatisfied(vector<int>& customers, vector<int>& grumpy, int minutes) {
        // min fix size sliding window..

        int sum=0;
        int maxAns=0;
        int base=0;
        for(int i=0; i<customers.size(); i++){
            if(grumpy[i]==0){
                base+=customers[i];
            }


        }


        for(int i=0; i<minutes; i++){
            if(grumpy[i]==1)
                sum+=customers[i];
        }

        int left=0;
        
        maxAns=sum;

        for(int right=minutes; right<customers.size(); right++){
            
            if(grumpy[right]==1)
                sum+=customers[right];

            if(grumpy[left]==1){
                sum-=customers[left];
            }
            left++;

            maxAns=max(maxAns, sum);
        }

        return base+maxAns;
    }
};
