/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {

int dfs(TreeNode *root, long long  tar,long long curr, unordered_map<long long, int> mp){
    if(!root){
        return 0;
    }
    curr+=root->val;

    int ans=mp[curr-tar];

    mp[curr]++;

    ans+=dfs(root->left, tar, curr, mp);
    ans+=dfs(root->right, tar, curr, mp);
    mp[curr]--;
    return ans;
}

public:
    int pathSum(TreeNode* root, int targetSum) {

        if(!root){
            return 0;
        }
        unordered_map<long long , int>mp;
        mp[0]=1;
        return  dfs(root, targetSum, 0, mp);
    }
};
