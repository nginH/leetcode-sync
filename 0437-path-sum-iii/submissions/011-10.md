/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {

private:

 int sumk(TreeNode* root, long long target) {
        if (!root)
            return 0;

        int res = 0;

        if (root->val == target)
            res++;

        res += sumk(root->left, target - root->val);
        res += sumk(root->right, target - root->val);

        return res;
    }

public:
    int pathSum(TreeNode* root, int k) {
        if(root==nullptr){
            return 0;
        }
        return 
                sumk(root, k)+
                pathSum(root->left,k)+ 
                pathSum(root->right,k);
    }
};
