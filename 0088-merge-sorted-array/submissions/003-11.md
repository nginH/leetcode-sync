class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
            int i=0;
            int j=0;
            vector<int> ans;
            while(i<m && j<n){
                if(nums1[i]<nums2[j]){
                    ans.push_back(nums1[i]);
                    i++;
                }else{
                    ans.push_back(nums2[j]);
                    j++;
                }
            }

            for(int k=i ; k<m; k++){
                ans.push_back(nums1[k]);
            }
            for(int k=j ; k<n; k++){
                ans.push_back(nums2[k]);
            }
            for(int k=0 ; k<m+n; k++){
                if(ans[k]!=0){
                    nums1[k]=ans[k];
                }
            }

        return ;


    }
};
