class Solution {
public:
       void merge(vector<int>& num1, int m, vector<int>& num2, int n) {
        int ans[m+n];
        int i=0, j=0;
        int k=0;
        while(i<m && j<n){
            if(num1[i] < num2[j]){
                ans[k++]= num1[i];
                i++;
            }else{
                ans[k++]= num2[j];
                j++;
            }
        }
        for(;i<m;){
            ans[k++] = num1[i++];
        }

        for(;j<n;){
            ans[k++]=num2[j++];
        }
        for(int i=0; i<k; i++){
            if(i<num1.size()){
                num1[i]=ans[i];
            }else{
                num1.push_back(ans[i]);
            }
        }
    }

};
