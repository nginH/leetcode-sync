class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        int level=1;
        vector<vector<int>>ans;
        while(numRows--){
            vector<int> temp;
            for(int i=1; i<=level; i++){
                if(i==1 || i==level){
                    temp.push_back(1);
                }else{
                    int leftVal=ans[level-2][i-2];
                    int rightVal=ans[level-2][i-1];
                    temp.push_back(leftVal+rightVal);
                }
            }
            level++;
            ans.push_back(temp);
        }

        return ans;
    }
};
