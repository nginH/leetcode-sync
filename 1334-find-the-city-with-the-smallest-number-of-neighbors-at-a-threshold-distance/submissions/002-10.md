class Solution {

    vector<int> dij(int n, vector<vector<int>>&edge, int start){
        vector<vector<pair<int,int>>> graph(n);
        vector<int> dist(n, INT_MAX);

        for(auto &x:edge){
            int u=x[0];
            int v=x[1];
            int w=x[2];
           graph[u].push_back({v, w});
           graph[v].push_back({u, w});
        }

        std::priority_queue<
            pair<int, int>,
            vector  <pair<int, int>>,
            greater <pair<int, int>>
        > pq;

        dist[start]=0;
        pq.push({0, start});


        while(!pq.empty()){
            auto [currDist, node] = pq.top();
            pq.pop();

            if (currDist > dist[node]) continue;

            for(auto [u, wt]: graph[node]){
                if(dist[u]>wt+currDist){
                    dist[u]=wt+currDist;
                    pq.push({dist[u], u});
                }
            }
        }

        return dist;
    }




public:
    int findTheCity(int n, vector<vector<int>>& edges, int distanceThreshold) {
        // total is n;
        int ans=-1;
        int minReachable=INT_MAX;

        for(int i=0; i<n; i++ ){
            vector<int>res=dij(n, edges, i);
            int reachable=0;

            for (int j = 0; j < n; j++) {
                if (i != j && res[j] <= distanceThreshold) {
                    reachable++;
                }
            }


             // If tie, choose larger city index
            if (reachable <= minReachable) {
                minReachable = reachable;
                ans = i;
            }
        }

        return ans;



    }
};
