class Solution {

    int bfs(vector<vector<pair<int, int>>> &adj,
            int startNode,
            int k,
            int n) {

        priority_queue<
            pair<int, int>,
            vector<pair<int, int>>,
            greater<>
        > pq;

        pq.push({0, startNode});

        vector<int> vis(n, INT_MAX);

        // FIX 1
        vis[startNode] = 0;

        int reachable = 0;

        while (!pq.empty()) {

            auto [w_, v] = pq.top();
            pq.pop();

            if (w_ > vis[v]) {
                continue;
            }

            if (v != startNode) {
                reachable++;
            }

            for (auto [u, wt] : adj[v]) {

                int cost = w_ + wt;

                // FIX 2: <= because threshold is inclusive
                if (cost <= k && cost < vis[u]) {
                    vis[u] = cost;
                    pq.push({cost, u});
                }
            }
        }

        return reachable;
    }

public:
    int findTheCity(
        int n,
        vector<vector<int>>& edges,
        int distanceThreshold
    ) {

        // FIX 3
        vector<vector<pair<int, int>>> adj(n);

        for (auto x : edges) {
            int u = x[0];
            int v = x[1];
            int w = x[2];

            adj[u].push_back({v, w});
            adj[v].push_back({u, w});
        }

        priority_queue<
            pair<int, int>,
            vector<pair<int, int>>,
            greater<>
        > pq;

        for (int i = 0; i < n; i++) {
            pq.push({
                bfs(adj, i, distanceThreshold, n),
                i
            });
        }

        int ans = -1;
        int bestCount = INT_MAX;

        while (!pq.empty()) {

            auto [ct, node] = pq.top();
            pq.pop();

            if (ct < bestCount ||
                (ct == bestCount && node > ans)) {

                bestCount = ct;
                ans = node;
            }
        }

        return ans;
    }
};
