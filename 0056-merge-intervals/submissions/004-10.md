class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& v) {
        sort(v.begin(), v.end());

        vector<vector<int>> ans;
        ans.push_back(v[0]);

        for(int i=1; i<v.size();i++){
            auto cs=v[i][0];
            auto ce=v[i][1];

            auto mergeBack= ans.back()[1];
            if(cs<=mergeBack){
                ans.back()[1]=max(mergeBack, ce);
            }else{
                ans.push_back(v[i]);
            }
        }
        return ans;
    }
};
