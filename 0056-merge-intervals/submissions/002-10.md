class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& arr) {
        sort(arr.begin(), arr.end());

        vector<vector<int>>  ans;

        for(auto x:arr){
            if(ans.size()==0){
                ans.push_back(x);
                continue;
            }
            auto last= ans.back();
            if(last[1]>=x[0]){
                ans[ans.size()-1][1]=max(x[1], last[1]);
            }else{
                ans.push_back(x);
            }
        }
        return ans;
    }
};
