class Solution {
public:
    int minimumDeletions(vector<int>& nums) {
        if(nums.size()<=2){
            return nums.size();
        }
        int n=nums.size();
        int i1=0;
        int prevSmall=INT_MAX;
        int prevGrt=INT_MIN;
        int i2=0;
        for(int i=0; i<nums.size(); i++){
            if(nums[i]>prevGrt){
                prevGrt=nums[i];
                i2=i;// one based index
            };
            if(nums[i]<prevSmall){
                prevSmall=nums[i];
                i1=i;
            }
        }


        int l= min(i1, i2);
        int r=max(i1, i2);
        

        int left=r+1;
        int right=n-l;

        int mid=l+1 + n-r;

        return min({left, right, mid});


    }
};
