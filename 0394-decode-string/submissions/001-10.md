class Solution {
public:
    string decodeString(string s) {
        
        stack<string>st;

        for(int i=0; i<s.size(); i++){
            char c=s[i];
            // digit

            if(isdigit(c)){
                string nums="";
                while(i<s.size() && isdigit(s[i])){
                    nums+=s[i];
                    i++;
                }
                st.push(nums);
                i--; // compansate of already moved
            } else if(c==']'){
                string rept="";
                while(!st.empty() && st.top()!="["){
                    if(st.top()!="[")
                        rept=st.top()+rept;
                    st.pop();
                }
                // pop [
                st.pop();
                int repeatTimes=stoi(st.top());
                st.pop();// pop number 
                string expanded="";
                while(repeatTimes--){ // copy number oftime ..
                    expanded+=rept;
                }
                st.push(expanded);
            }else{
                string ct=""; //
                ct+=c;
                st.push(ct);
            }
        }
        string ans="";
        while(!st.empty()){
            ans=st.top()+ans;
            st.pop();
        }
        return ans;

    }
};
